typedef struct { MNode *root; int M; } MWay;     /* M = order: up to M-1 keys and M children per node */

/* search: returns 1 if found; *visited = number of nodes looked at (= "disk reads") */
static int mw_search(const MWay *t, int k, int *visited)
{
    int v = 0;
    for (const MNode *x = t->root; x; ) {
        v++;
        int i = 0;
        while (i < x->n && x->key[i] < k) i++;           /* first key >= k */
        if (i < x->n && x->key[i] == k) { if (visited) *visited = v; return 1; }
        x = x->ch[i];                                    /* the interval k falls in */
    }
    if (visited) *visited = v;
    return 0;
}

/* insert: returns 0 if k is already there */
static int mw_insert(MWay *t, int k)
{
    if (!t->root) { t->root = mn_new(); t->root->key[0] = k; t->root->n = 1; return 1; }
    MNode *x = t->root;
    for (;;) {
        int i = 0;
        while (i < x->n && x->key[i] < k) i++;
        if (i < x->n && x->key[i] == k) return 0;        /* duplicate */
        if (x->ch[i]) { x = x->ch[i]; continue; }        /* the interval already has a subtree: go in */
        if (x->n < t->M - 1) {                           /* room: the empty interval is split in two empty ones */
            for (int j = x->n; j > i; j--) { x->key[j] = x->key[j - 1]; x->ch[j + 1] = x->ch[j]; }
            x->key[i] = k; x->ch[i + 1] = NULL; x->n++;
        } else {                                         /* full: the new key starts a new child node */
            MNode *c = mn_new(); c->key[0] = k; c->n = 1; x->ch[i] = c;
        }
        return 1;
    }
}

/* delete: returns the (possibly new) root of the subtree; *done = 1 if the key was found */
static MNode *mw_del(MNode *x, int k, int *done)
{
    if (!x) return NULL;
    int i = 0;
    while (i < x->n && x->key[i] < k) i++;
    if (i == x->n || x->key[i] != k) { x->ch[i] = mw_del(x->ch[i], k, done); return x; }
    *done = 1;
    if (x->ch[i]) {                                      /* inner key: take the predecessor from the left subtree */
        const MNode *p = x->ch[i];
        while (p->ch[p->n]) p = p->ch[p->n];
        int pk = p->key[p->n - 1];
        x->key[i] = pk;
        int d2 = 0; x->ch[i] = mw_del(x->ch[i], pk, &d2); assert(d2);
        return x;
    }
    if (x->ch[i + 1]) {                                  /* left interval empty: take the successor instead */
        const MNode *s = x->ch[i + 1];
        while (s->ch[0]) s = s->ch[0];
        int sk = s->key[0];
        x->key[i] = sk;
        int d2 = 0; x->ch[i + 1] = mw_del(x->ch[i + 1], sk, &d2); assert(d2);
        return x;
    }
    for (int j = i; j < x->n - 1; j++) x->key[j] = x->key[j + 1];     /* both neighbours empty: just remove */
    for (int j = i + 1; j < x->n; j++) x->ch[j] = x->ch[j + 1];
    x->ch[x->n] = NULL; x->n--;
    if (x->n == 0) { MNode *only = x->ch[0]; free(x); return only; } /* splice out an empty node */
    return x;
}
static int mw_delete(MWay *t, int k) { int d = 0; t->root = mw_del(t->root, k, &d); return d; }

/* validity: every node has 1..M-1 sorted keys, every key of ch[i] lies strictly between key[i-1] and key[i] */
static int mw_valid_(const MNode *x, int M, long lo, long hi)
{
    if (!x) return 1;
    if (x->n < 1 || x->n > M - 1) return 0;
    for (int i = 0; i < x->n; i++) {
        if (x->key[i] <= lo || x->key[i] >= hi) return 0;
        if (i && x->key[i] <= x->key[i - 1]) return 0;
    }
    for (int i = 0; i <= x->n; i++)
        if (!mw_valid_(x->ch[i], M, i ? x->key[i - 1] : lo, i < x->n ? x->key[i] : hi)) return 0;
    for (int i = x->n + 1; i <= MW_MAXM + 1; i++) if (x->ch[i]) return 0;
    return 1;
}

static int bt_min_keys(int m) { return (m + 1) / 2 - 1; }          /* ceil(m/2) - 1 */
static long bt_reads = 0;                                           /* nodes touched by the last operations */

static int bt_search(const BTree *t, int k)
{
    for (const MNode *x = t->root; x; ) {
        bt_reads++;
        int i = 0;
        while (i < x->n && x->key[i] < k) i++;
        if (i < x->n && x->key[i] == k) return 1;
        x = x->ch[i];                                   /* NULL at a leaf: not found */
    }
    return 0;
}

/* ---- insert ---- */
/* returns 0 = duplicate, 1 = inserted, 2 = inserted and x was split: *up (median) and *right (new sibling) */
static int bt_ins(MNode *x, int m, int k, int *up, MNode **right)
{
    bt_reads++;
    int i = 0;
    while (i < x->n && x->key[i] < k) i++;
    if (i < x->n && x->key[i] == k) return 0;
    MNode *newchild = NULL;
    if (x->ch[0]) {                                     /* inner node: insert below, maybe receive a split */
        int mk; MNode *mr = NULL;
        int r = bt_ins(x->ch[i], m, k, &mk, &mr);
        if (r != 2) return r;
        k = mk; newchild = mr;                          /* the median of the child goes into THIS node */
    }
    for (int j = x->n; j > i; j--) { x->key[j] = x->key[j - 1]; x->ch[j + 1] = x->ch[j]; }
    x->key[i] = k; x->ch[i + 1] = newchild; x->n++;
    if (x->n < m) return 1;                             /* at most m-1 keys: fits */

    int mid = (x->n - 1) / 2;                           /* overflow: m keys -> split at the median */
    MNode *r = mn_new();
    r->n = x->n - mid - 1;
    for (int j = 0; j < r->n; j++) r->key[j] = x->key[mid + 1 + j];
    for (int j = 0; j <= r->n; j++) { r->ch[j] = x->ch[mid + 1 + j]; x->ch[mid + 1 + j] = NULL; }
    *up = x->key[mid]; *right = r;
    x->n = mid;
    return 2;
}
static int bt_insert(BTree *t, int k)
{
    if (!t->root) { t->root = mn_new(); t->root->key[0] = k; t->root->n = 1; return 1; }
    int up; MNode *r = NULL;
    int res = bt_ins(t->root, t->m, k, &up, &r);
    if (res == 2) {                                     /* the root split: the tree grows by one level */
        MNode *nr = mn_new();
        nr->n = 1; nr->key[0] = up; nr->ch[0] = t->root; nr->ch[1] = r;
        t->root = nr;
    }
    return res != 0;
}

        for (int j = c->n; j > 0; j--) c->key[j] = c->key[j - 1];
        for (int j = c->n + 1; j > 0; j--) c->ch[j] = c->ch[j - 1];
        c->key[0] = x->key[i - 1];                      /* separator comes down */
        c->ch[0] = L->ch[L->n];
        x->key[i - 1] = L->key[L->n - 1];               /* largest key of the sibling goes up */
        L->ch[L->n] = NULL; L->n--; c->n++;
        bt_note("borrow-left");
    } else if (R && R->n > mn) {                        /* borrow from the right sibling */
        c->key[c->n] = x->key[i];
        c->ch[c->n + 1] = R->ch[0];
        x->key[i] = R->key[0];
        for (int j = 0; j < R->n - 1; j++) R->key[j] = R->key[j + 1];
        for (int j = 0; j < R->n; j++) R->ch[j] = R->ch[j + 1];
        R->ch[R->n] = NULL; R->n--; c->n++;
        bt_note("borrow-right");
    } else {                                            /* merge: left + separator + this (or this + separator + right) */
        int left = L && !(bt_merge_right_first && R);   /* which neighbour do we combine with? */
        int li = left ? i - 1 : i;                      /* index of the left node of the pair */
        MNode *a = x->ch[li], *b = x->ch[li + 1];
        a->key[a->n] = x->key[li];
        for (int j = 0; j < b->n; j++) a->key[a->n + 1 + j] = b->key[j];
        for (int j = 0; j <= b->n; j++) a->ch[a->n + 1 + j] = b->ch[j];
        a->n += 1 + b->n;
        free(b);
        for (int j = li; j < x->n - 1; j++) x->key[j] = x->key[j + 1];
        for (int j = li + 1; j < x->n; j++) x->ch[j] = x->ch[j + 1];
        x->ch[x->n] = NULL; x->n--;
        bt_note(left ? "merge-left" : "merge-right");
    }
}
static int bt_del(MNode *x, int m, int k)
{
    bt_reads++;
    int i = 0;
    while (i < x->n && x->key[i] < k) i++;
    int found = i < x->n && x->key[i] == k;
    if (!x->ch[0]) {                                    /* leaf */
        if (!found) return 0;
        for (int j = i; j < x->n - 1; j++) x->key[j] = x->key[j + 1];
        x->n--;
        return 1;
    }
    if (found && bt_use_pred) {                         /* inner key: replace by the in-order predecessor (a leaf key) */
        const MNode *p = x->ch[i];
        while (p->ch[0]) p = p->ch[p->n];
        int pk = p->key[p->n - 1];
        x->key[i] = pk;
        int r = bt_del(x->ch[i], m, pk); assert(r);
        bt_fix(x, i, m);
        return 1;
    }
    if (found) {                                        /* inner key: replace by the in-order successor (a leaf key) */
        const MNode *s = x->ch[i + 1];
        while (s->ch[0]) s = s->ch[0];
        int sk = s->key[0];
        x->key[i] = sk;
        int r = bt_del(x->ch[i + 1], m, sk); assert(r);
        bt_fix(x, i + 1, m);
        return 1;
    }
    int r = bt_del(x->ch[i], m, k);
    if (r) bt_fix(x, i, m);
    return r;
}
static int bt_delete(BTree *t, int k)
{
    bt_fixlog[0] = 0;
    if (!t->root) return 0;
    int r = bt_del(t->root, t->m, k);
    if (t->root->n == 0) {                              /* the root lost its last key: the tree shrinks */
        MNode *old = t->root;
        t->root = old->ch[0];                           /* NULL when the whole tree is now empty */
        free(old);
    }
    return r;
}

/* ---- validity: the four B-tree rules + search-tree order.  Returns the height or -1 ---- */
static int bt_valid_(const MNode *x, int m, int is_root, long lo, long hi)
{
    if (x->n < (is_root ? 1 : bt_min_keys(m)) || x->n > m - 1) return -1;
    for (int i = 0; i < x->n; i++) {
        if (x->key[i] <= lo || x->key[i] >= hi) return -1;
        if (i && x->key[i] <= x->key[i - 1]) return -1;
    }
    if (mn_leaf(x)) return 1;
    int h = -2;
    for (int i = 0; i <= x->n; i++) {
        if (!x->ch[i]) return -1;                       /* inner node: all n+1 children must exist */
        int c = bt_valid_(x->ch[i], m, 0, i ? x->key[i - 1] : lo, i < x->n ? x->key[i] : hi);
        if (c < 0) return -1;
        if (h == -2) h = c; else if (h != c) return -1; /* all leaves on one level */
    }
    return h + 1;
}
static int bt_valid(const BTree *t)
{
    if (!t->root) return 1;
    if (t->root->n < 1) return 0;
    return bt_valid_(t->root, t->m, 1, -2147483647L - 1, 2147483647L) > 0;
}

الفرق في الإدخال: تقسيم الورقة نسخ للمفتاح الأول من اليمين للأب (لأنه لازم يفضل في الورقة)، لكن تقسيم عقدة داخلية نقل زي B-tree. وفي الحذف: المفتاح بيتشال من الورقة بس؛ المفتاح في الفهرس ممكن يفضل (يفضل صالح للتوجيه).

typedef struct { MNode *root; int m; } BPlus;

static long bp_reads = 0;
static int bp_upper(const MNode *x, int k) { int i = 0; while (i < x->n && x->key[i] <= k) i++; return i; }  /* number of keys <= k */

static const MNode *bp_find_leaf(const BPlus *t, int k)
{
    const MNode *x = t->root;
    while (x && !mn_leaf(x)) { bp_reads++; x = x->ch[bp_upper(x, k)]; }
    if (x) bp_reads++;
    return x;
}
static int bp_search(const BPlus *t, int k)
{
    const MNode *l = bp_find_leaf(t, k);
    if (!l) return 0;
    for (int i = 0; i < l->n; i++) if (l->key[i] == k) return 1;
    return 0;
}
/* all keys in [lo, hi] by walking the leaf chain; returns how many */
static int bp_range(const BPlus *t, int lo, int hi, int *out)
{
    int c = 0;
    const MNode *first = bp_find_leaf(t, lo);
    for (const MNode *l = first; l; l = l->next) {
        if (l != first) bp_reads++;                          /* following `next` = one more leaf read */
        for (int i = 0; i < l->n; i++) {
            if (l->key[i] > hi) return c;
            if (l->key[i] >= lo) out[c++] = l->key[i];
        }
    }
    return c;
}

/* ---- insert ---- */
static int bp_ins(MNode *x, int m, int k, int *up, MNode **right)
{
    int i = bp_upper(x, k);
    if (mn_leaf(x)) {
        for (int j = 0; j < x->n; j++) if (x->key[j] == k) return 0;
        for (int j = x->n; j > i; j--) x->key[j] = x->key[j - 1];
        x->key[i] = k; x->n++;
        if (x->n < m) return 1;
        int keep = x->n / 2;                                 /* leaf split: left keeps n/2, right gets the rest */
        MNode *r = mn_new();
        r->n = x->n - keep;
        for (int j = 0; j < r->n; j++) r->key[j] = x->key[keep + j];
        x->n = keep;
        r->next = x->next; x->next = r;                      /* keep the chain */
        *up = r->key[0]; *right = r;                         /* COPY the first key of the right leaf up */
        return 2;
    }
    int mk; MNode *mr = NULL;
    int res = bp_ins(x->ch[i], m, k, &mk, &mr);
    if (res != 2) return res;
    for (int j = x->n; j > i; j--) { x->key[j] = x->key[j - 1]; x->ch[j + 1] = x->ch[j]; }
    x->key[i] = mk; x->ch[i + 1] = mr; x->n++;
    if (x->n < m) return 1;
    int mid = (x->n - 1) / 2;                                /* index split: the middle key MOVES up */
    MNode *r = mn_new();
    r->n = x->n - mid - 1;
    for (int j = 0; j < r->n; j++) r->key[j] = x->key[mid + 1 + j];
    for (int j = 0; j <= r->n; j++) { r->ch[j] = x->ch[mid + 1 + j]; x->ch[mid + 1 + j] = NULL; }
    *up = x->key[mid]; *right = r; x->n = mid;
    return 2;
}
static int bp_insert(BPlus *t, int k)
{
    if (!t->root) { t->root = mn_new(); t->root->key[0] = k; t->root->n = 1; return 1; }
    int up; MNode *r = NULL;
    int res = bp_ins(t->root, t->m, k, &up, &r);
    if (res == 2) {
        MNode *nr = mn_new();
        nr->n = 1; nr->key[0] = up; nr->ch[0] = t->root; nr->ch[1] = r;
        t->root = nr;                                        /* the tree grows by one level */
    }
    return res != 0;
}

/* ---- delete ---- */
static char bp_log[128];
static void bp_note(const char *s) { if (bp_log[0]) strcat(bp_log, " "); strcat(bp_log, s); }
static void bp_drop(MNode *x, int ci)                        /* remove child ci and one adjacent separator from x */
{
    int ki = ci > 0 ? ci - 1 : 0;
    for (int j = ki; j < x->n - 1; j++) x->key[j] = x->key[j + 1];
    for (int j = ci; j < x->n; j++) x->ch[j] = x->ch[j + 1];
    x->ch[x->n] = NULL; x->n--;
}
static void bp_unlink(BPlus *t, MNode *leaf)                 /* take a leaf out of the chain (walk from the leftmost leaf) */
{
    MNode *l = t->root;
    while (l && !mn_leaf(l)) l = l->ch[0];
    if (l == leaf) return;                                   /* the leftmost leaf has no predecessor */
    while (l && l->next != leaf) l = l->next;
    if (l) l->next = leaf->next;
}
static int bp_del(BPlus *t, MNode *x, int k)
{
    if (mn_leaf(x)) {
        for (int i = 0; i < x->n; i++) if (x->key[i] == k) {
            for (int j = i; j < x->n - 1; j++) x->key[j] = x->key[j + 1];
            x->n--; return 1;
        }
        return 0;
    }
    int i = bp_upper(x, k);
    MNode *c = x->ch[i];
    int r = bp_del(t, c, k);
    if (!r || c->n > 0) return r;
    if (mn_leaf(c)) {                                        /* an empty leaf: remove it and its separator */
        bp_unlink(t, c); free(c); bp_drop(x, i); bp_note("drop-leaf");
        return 1;
    }
    /* c is an index node with no key left (one child): merge with a sibling, or borrow when the merge would not fit */
    int m = t->m;
    MNode *L = i > 0 ? x->ch[i - 1] : NULL, *R = i < x->n ? x->ch[i + 1] : NULL;
    if (L && L->n + 1 <= m - 1) {                            /* merge into the left sibling: separator comes down */
        L->key[L->n] = x->key[i - 1]; L->ch[L->n + 1] = c->ch[0]; L->n++;
        free(c); bp_drop(x, i); bp_note("merge-left");
    } else if (!L && R && R->n + 1 <= m - 1) {               /* merge the right sibling into c */
        c->key[0] = x->key[i];
        for (int j = 0; j < R->n; j++) c->key[1 + j] = R->key[j];
        for (int j = 0; j <= R->n; j++) c->ch[1 + j] = R->ch[j];
        c->n = 1 + R->n;
        free(R); bp_drop(x, i + 1);                          /* removes ch[i+1] and key[i] */
        bp_note("merge-right");
    } else if (L) {                                          /* borrow the last child of L through the parent */
        c->ch[1] = c->ch[0]; c->ch[0] = L->ch[L->n];
        c->key[0] = x->key[i - 1]; c->n = 1;
        x->key[i - 1] = L->key[L->n - 1];
        L->ch[L->n] = NULL; L->n--; bp_note("borrow-left");
    } else {                                                 /* borrow the first child of R */
        c->key[0] = x->key[i]; c->ch[1] = R->ch[0]; c->n = 1;
        x->key[i] = R->key[0];
        for (int j = 0; j < R->n - 1; j++) R->key[j] = R->key[j + 1];
        for (int j = 0; j < R->n; j++) R->ch[j] = R->ch[j + 1];
        R->ch[R->n] = NULL; R->n--; bp_note("borrow-right");
    }
    return 1;
}
static int bp_delete(BPlus *t, int k)
{
    bp_log[0] = 0;
    if (!t->root) return 0;
    int r = bp_del(t, t->root, k);
    if (t->root->n == 0) {                                   /* empty root: an index root with one child collapses, an empty leaf means an empty tree */
        MNode *old = t->root;
        t->root = mn_leaf(old) ? NULL : old->ch[0];
        free(old);
    }
    return r;
}

/* ---- relink the leaf chain of a tree built by hand (mn_parse) ---- */
static void bp_link_(MNode *x, MNode **prev)
{
    if (mn_leaf(x)) { if (*prev) (*prev)->next = x; *prev = x; return; }
    for (int i = 0; i <= x->n; i++) bp_link_(x->ch[i], prev);
}
static void bp_relink(MNode *root) { MNode *prev = NULL; if (root) bp_link_(root, &prev); }

/* ---- validity ---- */
static int bp_valid_(const MNode *x, int m, int is_root, long lo, long hi, int depth, int *leafdepth, const MNode **prevleaf)
{
    if (x->n < 1 || x->n > m - 1) return 0;
    (void)is_root;
    for (int i = 0; i < x->n; i++) {
        if (x->key[i] < lo || x->key[i] >= hi) return 0;     /* leaf keys: lo <= key < hi (separator copies are lo) */
        if (i && x->key[i] <= x->key[i - 1]) return 0;
    }
    if (mn_leaf(x)) {
        if (*leafdepth < 0) *leafdepth = depth; else if (*leafdepth != depth) return 0;
        if (*prevleaf && (*prevleaf)->next != x) return 0;   /* the chain must visit the leaves in order */
        if (*prevleaf && (*prevleaf)->key[(*prevleaf)->n - 1] >= x->key[0]) return 0;
        *prevleaf = x;
        return 1;
    }
    for (int i = 0; i <= x->n; i++) {
        if (!x->ch[i]) return 0;
        if (!bp_valid_(x->ch[i], m, 0, i ? x->key[i - 1] : lo, i < x->n ? x->key[i] : hi, depth + 1, leafdepth, prevleaf)) return 0;
    }
    return 1;
}
static int bp_valid(const BPlus *t)
{
    if (!t->root) return 1;
    int ld = -1; const MNode *pl = NULL;
    if (!bp_valid_(t->root, t->m, 1, -2147483647L - 1, 2147483647L, 0, &ld, &pl)) return 0;
    return pl == NULL || pl->next == NULL;                   /* the last leaf ends the chain */
}

static int tt_search(const MNode *root, int x, char *path, size_t cap)
{
    path[0] = 0;
    for (const MNode *n = root; n; ) {
        size_t len = strlen(path);
        snprintf(path + len, cap - len, "%s[", len ? " " : "");
        for (int i = 0; i < n->n; i++) { len = strlen(path); snprintf(path + len, cap - len, i ? " %d" : "%d", n->key[i]); }
        len = strlen(path); snprintf(path + len, cap - len, "]");

}

Weiss بيعرّف B-tree بمعاملين: M للعقد الداخلية (⌈M/2⌉..M ابن، مفتاح i = أصغر مفتاح في الفرع i+1) وL للأوراق (⌈L/2⌉..L عنصر)، والبيانات في الأوراق بس (وهو بنفسه بيعلّق في الحاشية: «what is described is popularly known as a B+ tree»). الأرقام: Fig 4.63 فيها 16 ورقة و59 عنصر. الخطوات:

#define WMAX 40
typedef struct WN {
    int leaf;
    int n;                       /* leaf: number of items   inner: number of CHILDREN */
    int key[WMAX + 2];           /* leaf: the items          inner: key[i] = smallest key of subtree i+1  (n-1 keys) */
    struct WN *ch[WMAX + 2];
} WN;
typedef struct { WN *root; int M, L; int adopt; } WTree;

static WN *wn_new(int leaf) { WN *x = calloc(1, sizeof *x); if (!x) exit(1); x->leaf = leaf; return x; }
static void wn_free(WN *x) { if (!x) return; if (!x->leaf) for (int i = 0; i < x->n; i++) wn_free(x->ch[i]); free(x); }
static int w_min_leaf(const WTree *t) { return (t->L + 1) / 2; }
static int w_min_children(const WTree *t) { return (t->M + 1) / 2; }
static long w_reads = 0;

static int w_child_index(const WN *x, int k) { int i = 0; while (i < x->n - 1 && x->key[i] <= k) i++; return i; }
static int w_search(const WTree *t, int k)
{
    const WN *x = t->root;
    while (x && !x->leaf) { w_reads++; x = x->ch[w_child_index(x, k)]; }
    if (!x) return 0;
    w_reads++;
    for (int i = 0; i < x->n; i++) if (x->key[i] == k) return 1;
    return 0;
}

/* ---- insert ---- */
/* returns 0 dup, 1 ok, 2 split (*sep = smallest key of *right) */
static int w_ins(WTree *t, WN *x, int k, int *sep, WN **right, WN *left_sib, WN *right_sib, WN *parent, int pi)
{
    (void)left_sib; (void)right_sib;
    if (x->leaf) {
        int i = 0; while (i < x->n && x->key[i] < k) i++;
        if (i < x->n && x->key[i] == k) return 0;
        if (x->n == t->L && t->adopt && parent) {                 /* Exercise 4.41(c): look for a sibling with room before splitting */
            WN *L = pi > 0 ? parent->ch[pi - 1] : NULL, *R = pi < parent->n - 1 ? parent->ch[pi + 1] : NULL;
            if (L && L->n < t->L) {                               /* give our smallest item to the left sibling */
                int tmp[WMAX + 2], c = 0;
                for (int j = 0; j < i; j++) tmp[c++] = x->key[j];
                tmp[c++] = k;
                for (int j = i; j < x->n; j++) tmp[c++] = x->key[j];
                L->key[L->n++] = tmp[0];
                x->n = 0; for (int j = 1; j < c; j++) x->key[x->n++] = tmp[j];
                parent->key[pi - 1] = x->key[0];
                return 1;
            }
            if (R && R->n < t->L) {                               /* give our largest item to the right sibling */
                int tmp[WMAX + 2], c = 0;
                for (int j = 0; j < i; j++) tmp[c++] = x->key[j];
                tmp[c++] = k;
                for (int j = i; j < x->n; j++) tmp[c++] = x->key[j];
                for (int j = R->n; j > 0; j--) R->key[j] = R->key[j - 1];
                R->key[0] = tmp[c - 1]; R->n++;
                x->n = 0; for (int j = 0; j < c - 1; j++) x->key[x->n++] = tmp[j];
                parent->key[pi] = R->key[0];
                return 1;
            }
        }
        for (int j = x->n; j > i; j--) x->key[j] = x->key[j - 1];
        x->key[i] = k; x->n++;
        if (x->n <= t->L) return 1;
        int left = x->n / 2;                                      /* L+1 items -> floor((L+1)/2) and the rest */
        WN *r = wn_new(1);
        for (int j = left; j < x->n; j++) r->key[r->n++] = x->key[j];
        x->n = left;
        *sep = r->key[0]; *right = r;
        return 2;
    }
    int i = w_child_index(x, k);
    int s; WN *r = NULL;
    int res = w_ins(t, x->ch[i], k, &s, &r, NULL, NULL, x, i);
    if (res != 2) return res;
    for (int j = x->n - 1; j > i; j--) x->key[j] = x->key[j - 1];
    for (int j = x->n; j > i + 1; j--) x->ch[j] = x->ch[j - 1];
    x->key[i] = s; x->ch[i + 1] = r; x->n++;
    if (x->n <= t->M) return 1;
    int left = x->n / 2;                                          /* M+1 children -> floor((M+1)/2) and the rest */
    WN *nr = wn_new(0);
    nr->n = x->n - left;
    for (int j = 0; j < nr->n; j++) nr->ch[j] = x->ch[left + j];
    for (int j = 0; j < nr->n - 1; j++) nr->key[j] = x->key[left + j];
    *sep = x->key[left - 1];                                      /* the key between the halves goes up */
    for (int j = left; j < x->n; j++) x->ch[j] = NULL;
    x->n = left;
    *right = nr;
    return 2;
}
static int w_insert(WTree *t, int k)
{
    if (!t->root) { t->root = wn_new(1); t->root->key[0] = k; t->root->n = 1; return 1; }
    int s; WN *r = NULL;
