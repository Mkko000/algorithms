الإجابات

1) Overflow: push على stack ممتلئ (TOP == MAX−1). Underflow: pop (أو peek) من stack فاضي (TOP == −1). في الـ linked stack مفيش overflow غير لو malloc فشل.

2) الـ array: وصول لأي عنصر بالـ index وإدخال/حذف في أي مكان. الـ stack: واجهة مقيّدة (LIFO) تتعامل بس مع القمة؛ ممكن تتنفّذ فوق array أو list.

3) Array stack: حجم ثابت (overflow)، ذاكرة متصلة، من غير pointers. Linked stack: حجم ديناميكي، كل عقدة فيها pointer زيادة، وpush/pop = إدخال/حذف عند head (O(1)).

4) peek بترجّع القمة من غير ما تشيلها، وpop بترجّعها وتشيلها.

5) لأن ترتيب المعاملات والعمليات في postfix/prefix بيحدد ترتيب التنفيذ بالكامل: كل عملية بتشتغل على آخر معاملين جاهزين (postfix) أو التاليين ليها (prefix)، فمفيش لبس يحتاج أقواس أو أولويات.

6) أي algorithm recursive ممكن نكتبه من غير recursion بـ stack صريح بنخزّن فيه اللي الـ call stack كان هيخزّنه (arguments، حالة، عنوان الاستكمال). أمثلة محلولة بالكود: factorial وfib وHanoi وAckermann في recursion_stack.c، والـ traversals غير الـ recursive للأشجار في الدرس الجاي.

7) Multiple stack = أكتر من stack في نفس الـ array (مثلًا واحد من كل طرف). مفيد لأنه بيستغل المساحة كلها: الـ overflow بس لما المصفوفة كلها تتملي، مش لما نصّها يتملي (راجع two_stacks).

15) iterative = حلقة (O(1) مساحة stack، أسرع عادةً). recursive = دالة بتنادي نفسها (كود أقصر وأقرب للتعريف، بس بتكلّف frame لكل نداء وممكن stack overflow). نفضّل الـ recursive لو المشكلة تعريفها recursive طبيعيًا والعمق صغير (أشجار، divide and conquer، backtracking)، والـ iterative لو فيه حلقة واضحة (factorial، gcd، fib)، وخصوصًا لو العمق ممكن يوصل آلاف. مثال: fib recursive أُسّي مقابل O(n) بحلقة.

16) أبراج هانوي: 3 قضبان وn أقراص بأحجام مختلفة على الأول؛ انقلهم للتالت بقرص واحد في المرة، من غير كبير فوق صغير. الحل recursive: انقل n−1 لـ B، الأكبر لـ C، n−1 من B لـ C. الأدنى 2ⁿ − 1 حركة. (الشرح والصورة والكود في الدرس، والبرنامج بيتأكد إن الـ explicit-stack والـ recursive بيطلّعوا نفس الحركات لكل n حتى 10.)

ف7 RQ 8 و9: infix → postfix و prefix

نص السؤالين في النسخة اللي عندي بيقولوا الاتنين «postfix»، فحليت الشكلين لكل تعبير وأي واحد المقصود تلاقيه.

/* ---- RQ 8 and 9: infix -> postfix and infix -> prefix (the OCR of the book words both as "postfix";
        we give BOTH forms for every expression, so whichever the question meant is answered) ---- */
static void conv(const char *infix, const char *post_want, const char *pre_want)
{
    toks in, pf, pr; char b1[256], b2[256];
    assert(tokenize(infix, &in) > 0);
    assert(to_postfix(&in, &pf) > 0 && to_prefix_reverse(&in, &pr) > 0);
    join(&pf, b1, sizeof b1); join(&pr, b2, sizeof b2);
    printf("  %-30s postfix: %-26s prefix: %s\n", infix, b1, b2);
    assert(strcmp(b1, post_want) == 0 && strcmp(b2, pre_want) == 0);
}
static void rq8_rq9(void)
{
    conv("A - B + C",                   "A B - C +",                     "+ - A B C");
    conv("A * B + C / D",               "A B * C D / +",                 "+ * A B / C D");
    conv("(A - B) + C * D / E - C",     "A B - C D * E / + C -",         "- + - A B / * C D E C");
    conv("(A * B) + (C / D) - (D + E)", "A B * C D / + D E + -",         "- + * A B / C D + D E");
    conv("((A - B) + D / ((E + F) * G))", "A B - D E F + G * / +",       "+ - A B / D * + E F G");
    conv("(A - 2 * (B + C) / D * E) + F", "A 2 B C + * D / E * - F +",   "+ - A * / * 2 + B C D E F");
    conv("14 / 7 * 3 - 4 + 9 / 2",      "14 7 / 3 * 4 - 9 2 / +",        "+ - * / 14 7 3 4 / 9 2");
}
بنفكّر إزايفي (f) و(g) القسمة والضرب نفس الأولوية وشماليين: 2*(B+C)/D*E = ((2*(B+C))/D)*E فالـ postfix 2 B C + * D / E *. وفي (g): 14/7*3 = (14/7)*3 = 6، فالـ postfix 14 7 / 3 * .... وقيمة (g) = 6 − 4 + 4.5 = 6.5.

ف7 RQ 10 و11: postfix/prefix → infix

/* ---- RQ 10 and 11: postfix / prefix -> infix.  Scan, push strings, and each operator glues "(x op y)". ---- */
static void to_infix(const char *expr, int prefix, const char *want)
{
    toks t; assert(tokenize(expr, &t) > 0);
    char *r = convert(&t, prefix ? 2 : 1);
    printf("  %-18s -> %s\n", expr, r);
    assert(r && strcmp(r, want) == 0);
    free(r);
}
static void rq10_rq11(void)
{
    puts("RQ10 (postfix -> infix):");
    to_infix("A B + C * D -", 0, "(((A + B) * C) - D)");
    to_infix("A B C * + D -", 0, "((A + (B * C)) - D)");
    puts("RQ11 (prefix -> infix):");
    to_infix("* - + A B C D", 1, "(((A + B) - C) * D)");
    to_infix("+ - a * B C D", 1, "((a - (B * C)) + D)");
}

ف7 RQ 12: حوّل وقيّم 10 + ((7 − 5) + 10) / 2

/* ---- RQ 12: 10 + ((7 - 5) + 10) / 2   ->  postfix, then evaluate with a visible stack ---- */
static double eval_trace(const toks *pf)
{
    double st[MAXT]; int sp = 0;
    for (int i = 0; i < pf->n; i++) {
        const char *t = pf->t[i];
        if (!is_op(t)) st[sp++] = atof(t);
        else { double y = st[--sp], x = st[--sp]; st[sp++] = apply(t[0], x, y); }
        printf("    read %-3s stack: [", t);
        for (int k = 0; k < sp; k++) printf("%s%g", k ? " " : "", st[k]);
        puts("]");
    }
    assert(sp == 1);
    return st[0];
}
static void rq12(void)
{
    toks in, pf; char buf[128];
    tokenize("10 + ((7 - 5) + 10) / 2", &in); to_postfix(&in, &pf);
    join(&pf, buf, sizeof buf);
    printf("  postfix: %s\n", buf);
    assert(strcmp(buf, "10 7 5 - 10 + 2 / +") == 0);
    double v = eval_trace(&pf);
    printf("  value = %g\n", v);
    assert(v == 16);
}
الحل

postfix: 10 7 5 − 10 + 2 / + والقيمة 16 (7−5=2، 2+10=12، 12/2=6، 10+6=16). برنامج التقييم المطلوب هو eval_postfix في الدرس (وأعلاه نسخة بتطبع الـ stack بعد كل توكن).

ف7 RQ 13 و PE 5: نسخ ومقارنة stacks

typedef struct { int a[CAP]; int n; } istack;
static void is_push(istack *s, int x) { assert(s->n < CAP); s->a[s->n++] = x; }
static int  is_pop(istack *s)          { assert(s->n > 0);   return s->a[--s->n]; }
/* copy preserving order: pop into a temporary (reverses), pop back into both (restores) */
static void is_copy(istack *src, istack *dst)
{
    istack tmp = {{0}, 0};
    while (src->n) is_push(&tmp, is_pop(src));
    while (tmp.n) { int x = is_pop(&tmp); is_push(src, x); is_push(dst, x); }
}
/* compare two stacks: they are equal iff the pops agree one by one (both are restored) */
static int is_equal(istack *a, istack *b)
{
    if (a->n != b->n) return 0;
    istack ta = {{0}, 0}, tb = {{0}, 0}; int eq = 1;
    while (a->n) { int x = is_pop(a), y = is_pop(b); is_push(&ta, x); is_push(&tb, y); if (x != y) eq = 0; }
    while (ta.n) is_push(a, is_pop(&ta));
    while (tb.n) is_push(b, is_pop(&tb));
    return eq;
}
/* reverse a stack using only push/pop and two temporary stacks (three moves = identity on order twice, reverse once) */
static void is_reverse(istack *s)
{
    istack t1 = {{0}, 0}, t2 = {{0}, 0};
    while (s->n) is_push(&t1, is_pop(s));
    while (t1.n) is_push(&t2, is_pop(&t1));
    while (t2.n) is_push(s, is_pop(&t2));
}
بنفكّر إزايالـ pop بيعكس. لو نسخنا stack لـ temp بالـ pop هنحصل على معكوس؛ فنحتاج مرور تاني من temp لـ (الأصل + النسخة) فيرجع الترتيب. نفس الفكرة في المقارنة: نشيل الاتنين مع بعض ونقارن، ونحفظ اللي شلناه عشان نرجّعه.

ف7 RQ 14: الـ stack بعد كل عملية

/* ---- RQ 14: draw the stack after each operation ---- */
static void show_stack(const char *what, const char *st, int n)
{
    printf("  %-22s [", what);
    for (int i = 0; i < n; i++) printf("%s%c", i ? " " : "", st[i]);
    printf("]%s\n", n ? "  <- top" : "");
}
static void rq14(void)
{
    char st[16]; int n = 0;
    for (char c = 'A'; c <= 'F'; c++) st[n++] = c;
    show_stack("(a) add A..F", st, n);          assert(n == 6 && st[5] == 'F');
    n -= 2;  show_stack("(b) delete two letters", st, n);   assert(n == 4 && st[3] == 'D');   /* F and E leave */
    st[n++] = 'G'; show_stack("(c) add G", st, n);          assert(n == 5 && st[4] == 'G');
    st[n++] = 'H'; show_stack("(d) add H", st, n);          assert(n == 6 && st[5] == 'H');
    n -= 4;  show_stack("(e) delete four letters", st, n);  assert(n == 2 && st[1] == 'B');   /* H G D C leave */
    st[n++] = 'I'; show_stack("(f) add I", st, n);          assert(n == 3 && strncmp(st, "ABI", 3) == 0);
}

ف7 PE 1: stack بـ linked list، PE 2 و3 و4

/* ---- PE 1: a stack as a linked list: push = insert at head, pop = delete at head, both O(1) ---- */
typedef struct snode { int val; struct snode *next; } snode;
static void ls_push(snode **top, int v)
{
    snode *n = malloc(sizeof *n);
    if (!n) { perror("malloc"); exit(1); }
    n->val = v; n->next = *top; *top = n;
}
static int ls_pop(snode **top, int *v)
{
    if (!*top) return 0;                          /* underflow */
    snode *t = *top; *v = t->val; *top = t->next; free(t); return 1;
}
/* ---- PE 2 and 3: "a+b" -> "ab+" and "a+b" -> "+ab" (single-letter operands, no spaces) ----
   the general engine handles any expression; here we strip the spaces so the result looks like the book's */
static void nospace(const toks *t, char *out)
{
    out[0] = 0;
    for (int i = 0; i < t->n; i++) strcat(out, t->t[i]);
}
static void pe2_pe3(void)
{
    toks in, pf, pr; char b[64];
    tokenize("a+b", &in);
    to_postfix(&in, &pf); nospace(&pf, b); printf("  a+b -> %s\n", b); assert(strcmp(b, "ab+") == 0);
    to_prefix_reverse(&in, &pr); nospace(&pr, b); printf("  a+b -> %s\n", b); assert(strcmp(b, "+ab") == 0);
}
/* ---- PE 4: a stack of student names: the element type is a string (copied in, so the caller's buffer may change) ---- */
#define NAMELEN 32
typedef struct { char name[CAP][NAMELEN]; int top; } name_stack;
static int ns_push(name_stack *s, const char *nm)
{
    if (s->top == CAP) return 0;
    size_t k = strlen(nm); if (k > NAMELEN - 1) k = NAMELEN - 1;      /* truncate long names, never overflow */
    memcpy(s->name[s->top], nm, k); s->name[s->top][k] = 0; s->top++; return 1;
}
static const char *ns_pop(name_stack *s) { return s->top ? s->name[--s->top] : NULL; }   /* valid until the next push */

ف7 PE 6–10: التعريفات الـ recursive

/* ---- PE 6: F(x,y) = F(x-y, y) + 1 if y <= x, else 0.  Each call subtracts y and adds 1 -> F = floor(x / y).
        (y = 0 would never terminate, so the function guards it.) ---- */
static int F6(int x, int y) { assert(y > 0); return y <= x ? F6(x - y, y) + 1 : 0; }
/* ---- PE 7: F(n,r) = F(n-1,r) + F(n-1,r-1).  The book gives no base cases; the standard ones
        F(n,0) = F(n,n) = 1 turn it into Pascal's rule, i.e. the binomial coefficient C(n,r). ---- */
static long F7(int n, int r) { return (r == 0 || r == n) ? 1 : F7(n - 1, r) + F7(n - 1, r - 1); }
/* ---- PE 8: Lambda(n) = Lambda(n/2) + 1 for n > 1, Lambda(1) = 0  ->  floor(log2 n) ---- */
static int Lambda(int n) { return n > 1 ? Lambda(n / 2) + 1 : 0; }
/* ---- PE 9: F(M,N) = 1 if M = 0 or M >= N >= 1, else F(M-1,N) + F(M-1,N-1) ---- */
static long F9(int M, int N) { return (M == 0 || (M >= N && N >= 1)) ? 1 : F9(M - 1, N) + F9(M - 1, N - 1); }
/* ---- PE 10: reverse a string with recursion: reverse the middle, swap the ends ---- */
static void rev(char *s, int lo, int hi) { if (lo >= hi) return; char t = s[lo]; s[lo] = s[hi]; s[hi] = t; rev(s, lo + 1, hi - 1); }
ملاحظات على كل دالة

PE 6 F(x,y) = ⌊x / y⌋ (قسمة صحيحة بالطرح المتكرر). y = 0 لانهائية، فبنحرسها بـ assert.

PE 7 نص السؤال ما حددش الحالات الأساسية. اخترت F(n,0) = F(n,n) = 1، وساعتها F(n,r) = C(n,r) (قاعدة باسكال). البرنامج بيقارنها بالصيغة المباشرة لـ n ≤ 12 (مثلًا F(10,5) = 252).

PE 8 Lambda(n) = ⌊log₂ n⌋ (عدد مرات القسمة على 2 لحد ما نوصل 1). اتأكد لـ n = 1…1000.

PE 9 F(M,N): الحالة الأساسية «M = 0 أو M ≥ N ≥ 1» = 1. لما N ≤ 0 و M > 0 مفيش حالة أساسية بتتحقق غير M = 0، فكل الـ 2M فرع بينتهي بـ 1: F(M,0) = 2M. ولما N > M ≥ 1 القيم زي باسكال (F(2,3) = 4، F(3,4) = 8). الجدول المطبوع (M صفوف 0–6، N أعمدة 0–6) بيوضح الشكل: مثلث الوحايد، وأعمدة 2M على الأطراف. (الدالة بتنتهي دايمًا لأن M بتقلّ في كل نداء.)

PE 10 عكس string بالـ recursion: بدّل الطرفين وانزل للوسط. حالة الـ string الفاضية والحرف الواحد بتقف على lo >= hi.

ف8 RQ 4: queue فاضية والعمليات (a)–(f)

/* A "book-style" linear queue: the array is printed with front and rear so the drawings can be compared. */
#define MAXQ 10
typedef struct { char a[MAXQ]; int front, rear; } lqueue;            /* front = rear = -1 : empty */
static void lq_init(lqueue *q) { memset(q->a, '.', MAXQ); q->front = q->rear = -1; }
static int lq_add(lqueue *q, char c)
{
    if (q->rear == MAXQ - 1) return 0;                               /* OVERFLOW: rear hit the end, even if cells before front are free */
    if (q->front == -1) q->front = 0;
    q->a[++q->rear] = c; return 1;
}
static int lq_del(lqueue *q, char *c)
{
    if (q->front == -1 || q->front > q->rear) return 0;              /* UNDERFLOW */
    *c = q->a[q->front]; q->a[q->front++] = '.';
    return 1;
}
static void lq_show(const lqueue *q, const char *what)
{
    printf("  %-22s |", what);
    for (int i = 0; i < MAXQ; i++) printf("%c", q->a[i]);
    printf("|  front=%d rear=%d\n", q->front, q->rear);
}
static void rq4(void)
{
    lqueue q; lq_init(&q); char c;
    for (char x = 'A'; x <= 'F'; x++) assert(lq_add(&q, x));
    lq_show(&q, "(a) add A..F");              assert(q.front == 0 && q.rear == 5);
    lq_del(&q, &c); assert(c == 'A'); lq_del(&q, &c); assert(c == 'B');
    lq_show(&q, "(b) delete two letters");    assert(q.front == 2 && q.rear == 5);
    lq_add(&q, 'G'); lq_show(&q, "(c) add G");  assert(q.rear == 6);
    lq_add(&q, 'H'); lq_show(&q, "(d) add H");  assert(q.rear == 7);
    for (int i = 0; i < 4; i++) lq_del(&q, &c);                      /* C D E F leave */
    lq_show(&q, "(e) delete four letters");   assert(q.front == 6 && q.rear == 7 && c == 'F');
    lq_add(&q, 'I'); lq_show(&q, "(f) add I");  assert(q.front == 6 && q.rear == 8 && q.a[6] == 'G' && q.a[8] == 'I');
}

ف8 RQ 5: أول FRONT=1 وREAR=5 (A..E)

السؤال مابيذكرش حجم المصفوفة، فحليته لحجمين: MAX = 10 (الـ linear يكفي)، وMAX = 6 (الخانات 0..5) فالـ linear بيعمل OVERFLOW من أول عملية والـ circular بيكمّل: كل العمليات بتنجح، والمصفوفة بتتملي بالظبط (6 عناصر) مرتين.

/* RQ 5: the book gives FRONT = 1, REAR = 5 holding A..E but not the array size.  We answer for two sizes:
   MAX = 10 (a linear queue just works) and MAX = 6 (indices 0..5, A..E sit in 1..5) where the LINEAR queue
   overflows at the very first operation, while the CIRCULAR queue copes. */
static void rq5(void)
{
    lqueue q; lq_init(&q); char c;
    memcpy(q.a + 1, "ABCDE", 5); q.front = 1; q.rear = 5;          /* the picture of the question */
    lq_show(&q, "start");
    assert(lq_add(&q, 'F'));  lq_show(&q, "(a) add F");
    lq_del(&q, &c); lq_del(&q, &c); lq_show(&q, "(b) delete two letters");
    lq_add(&q, 'G'); lq_show(&q, "(c) add G");
    lq_add(&q, 'H'); lq_show(&q, "(d) add H");
    for (int i = 0; i < 4; i++) lq_del(&q, &c);
    lq_show(&q, "(e) delete four letters");
    assert(lq_add(&q, 'I')); lq_show(&q, "(f) add I");
    assert(q.front == 7 && q.rear == 9 && q.a[7] == 'G' && q.a[8] == 'H' && q.a[9] == 'I');

    puts("  with MAX = 6 and the SAME start (FRONT=1, REAR=5):");
    /* linear: REAR = MAX-1 already, so "add F" is an overflow although cell 0 is free */
    int rear = 5, MAXS = 6;
    assert(rear == MAXS - 1);
    puts("    linear queue : (a) add F -> OVERFLOW (REAR = MAX-1), the free cell 0 cannot be used");
    /* circular, book convention: indices wrap */
    char a[6] = {'.', 'A', 'B', 'C', 'D', 'E'}; int front = 1, rr = 5, count = 5;
    rr = (rr + 1) % 6; a[rr] = 'F'; count++;                                       /* (a) F goes to cell 0 */
    assert(rr == 0 && count == 6);                                                 /* the queue is now exactly full */
    front = (front + 2) % 6; count -= 2;                                           /* (b) A, B leave */
    rr = (rr + 1) % 6; a[rr] = 'G'; count++;                                       /* (c) G -> cell 1 */
    rr = (rr + 1) % 6; a[rr] = 'H'; count++;                                       /* (d) H -> cell 2 */
    assert(count == 6 && front == 3 && rr == 2);                                   /* C D E F G H : full again */
    front = (front + 4) % 6; count -= 4;                                           /* (e) C D E F leave */
    rr = (rr + 1) % 6; a[rr] = 'I'; count++;                                       /* (f) I -> cell 3 */
    printf("    circular queue: front=%d rear=%d, contents:", front, rr);
    for (int i = 0, k = front; i < count; i++, k = (k + 1) % 6) printf(" %c", a[k]);
    puts("");
    assert(front == 1 && rr == 3 && a[1] == 'G' && a[2] == 'H' && a[3] == 'I' && count == 3);
}

ف8 RQ 6: الـ deque (LEFT=1، RIGHT=5، المصفوفة 10 خانات دايرية)

/* RQ 6: deque in a circular array of 10 cells, LEFT = 1, RIGHT = 5 holding A..E.
   LEFT moves back (wrapping) on insert-left and forward on delete-left; RIGHT is the mirror image. */
#define DN 10
typedef struct { char a[DN]; int left, right, n; } bdeque;
static void bd_show(const bdeque *d, const char *what)
{
    printf("  %-26s |", what);
    for (int i = 0; i < DN; i++) printf("%c", d->a[i]);
    printf("|  LEFT=%d RIGHT=%d\n", d->left, d->right);
}
static void bd_left(bdeque *d, char c)  { assert(d->n < DN); d->left = (d->left + DN - 1) % DN; d->a[d->left] = c; d->n++; }
static void bd_right(bdeque *d, char c) { assert(d->n < DN); d->right = (d->right + 1) % DN; d->a[d->right] = c; d->n++; }
static char bd_dell(bdeque *d) { assert(d->n); char c = d->a[d->left]; d->a[d->left] = '.'; d->left = (d->left + 1) % DN; d->n--; return c; }
static char bd_delr(bdeque *d) { assert(d->n); char c = d->a[d->right]; d->a[d->right] = '.'; d->right = (d->right + DN - 1) % DN; d->n--; return c; }
static void rq6(void)
{
    bdeque d; memset(d.a, '.', DN); memcpy(d.a + 1, "ABCDE", 5); d.left = 1; d.right = 5; d.n = 5;
    bd_show(&d, "start");
    bd_left(&d, 'F');  bd_show(&d, "(a) add F on the left");
    bd_right(&d, 'G'); bd_show(&d, "(b) add G on the right");
    bd_right(&d, 'H'); bd_show(&d, "(c) add H on the right");
    char x = bd_dell(&d), y = bd_dell(&d); bd_show(&d, "(d) delete 2 from left");   assert(x == 'F' && y == 'A');
    bd_right(&d, 'I'); bd_show(&d, "(e) add I on the right");
    bd_left(&d, 'J');  bd_show(&d, "(f) add J on the left");
    x = bd_delr(&d); y = bd_delr(&d); bd_show(&d, "(g) delete 2 from right");        assert(x == 'I' && y == 'H');
    assert(d.left == 1 && d.right == 6 && d.n == 6);
    char s[8]; int w = 0; for (int i = 0, k = d.left; i < d.n; i++, k = (k + 1) % DN) s[w++] = d.a[k]; s[w] = 0;
    printf("  final deque, left to right: %s\n", s);
    assert(strcmp(s, "JBCDEG") == 0);
}

/* ---- PE 1: number of items. Linear: rear - front + 1.  Circular: (rear - front + MAX) % MAX + 1 (empty handled apart) ---- */
static int lq_count(const lqueue *q) { return q->front == -1 || q->front > q->rear ? 0 : q->rear - q->front + 1; }
static int circ_count(int front, int rear, int max) { return front == -1 ? 0 : (rear - front + max) % max + 1; }
/* ---- PE 4: deque as a doubly linked list with head and tail: all four operations O(1) ---- */
typedef struct dnode { int val; struct dnode *prev, *next; } dnode;
typedef struct { dnode *head, *tail; int n; } ldeque;
static void ld_push_left(ldeque *d, int v)
{
    dnode *x = malloc(sizeof *x); if (!x) exit(1);
    x->val = v; x->prev = NULL; x->next = d->head;
    if (d->head) d->head->prev = x; else d->tail = x;
    d->head = x; d->n++;
}
static void ld_push_right(ldeque *d, int v)
{
    dnode *x = malloc(sizeof *x); if (!x) exit(1);
    x->val = v; x->next = NULL; x->prev = d->tail;
    if (d->tail) d->tail->next = x; else d->head = x;
    d->tail = x; d->n++;
}
static int ld_pop_left(ldeque *d, int *v)
{
    if (!d->head) return 0;
    dnode *x = d->head; *v = x->val; d->head = x->next;
    if (d->head) d->head->prev = NULL; else d->tail = NULL;
    free(x); d->n--; return 1;
}
static int ld_pop_right(ldeque *d, int *v)
{
    if (!d->tail) return 0;
    dnode *x = d->tail; *v = x->val; d->tail = x->prev;
    if (d->tail) d->tail->next = NULL; else d->head = NULL;
    free(x); d->n--; return 1;
}
/* ---- PE 11: reverse a queue with a stack: dequeue everything onto the stack, pop everything back.
        PE 12: compare two queues: dequeue/enqueue each exactly n times so both are rotated back to where they started. ---- */
static void queue_reverse(cqueue *q)
{
    istack s = {{0}, 0}; int x;
    while (cq_get(q, &x)) is_push(&s, x);
    while (s.n) cq_put(q, is_pop(&s));
}
static int queue_equal(cqueue *a, cqueue *b)
{
    if (a->n != b->n) return 0;
    int n = a->n, eq = 1, x = 0, y = 0;
    for (int i = 0; i < n; i++) {
        cq_get(a, &x); cq_get(b, &y);
        if (x != y) eq = 0;
        cq_put(a, x); cq_put(b, y);                    /* rotate: after n steps both are back in the original order */
    }
    return eq;
}
