1) الـ BST: التعريف والعمليات (Thareja 10.1–10.2)

Binary search tree: شجرة ثنائية فيها لكل عقدة N: كل المفاتيح في شمالها أصغر من مفتاحها، وكل اللي في يمينها أكبر (أو يساوي، في طريقة الكتاب اللي بتحط المكرر يمين). النتيجة المباشرة: in-order traversal بيطلّع المفاتيح مرتبة. البحث بيمشي من الجذر شمال/يمين زي binary search، وتكلفته = عمق العقدة.

static INode *bst_search(INode *t, int v)
{
    while (t && t->key != v) t = v < t->key ? t->l : t->r;
    return t;
}
/* equal keys go to the right subtree (Thareja: "else insert in the right sub-tree") */
static INode *bst_insert(INode *t, int v)
{
    if (!t) return inode(v);
    if (v < t->key) t->l = bst_insert(t->l, v);
    else            t->r = bst_insert(t->r, v);
    return t;
}
static INode *bst_min(INode *t) { while (t && t->l) t = t->l; return t; }   /* leftmost node  */
static INode *bst_max(INode *t) { while (t && t->r) t = t->r; return t; }   /* rightmost node */

الحذف: ثلاث حالات

(1) ورقة: نشيلها. (2) ابن واحد: الابن بياخد مكان أبوه. (3) ابنين: نبدّل المفتاح بالـ predecessor (أكبر مفتاح في الشمال) أو الـ successor (أصغر مفتاح في اليمين) وبعدين نحذف العقدة اللي اتنقل منها (وهي بالضرورة حالة 1 أو 2، لأن أكبر عقدة في الشمال ملهاش ابن يمين). في الكود ده بنستعمل الـ predecessor عشان تتطابق نتايج تمارين الكتاب.

/* Three cases (Thareja 10.2.3):
     1. leaf            -> just remove it
     2. one child       -> the child takes its place
     3. two children    -> copy the in-order PREDECESSOR (largest key of the left subtree) into the node,
                           then delete that predecessor node (which has at most one child: case 1 or 2).
   use_succ = 1 switches to the in-order SUCCESSOR (smallest key of the right subtree). */
static INode *bst_delete(INode *t, int v, int use_succ)
{
    if (!t) return NULL;                                    /* not found: nothing to do */
    if (v < t->key) t->l = bst_delete(t->l, v, use_succ);
    else if (v > t->key) t->r = bst_delete(t->r, v, use_succ);
    else if (t->l && t->r) {                                /* case 3 */
        if (use_succ) { INode *s = bst_min(t->r); t->key = s->key; t->r = bst_delete(t->r, s->key, use_succ); }
        else          { INode *p = bst_max(t->l); t->key = p->key; t->l = bst_delete(t->l, p->key, use_succ); }
    } else {                                                /* case 1 and 2 together */
        INode *child = t->l ? t->l : t->r;
        free(t);
        return child;
    }
    return t;
}
static int bst_height(const INode *t)  { return iheight(t); }                    /* empty tree: 0, one node: 1 */
static int bst_nodes(const INode *t)   { return isize(t); }
static int bst_leaves(const INode *t)  { return !t ? 0 : (!t->l && !t->r) ? 1 : bst_leaves(t->l) + bst_leaves(t->r); }
static int bst_internal(const INode *t){ return bst_nodes(t) - bst_leaves(t); }
/* external nodes = the NULL pointers = n + 1 for n nodes */
static int bst_external(const INode *t){ return !t ? 1 : bst_external(t->l) + bst_external(t->r); }
static INode *bst_mirror(INode *t)                       /* swap the children at every node */
{
    if (t) { INode *x = t->l; t->l = bst_mirror(t->r); t->r = bst_mirror(x); }
    return t;
}

كام BST ممكن من نفس المفاتيح؟

نفس n مفتاح ممكن تدّي أشجار مختلفة حسب ترتيب الإدخال. عدد الأشكال الممكنة (من غير ما نهتم بترتيب الإدخال) = كاتالان C(n). مع {7, 9, 11}: 5 أشجار (Thareja Q19)، والكود بيولّدها كلها:

/* all distinct BSTs on the keys sorted in k[lo..hi]:  choose each key as root, combine every left shape
   with every right shape.  Writes specs into out[]; returns how many. */
static int all_bsts(const int *k, int lo, int hi, char out[][96])
{
    if (lo > hi) { strcpy(out[0], "."); return 1; }
    int cnt = 0;
    for (int r = lo; r <= hi; r++) {
        char L[16][96], R[16][96];
        int a = all_bsts(k, lo, r - 1, L), b = all_bsts(k, r + 1, hi, R);
        for (int i = 0; i < a; i++)
            for (int j = 0; j < b; j++) {
                if (L[i][0] == '.' && R[j][0] == '.') snprintf(out[cnt], 96, "%d", k[r]);
                else { snprintf(out[cnt], 96, "%d(", k[r]); strcat(out[cnt], L[i]); strcat(out[cnt], ","); strcat(out[cnt], R[j]); strcat(out[cnt], ")"); }
                cnt++;
            }
    }
    return cnt;
}

static LN *ins(LN *t, int k, long *depth_acc, int d)     /* returns new root; adds the depth of a NEW node to *depth_acc */
{

    else if (k > t->key) t->r = ins(t->r, k, depth_acc, d + 1);
    else t->dead = 0;                                    /* lazy deletion: inserting a dead key revives it */

}

الحذف الكسول (lazy deletion) — Weiss 4.3.6

لو الحذف نادر نحط علامة «محذوف» على العقدة بدل ما نشيلها. فائدته: بساطة، وإعادة الإدخال لنفس المفتاح مجانية. عيبه: الشجرة بتكبر بعقد ميتة، فنعمل purge (إعادة بناء) لما عدد الميتين يعدّي نصف الشجرة مثلًا.

static int lcontains(const LN *t, int k)
{
    while (t && t->key != k) t = k < t->key ? t->l : t->r;
    return t && !t->dead;                                /* found AND not marked deleted */
}
static int ldelete(LN *t, int k)                         /* O(depth), no relinking, no free */
{
    while (t && t->key != k) t = k < t->key ? t->l : t->r;
    if (!t || t->dead) return 0;
    t->dead = 1; return 1;
}
static int llive(const LN *t) { return t ? (!t->dead) + llive(t->l) + llive(t->r) : 0; }
static void collect(const LN *t, int *a, int *n) { if (t) { collect(t->l, a, n); if (!t->dead) a[(*n)++] = t->key; collect(t->r, a, n); } }
static LN *build_balanced(const int *a, int lo, int hi)  /* rebuild from the sorted live keys: purge the dead ones, height ~ log n */
{
    if (lo > hi) return NULL;
    int m = (lo + hi) / 2; LN *n = calloc(1, sizeof *n); n->key = a[m];
    n->l = build_balanced(a, lo, m - 1); n->r = build_balanced(a, m + 1, hi);
    return n;
}

2) الـ Threaded Binary Trees (Thareja 10.3، Lipschutz 7.6–7.7)

شجرة ثنائية بـ n عقدة فيها n + 1 مؤشر NULL (كل عقدة ليها مؤشرين: 2n مؤشر، منها n − 1 بتشاور على أبناء). بدل ما نسيبهم فاضيين نخلّيهم threads بتشاور على جيران الـ in-order: الـ NULL اليمين → الـ successor (one-way threading)، والـ NULL الشمال → الـ predecessor (two-way أو fully threaded). علامة (flag) لكل مؤشر بتقول «ابن» ولا «thread». الفايدة: traversal من غير stack ولا recursion.

typedef struct {
    int l[N], r[N];                 /* LEFT, RIGHT: node number, or -1 for NULL                    */
    int lth[N], rth[N];             /* 1 if that pointer is a thread, 0 if it is a real child       */
    int root, n;                    /* ROOT and number of nodes (node numbers are 1..n)             */
} TA;

static void ta_from(TA *a, const INode *t)       /* node number = key (the figures use 1..12) */
{
    if (!t) return;
    if (t->key > a->n) a->n = t->key;
    a->l[t->key] = t->l ? t->l->key : -1;
    a->r[t->key] = t->r ? t->r->key : -1;
    ta_from(a, t->l); ta_from(a, t->r);
}
static void ta_init(TA *a, const INode *t)
{
    memset(a, 0, sizeof *a);
    for (int i = 0; i < N; i++) a->l[i] = a->r[i] = -1;
    a->root = t->key; ta_from(a, t);
}
static void order_in(const TA *a, int x, int *o, int *k)  { if (x == -1) return; order_in(a, a->l[x], o, k); o[(*k)++] = x; order_in(a, a->r[x], o, k); }
static void order_pre(const TA *a, int x, int *o, int *k) { if (x == -1) return; o[(*k)++] = x; order_pre(a, a->l[x], o, k); order_pre(a, a->r[x], o, k); }

/* one-way in-order: every NULL right pointer becomes a thread to the in-order successor */
static void thread_one_way_in(TA *a)
{
    int o[N], k = 0; order_in(a, a->root, o, &k);
    for (int i = 0; i < k; i++)
        if (a->r[o[i]] == -1 && i + 1 < k) { a->r[o[i]] = o[i + 1]; a->rth[o[i]] = 1; }
}
/* one-way pre-order: NULL right pointers point to the PRE-order successor */
static void thread_one_way_pre(TA *a)
{
    int o[N], k = 0; order_pre(a, a->root, o, &k);
    for (int i = 0; i < k; i++)
        if (a->r[o[i]] == -1 && i + 1 < k) { a->r[o[i]] = o[i + 1]; a->rth[o[i]] = 1; }
}
/* two-way in-order: NULL left pointers point to the predecessor as well */
static void thread_two_way_in(TA *a)
{
    thread_one_way_in(a);
    int o[N], k = 0; TA tmp = *a; for (int i = 0; i < N; i++) if (tmp.rth[i]) tmp.r[i] = -1;
    order_in(&tmp, a->root, o, &k);
    for (int i = 0; i < k; i++)
        if (a->l[o[i]] == -1 && i > 0) { a->l[o[i]] = o[i - 1]; a->lth[o[i]] = 1; }
}
/* walk without a stack or recursion */
static int walk_in_one_way(const TA *a, int *out)
{
    int k = 0, x = a->root;
    while (a->l[x] != -1 && !a->lth[x]) x = a->l[x];              /* leftmost node */
    while (x != -1) {
        out[k++] = x;
        if (a->rth[x]) x = a->r[x];                               /* thread: successor in one step   */
        else {                                                     /* real child: leftmost of right   */
            x = a->r[x];
            while (x != -1 && a->l[x] != -1 && !a->lth[x]) x = a->l[x];
        }
    }
    return k;
}
static int walk_pre_one_way(const TA *a, int *out)
{
    int k = 0, x = a->root;
    while (x != -1) {
        out[k++] = x;
        x = (a->l[x] != -1 && !a->lth[x]) ? a->l[x] : a->r[x];    /* left child if any, else right/thread */
    }
    return k;
}

النسخة بالمؤشرات مع Header node

Lipschutz 7.6: بنضيف عقدة header بتخلّي الـ thread الشمال للعقدة الأولى والـ thread اليمين للعقدة الأخيرة يشاوروا عليها، فمفيش NULL استثناء في أي مكان والـ traversal بيقف لما نرجع للـ header.

/* Lipschutz's layout: a HEADER node H. H.left -> root, H.right -> H (itself). The root's tree is
   threaded so that the leftmost node's left thread and the rightmost node's right thread both point to H.
   An empty tree is H with H.left -> H (a thread). */
typedef struct TN {
    int key;
    struct TN *l, *r;
    int lth, rth;                   /* 1 = thread, 0 = child */
} TN;

static TN *th_new(void)
{
    TN *h = malloc(sizeof *h);
    h->key = 0; h->l = h; h->r = h; h->lth = 1; h->rth = 0;
    return h;
}
static TN *th_succ(const TN *p)                 /* in-order successor */
{
    TN *s = p->r;
    if (!p->rth) while (!s->lth) s = s->l;     /* real right child: go to its leftmost */
    return s;
}
static TN *th_pred(const TN *p)                 /* in-order predecessor (mirror image) */
{
    TN *s = p->l;
    if (!p->lth) while (!s->rth) s = s->r;
    return s;
}
/* BST insert (duplicates ignored). New leaf inherits the thread it is replacing. */
static int th_insert(TN *h, int key)
{
    TN *n = malloc(sizeof *n); n->key = key; n->lth = n->rth = 1;
    if (h->lth) {                               /* empty tree */
        n->l = h; n->r = h; h->l = n; h->lth = 0;
        return 1;
    }
    TN *p = h->l;
    for (;;) {
        if (key == p->key) { free(n); return 0; }
        if (key < p->key) {
            if (p->lth) { n->l = p->l; n->r = p; p->l = n; p->lth = 0; return 1; }
            p = p->l;
        } else {
            if (p->rth) { n->r = p->r; n->l = p; p->r = n; p->rth = 0; return 1; }
            p = p->r;
        }
    }
}
static void th_inorder(const TN *h, int *out, int *k)
{
    *k = 0;
    for (const TN *p = th_succ(h); p != h; p = th_succ(p)) out[(*k)++] = p->key;   /* H's "successor" is the first node */
}
static void th_reverse(const TN *h, int *out, int *k)
{
    *k = 0;
    for (const TN *p = th_pred(h); p != h; p = th_pred(p)) out[(*k)++] = p->key;
}
static void th_free(TN *h)
{
    TN *p = th_succ(h);
    while (p != h) { TN *nx = th_succ(p); free(p); p = nx; }
    free(h);
}

3) الـ AVL Trees (Thareja 10.4، Weiss 4.4، Lipschutz 7.11–7.14)

شجرة BST بحيث فرق ارتفاع الشمال واليمين ≤ 1 عند كل عقدة. Balance factor BF = ارتفاع الشمال − ارتفاع اليمين (تعريف Thareja، Weiss بيستعمله بالعكس أحيانًا) ∈ {−1, 0, +1}. الارتفاع O(log n): لو S(h) أقل عدد عقد لـ AVL بارتفاع h فـ S(h) = S(h−1) + S(h−2) + 1 (شجرة بشمال من h−1 ويمين h−2، زي فيبوناتشي) فبيطلع S(h) = 1, 2, 4, 7, 12, 20, 33, 54 ... وبالتالي h ≤ 1.44 log₂(n+2) − 1.33. الكود بيتأكد من الأرقام دي بالحساب وبالبناء.

static int H(const INode *n) { return n ? n->h : 0; }
static int BF(const INode *n) { return n ? H(n->l) - H(n->r) : 0; }
static void upd(INode *n) { int a = H(n->l), b = H(n->r); n->h = 1 + (a > b ? a : b); }

/*        y                x
         / \              / \
        x   C    -->     A   y        rotate RIGHT at y  (x = y->l)
       / \                  / \
      A   B                B   C                                                          */
static INode *rot_right(INode *y)
{
    INode *x = y->l;
    y->l = x->r; x->r = y;
    upd(y); upd(x);
    return x;
}
static INode *rot_left(INode *x)              /* mirror image */
{
    INode *y = x->r;
    x->r = y->l; y->l = x;
    upd(x); upd(y);
    return y;
}
/* restore the AVL property at n (its subtrees are already AVL). `del` = 0 insert, 1 = deleted from
   the left subtree, 2 = deleted from the right subtree (only used for the log names). */
static INode *rebalance(INode *n, int del)
{
    upd(n);
    int bf = BF(n);
    if (bf > 1) {                                           /* left heavy */
        int cb = BF(n->l);
        if (del == 2) logr(cb == 0 ? "R0" : cb == 1 ? "R1" : "R-1", n->key);
        else if (!del) logr(cb >= 0 ? "LL" : "LR", n->key);
        if (cb < 0) n->l = rot_left(n->l);
        return rot_right(n);
    }
    if (bf < -1) {                                          /* right heavy */
        int cb = BF(n->r);
        if (del == 1) logr(cb == 0 ? "L0" : cb == -1 ? "L-1" : "L1", n->key);
        else if (!del) logr(cb <= 0 ? "RR" : "RL", n->key);
        if (cb > 0) n->r = rot_right(n->r);
        return rot_left(n);
    }
    return n;
}
static INode *avl_insert(INode *t, int v)
{
    if (!t) return inode(v);
    if (v < t->key) t->l = avl_insert(t->l, v);
    else if (v > t->key) t->r = avl_insert(t->r, v);
    else return t;                                          /* duplicates ignored */
    return rebalance(t, 0);
}

static INode *avl_delete(INode *t, int v)
{
    if (!t) return NULL;
    int side = 0;
    if (v < t->key) { t->l = avl_delete(t->l, v); side = 1; }
    else if (v > t->key) { t->r = avl_delete(t->r, v); side = 2; }
    else if (t->l && t->r) {                                /* two children: predecessor replaces it */
        INode *p = t->l; while (p->r) p = p->r;
        t->key = p->key;
        t->l = avl_delete(t->l, p->key); side = 1;
    } else {
        INode *c = t->l ? t->l : t->r;
        free(t);
        return c;
    }
    return rebalance(t, side);
}

4) الـ Red-Black Trees (Thareja 10.5، Weiss 12.2)

BST كل عقدة فيها لون (أحمر/أسود)، والقواعد الأربعة: (1) الجذر أسود. (2) كل الـ NIL leaves (الأطراف الخارجية) سوداء. (3) العقدة الحمراء ملهاش ابن أحمر (مفيش حمرتين ورا بعض). (4) كل المسارات من أي عقدة لأي NIL تحتها فيها نفس عدد السواد (black-height). النتيجة: أطول مسار ≤ 2× أقصر مسار، فالارتفاع ≤ 2 log₂(n+1). أبسط من AVL في الحذف والإدخال (ثابت عدد rotations: حد أقصى 2 للإدخال و3 للحذف) والتوازن أقل صرامة.

static RB NILNODE = { 0, BLACK, &NILNODE, &NILNODE, &NILNODE };   /* ONE shared black sentinel */
#define NIL (&NILNODE)

typedef struct { RB *root; char log[256]; } RBT;

static void rlog(RBT *t, const char *s)
{
    if (strlen(t->log) + strlen(s) + 2 < sizeof t->log) { if (t->log[0]) strcat(t->log, " "); strcat(t->log, s); }
}
static void rot_left(RBT *t, RB *x)
{
    RB *y = x->r;
    x->r = y->l; if (y->l != NIL) y->l->p = x;
    y->p = x->p;
    if (x->p == NIL) t->root = y; else if (x == x->p->l) x->p->l = y; else x->p->r = y;
    y->l = x; x->p = y;
}
static void rot_right(RBT *t, RB *y)
{
    RB *x = y->l;
    y->l = x->r; if (x->r != NIL) x->r->p = y;
    x->p = y->p;
    if (y->p == NIL) t->root = x; else if (y == y->p->l) y->p->l = x; else y->p->r = x;
    x->r = y; y->p = x;
}

static void insert_fix(RBT *t, RB *z)
{
    while (z->p->color == RED) {                       /* parent red => grandparent exists and is black */
        RB *g = z->p->p;
        if (z->p == g->l) {
            RB *u = g->r;                              /* the uncle */
            if (u->color == RED) {                     /* case 1: recolour and move up */
                z->p->color = BLACK; u->color = BLACK; g->color = RED; rlog(t, "I1");
                z = g;
            } else {
                if (z == z->p->r) { z = z->p; rot_left(t, z); rlog(t, "I2"); }   /* case 2: triangle -> line */
                z->p->color = BLACK; g->color = RED; rot_right(t, g); rlog(t, "I3"); /* case 3 */
            }
        } else {                                       /* mirror image */
            RB *u = g->l;
            if (u->color == RED) {
                z->p->color = BLACK; u->color = BLACK; g->color = RED; rlog(t, "I1");
                z = g;
            } else {
                if (z == z->p->l) { z = z->p; rot_right(t, z); rlog(t, "I2"); }
                z->p->color = BLACK; g->color = RED; rot_left(t, g); rlog(t, "I3");
            }
        }
    }
    t->root->color = BLACK;                            /* rule 1 */
}
static int rb_insert(RBT *t, int key)
{
    RB *y = NIL, *x = t->root;
    while (x != NIL) { y = x; if (key == x->key) return 0; x = key < x->key ? x->l : x->r; }
    RB *z = malloc(sizeof *z);
    z->key = key; z->color = RED; z->l = z->r = NIL; z->p = y;
    if (y == NIL) t->root = z; else if (key < y->key) y->l = z; else y->r = z;
    insert_fix(t, z);
    return 1;
}

الحذف: الـ «extra black»

حذف عقدة حمراء مفيهوش مشكلة. حذف سودا بيخلّي مكانها «سواد زايد» بيتحرك لفوق بـ 4 حالات حسب أخ العقدة w: D1 الأخ أحمر (نلوّنه ونعمل rotation فتتحول لحالة تانية). D2 الأخ أسود وابناه سود (نلوّن الأخ أحمر ونصعّد المشكلة للأب). D3 الأخ أسود وابنه «القريب» أحمر (rotation تحوّلها لـ D4). D4 الأخ أسود وابنه «البعيد» أحمر (rotation وخلصنا).

static void transplant(RBT *t, RB *u, RB *v)
{
    if (u->p == NIL) t->root = v; else if (u == u->p->l) u->p->l = v; else u->p->r = v;
    v->p = u->p;                                       /* legal even when v is NIL: we need NIL->p below */
}
static void delete_fix(RBT *t, RB *x)                  /* x carries an "extra black" */
{
    while (x != t->root && x->color == BLACK) {
        if (x == x->p->l) {
            RB *w = x->p->r;                           /* sibling */
            if (w->color == RED) {                     /* case 1: red sibling -> make it black */
                w->color = BLACK; x->p->color = RED; rot_left(t, x->p); w = x->p->r; rlog(t, "D1");
            }
            if (w->l->color == BLACK && w->r->color == BLACK) {   /* case 2: push the extra black up */
                w->color = RED; x = x->p; rlog(t, "D2");
            } else {
                if (w->r->color == BLACK) {            /* case 3: near child red -> rotate to case 4 */
                    w->l->color = BLACK; w->color = RED; rot_right(t, w); w = x->p->r; rlog(t, "D3");
                }
                w->color = x->p->color; x->p->color = BLACK; w->r->color = BLACK;   /* case 4: far child red */
                rot_left(t, x->p); rlog(t, "D4");
                x = t->root;
            }
        } else {                                       /* mirror image */
            RB *w = x->p->l;
            if (w->color == RED) { w->color = BLACK; x->p->color = RED; rot_right(t, x->p); w = x->p->l; rlog(t, "D1"); }
            if (w->r->color == BLACK && w->l->color == BLACK) { w->color = RED; x = x->p; rlog(t, "D2"); }
            else {
                if (w->l->color == BLACK) { w->r->color = BLACK; w->color = RED; rot_left(t, w); w = x->p->l; rlog(t, "D3"); }
                w->color = x->p->color; x->p->color = BLACK; w->l->color = BLACK;
                rot_right(t, x->p); rlog(t, "D4");
                x = t->root;
            }
        }
    }
    x->color = BLACK;
}
static RB *rb_find(RB *x, int key) { while (x != NIL && x->key != key) x = key < x->key ? x->l : x->r; return x; }
static int rb_delete(RBT *t, int key)
{
    RB *z = rb_find(t->root, key);
    if (z == NIL) return 0;
    RB *y = z, *x; int ycol = y->color;
    if (z->l == NIL) { x = z->r; transplant(t, z, z->r); }
    else if (z->r == NIL) { x = z->l; transplant(t, z, z->l); }
    else {                                             /* two children: use the in-order successor */
        y = z->r; while (y->l != NIL) y = y->l;
        ycol = y->color; x = y->r;
        if (y->p == z) x->p = y;
        else { transplant(t, y, y->r); y->r = z->r; y->r->p = y; }
        transplant(t, z, y);
        y->l = z->l; y->l->p = y; y->color = z->color;
    }
    free(z);
    if (ycol == BLACK) delete_fix(t, x);               /* only a removed BLACK node breaks rule 4 */
    return 1;
}

5) الـ Splay Trees (Thareja 10.6، Weiss 4.5 وتحليلها في 11.5)

شجرة BST من غير أي معلومة توازن (لا ارتفاع ولا لون): كل مرة بنوصل لعقدة (بحث/إدخال/حذف) بنرفعها للـ جذر بسلسلة خطوات splay. الفكرة: اللي اتستخدم قريب غالبًا هيتستخدم تاني. العملية الواحدة ممكن تبقى O(n)، لكن amortised O(log n) لكل عملية (تحليل Weiss 11.5 بالـ potential function Σ log size).

static void slog(ST *t, const char *s) { if (strlen(t->log) + strlen(s) + 2 < sizeof t->log) { if (t->log[0]) strcat(t->log, " "); strcat(t->log, s); } }

static void rotate_up(ST *t, SN *x)           /* rotate x above its parent (right rotation if x is a left child) */
{
    SN *p = x->p, *g = p->p;
    if (x == p->l) { p->l = x->r; if (x->r) x->r->p = p; x->r = p; }
    else           { p->r = x->l; if (x->l) x->l->p = p; x->l = p; }
    p->p = x; x->p = g;
    if (!g) t->root = x; else if (g->l == p) g->l = x; else g->r = x;
}
static void splay(ST *t, SN *x)
{
    while (x->p) {
        SN *p = x->p, *g = p->p;
        if (!g) { rotate_up(t, x); slog(t, "zig"); }                           /* P is the root */
        else if ((g->l == p) == (p->l == x)) { rotate_up(t, p); rotate_up(t, x); slog(t, "zigzig"); }   /* same side */
        else { rotate_up(t, x); rotate_up(t, x); slog(t, "zigzag"); }          /* opposite sides */
    }
}
static SN *s_new(int k, SN *p) { SN *n = malloc(sizeof *n); n->key = k; n->l = n->r = NULL; n->p = p; return n; }

static int s_insert(ST *t, int k)            /* ordinary BST insert, then splay the new node */
{
    SN *y = NULL, *x = t->root;
    while (x) { y = x; if (k == x->key) { splay(t, x); return 0; } x = k < x->key ? x->l : x->r; }
    SN *z = s_new(k, y);
    if (!y) t->root = z; else if (k < y->key) y->l = z; else y->r = z;
    splay(t, z);
    return 1;
}
static SN *s_search(ST *t, int k)            /* a search ALSO restructures: found node (or last node on the path) goes to the root */
{
    SN *x = t->root, *last = NULL;
    while (x) { last = x; if (k == x->key) { splay(t, x); return x; } x = k < x->key ? x->l : x->r; }
    if (last) splay(t, last);
    return NULL;
}
static int s_delete(ST *t, int k)
{
    if (!s_search(t, k) || t->root->key != k) return 0;
    SN *z = t->root, *L = z->l, *R = z->r;
    if (L) L->p = NULL;
    if (R) R->p = NULL;
    free(z);
    if (!L) { t->root = R; return 1; }
    t->root = L;                              /* splay the MAXIMUM of the left tree to its root: it then has no right child */
    SN *m = L; while (m->r) m = m->r;
    splay(t, m);
    m->r = R; if (R) R->p = m;
    return 1;
}
