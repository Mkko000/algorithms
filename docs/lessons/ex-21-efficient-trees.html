RQ 1 و 2 و 3 و 5 و 6: أسئلة نظرية

الإجابات

1) BST: شجرة ثنائية، كل مفتاح أكبر من كل اللي في شماله وأصغر من كل اللي في يمينه (من غير ما نكرر الـ duplicates). in-order يطلّع المفاتيح مرتبة. التعريف recursive.

2) search: قارن مع الجذر وانزل شمال/يمين. insert: ابحث لحد ما توصل لـ NULL وحط العقدة هناك (دايمًا ورقة). delete: 3 حالات (ورقة / ابن واحد / ابنين: استبدل بـ predecessor أو successor). كمان: أصغر وأكبر مفتاح (أقصى شمال وأقصى يمين)، الارتفاع، عدّ العقد، mirror image، حذف الشجرة كلها. كلهم في bst_ops.c.

3) كل العمليات O(ارتفاع الشجرة). متوسط عشوائي ≈ 1.39 log₂ n؛ مرتب = chain بارتفاع n (قياس: 1024 مفتاح عشوائي بمتوسط عمق 11.09 مقابل 511 للمرتب، راجع bst_analysis.c). عشان كده بنوازن (AVL / red-black) أو نتأقلم (splay).

5) Threaded binary tree: شجرة بتستعمل الـ NULL pointers (عددهم n+1) كـ threads بتشاور على جيران in-order: الـ right thread على الـ successor، والـ left thread على الـ predecessor. بنميّز الـ thread عن الابن بـ flag. أنواعها: one-way (right threads فقط) وtwo-way (الاتنين)، in-order أو pre-order.

6) ليه efficient: الـ traversal من غير stack ولا recursion (O(1) مساحة)، وإيجاد الـ successor والـ predecessor في O(1) تقريبًا من غير ما نطلع للأب، واستغلال مؤشرات كانت ضايعة. عيوبها: الإدخال والحذف أعقد، وعدد flags زيادة.

RQ 4: complete tree بـ 27 عقدة

/* ---- RQ 4: a complete binary tree with 27 nodes: how many in the last level, what height? ----
   Levels 0..3 hold 1+2+4+8 = 15 nodes, level 4 holds the other 27 - 15 = 12 (capacity 16).
   Height (counting nodes on the longest path) = 5.                                              */
static void rq4(void)
{
    int n = 27, full = 0, level = 0;
    while (full + (1 << level) < n) { full += 1 << level; level++; }        /* last level number = 4 */
    int last = n - full, height = level + 1;
    printf("RQ4: n=27 -> last level has %d nodes, height %d\n", last, height);
    assert(last == 12 && height == 5);
    /* height of a complete tree is ceil(log2(n+1)) */
    int h2 = 0; while ((1 << h2) < n + 1) h2++;
    assert(h2 == height);
}
الإجابة

المستويات 0–3 فيها 1 + 2 + 4 + 8 = 15 عقدة، فالباقي 27 − 15 = 12 عقدة في المستوى الأخير (سعته 16)، والارتفاع 5 = ⌈log₂ 28⌉.

RQ 7 و 8 و 9 و 10: AVL وred-black وsplay (مقارنات)

الإجابات

7) مزايا AVL: ارتفاع مضمون ≤ 1.44 log₂(n+2) فالبحث والإدخال والحذف O(log n) في أسوأ حالة، توازن صارم يخلّي البحث أسرع ما يمكن بين الأشجار الموزونة، وإعادة التوازن بعدد صغير من الـ rotations (واحدة أو اتنين للإدخال).

8) AVL مقابل BST: BST العادي ممكن يبقى chain (O(n)) حسب ترتيب الإدخال، أما AVL فبيصلّح نفسه بعد كل تعديل فالارتفاع دايمًا O(log n). الثمن: تخزين الارتفاع/BF وrotations.

9) red-black مقابل BST: القواعد الأربعة بتحط حد للارتفاع 2 log₂(n+1). في الكود: 1024 مفتاح مرتب → BST ارتفاعها 1024 وred-black ارتفاعها 18 (الحد 22).

10) splay: مزايا: كود بسيط، من غير مساحة إضافية، amortised O(log n)، بيتأقلم مع نمط الاستعمال (الأكتر استعمالًا قريب من الجذر)، وبيدعم عمليات زي split/join. عيوب: عملية واحدة ممكن تبقى O(n)، والقراءة بتعدّل الشجرة (مش آمنة للـ concurrency)، وما بتضمنش ارتفاع صغير لو الوصول منتظم.

/* ---- RQ 9: how does a red-black tree beat a plain BST?  The height of a BST depends on the insertion order
   (a sorted input gives a chain, height n), the red-black rules cap it at 2*log2(n+1).  ---- */
static int bst_chain_height(int n) { return n; }
static void rq9(void)
{
    RBT t = { NIL, "" };
    for (int i = 1; i <= 1024; i++) rb_insert(&t, i);
    int h = rb_height(t.root);
    printf("RQ9: 1024 sorted keys: plain BST height %d, red-black height %d (bound %d)\n", bst_chain_height(1024), h, 2 * 11);
    assert(h <= 22 && rb_valid(&t));
    rb_free(t.root);
}

/* Three cases (Thareja 10.2.3):
     1. leaf            -> just remove it
     2. one child       -> the child takes its place
     3. two children    -> copy the in-order PREDECESSOR (largest key of the left subtree) into the node,
                           then delete that predecessor node (which has at most one child: case 1 or 2).
   use_succ = 1 switches to the in-order SUCCESSOR (smallest key of the right subtree). */
static INode *bst_delete(INode *t, int v, int use_succ)
{
    if (!t) return NULL;                                    /* not found: nothing to do */
    if (v < t->key) t->l = bst_delete(t->l, v, use_succ);
    else if (v > t->key) t->r = bst_delete(t->r, v, use_succ);
    else if (t->l && t->r) {                                /* case 3 */
        if (use_succ) { INode *s = bst_min(t->r); t->key = s->key; t->r = bst_delete(t->r, s->key, use_succ); }
        else          { INode *p = bst_max(t->l); t->key = p->key; t->l = bst_delete(t->l, p->key, use_succ); }
    } else {                                                /* case 1 and 2 together */
        INode *child = t->l ? t->l : t->r;
        free(t);
        return child;
    }
    return t;
}

static INode *avl_insert(INode *t, int v)
{
    if (!t) return inode(v);
    if (v < t->key) t->l = avl_insert(t->l, v);
    else if (v > t->key) t->r = avl_insert(t->r, v);
    else return t;                                          /* duplicates ignored */
    return rebalance(t, 0);
}

RQ 14: خصائص red-black

(1) الجذر أسود. (2) كل NIL leaf أسود. (3) العقدة الحمراء ابناها أسودين (مفيش أحمرين ورا بعض). (4) نفس عدد العقد السودا على كل مسار من أي عقدة لأي NIL تحتها. وهي بتضمن الارتفاع ≤ 2 log₂(n+1). الإدخال: العقدة الجديدة حمراء ونعالج بالـ uncle: I1 (uncle أحمر): recolour، I2 (uncle أسود triangle): rotation تحوّلها لـ I3، I3 (uncle أسود line): recolour + rotation.

/* ---- RQ 14: properties and the insertion cases — every property is checked by rb_valid() after each step;
   here we additionally watch which cases fire for a few sequences ---- */
static void rq14(void)
{
    struct { const char *name; int keys[8]; int n; const char *cases; } T[] = {
        { "uncle is NIL (rotation, case 3)",        { 10, 5, 1 }, 3, "I3" },
        { "uncle is red (recolour, case 1)",        { 10, 5, 15, 1 }, 4, "I1" },
        { "triangle (double: case 2 then 3)",       { 10, 5, 7 }, 3, "I2 I3" },
    };
    for (int i = 0; i < 3; i++) {
        RBT t = { NIL, "" }; char sp[128];
        for (int j = 0; j < T[i].n; j++) rb_insert(&t, T[i].keys[j]);
        printf("RQ14: %-36s -> %-28s cases: %s\n", T[i].name, rb_str(&t, sp), t.log);
        assert(!strcmp(t.log, T[i].cases));
        assert(rb_valid(&t));
        rb_free(t.root);
    }
    /* TF 10 (False in the book): "some leaf nodes can be red".  In the book's drawings the leaves are the NIL
       externals, which are always black.  If you call the real nodes without children "leaves", RED ones DO exist:
       insert 10, 5, 15 -> both children are red leaves.  We print this to be precise about the wording. */
    RBT t = { NIL, "" };
    rb_insert(&t, 10); rb_insert(&t, 5); rb_insert(&t, 15);
    assert(t.root->l->color == RED && t.root->l->l == NIL && t.root->r->color == RED);
    assert(NIL->color == BLACK);                                 /* the external leaves are black */
    rb_free(t.root);
    printf("RQ14: root is black, every NIL leaf is black (Fill-in 8: black and black)\n");
}
RQ 15: splay

3 خطوات: zig / zig-zig / zig-zag. المثال: إدخال 1..7 يطلع chain، وsearch(1) بيعمل zig-zig ثلاث مرات والنتيجة 1(.,6(4(2(.,3),5),7)) (الارتفاع نزل من 7 لـ 5)، وبعدها search(2) يعمل zig-zig ثم zig: 2(1,4(3,6(5,7))) (ارتفاع 4).

/* ---- RQ 15, MCQ 5, Fill-in 9: splay trees ---- */
static void rq15(void)
{
    ST t = { NULL, "" };
    for (int i = 1; i <= 15; i++) s_insert(&t, i);
    t.log[0] = 0;
    s_search(&t, 1);                            /* MCQ 5: the accessed node becomes the ROOT */
    assert(t.root->key == 1);
    /* Fill-in 9: zig is used when the parent of N is the root */
    ST u = { NULL, "" };
    s_insert(&u, 5); s_insert(&u, 3);           /* 3 is now root, 5 its right child */
    u.log[0] = 0; s_search(&u, 5);
    assert(!strcmp(u.log, "zig"));
    /* zig-zig needs N and P on the same side, zig-zag on opposite sides */
    ST v = { NULL, "" };
    s_insert(&v, 1); s_insert(&v, 2); s_insert(&v, 3);          /* 3(2(1)) */
    v.log[0] = 0; s_search(&v, 1);
    assert(!strcmp(v.log, "zigzig"));
    ST w = { NULL, "" };                                        /* build 3(1(.,2)) by hand: 2 is a RIGHT child of a LEFT child */
    SN *a = s_new(3, NULL), *b = s_new(1, a), *c = s_new(2, b);
    a->l = b; b->r = c; w.root = a; w.log[0] = 0;
    splay(&w, c);
    assert(!strcmp(w.log, "zigzag") && w.root->key == 2);
    printf("RQ15: zig, zig-zig and zig-zag all observed; accessed key always ends at the root\n");
    s_free(t.root); s_free(u.root); s_free(v.root); s_free(w.root);
}

/* all distinct BSTs on the keys sorted in k[lo..hi]:  choose each key as root, combine every left shape
   with every right shape.  Writes specs into out[]; returns how many. */
static int all_bsts(const int *k, int lo, int hi, char out[][96])
{
    if (lo > hi) { strcpy(out[0], "."); return 1; }
    int cnt = 0;
    for (int r = lo; r <= hi; r++) {
        char L[16][96], R[16][96];
        int a = all_bsts(k, lo, r - 1, L), b = all_bsts(k, r + 1, hi, R);
        for (int i = 0; i < a; i++)
            for (int j = 0; j < b; j++) {
                if (L[i][0] == '.' && R[j][0] == '.') snprintf(out[cnt], 96, "%d", k[r]);
                else { snprintf(out[cnt], 96, "%d(", k[r]); strcat(out[cnt], L[i]); strcat(out[cnt], ","); strcat(out[cnt], R[j]); strcat(out[cnt], ")"); }
                cnt++;
            }
    }
    return cnt;
}

    INode *x = iparse("2(1,.)"); assert(BF(x) == 1); ifree(x);
