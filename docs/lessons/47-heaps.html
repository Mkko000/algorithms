3) الإدخال (Thareja Fig 12.7، Example 12.1، P&H Algorithm Insert)

نحط العنصر في أول خانة فاضية (الشجرة تفضل كاملة) وبعدين نطلّعه (percolate up / shift-up) طول ما هو أحسن من أبوه. أقصى حاجة = عمق الشجرة.

typedef struct { int *a; int n, cap; int mx; long cmp; } BHeap;   /* a[0] unused (or a sentinel); mx=1: max-heap */

static BHeap bh_new(int cap, int mx) { BHeap h = { calloc((size_t)cap + 2, sizeof(int)), 0, cap, mx, 0 }; return h; }
static void  bh_free(BHeap *h) { free(h->a); h->a = NULL; }
/* "x belongs above y": the one comparison the whole structure is made of */
static int   bh_above(BHeap *h, int x, int y) { h->cmp++; return h->mx ? x > y : x < y; }
static void  bh_swap(int *x, int *y) { int t = *x; *x = *y; *y = t; }

/* Thareja Fig 12.7: put VAL at the end (the tree stays complete), then let it rise while it beats its parent */
static int bh_insert(BHeap *h, int val)
{
    assert(h->n < h->cap);
    int pos = ++h->n;                                  /* Step 1 */
    h->a[pos] = val;                                   /* Step 2 */
    while (pos > 1) {                                  /* Steps 3-5 */
        int par = pos / 2;
        if (!bh_above(h, h->a[pos], h->a[par])) break;
        bh_swap(&h->a[pos], &h->a[par]);
        pos = par;
    }
    return pos;                                        /* where VAL finally rests */
}

/* Thareja Fig 12.10: remove the root. LAST replaces it and sinks. The book's Step 5 reads HEAP[RIGHT] even when
   RIGHT > N (a node with a single child). That works in the book because cell N+1 still holds LAST; here we test RIGHT <= N first, so the code never relies on a stale cell (see exercises/ex23_01). */
static int bh_delete_root(BHeap *h)
{
    assert(h->n > 0);
    int top = h->a[1], last = h->a[h->n--];            /* Step 1 */
    if (h->n == 0) return top;
    int ptr = 1, left = 2;                             /* Step 2 */
    h->a[ptr] = last;                                  /* Step 3 */
    while (left <= h->n) {                             /* Step 4 */
        int c = left;                                  /* the better child */
        if (left + 1 <= h->n && bh_above(h, h->a[left + 1], h->a[left])) c = left + 1;
        if (!bh_above(h, h->a[c], h->a[ptr])) break;   /* Step 5: PTR is at least as good as both children */
        bh_swap(&h->a[ptr], &h->a[c]);                 /* Steps 6-7 */
        ptr = c; left = 2 * ptr;
    }
    return top;
}

5) بناء الـ heap من n عنصر (Thareja Ex 12.2، Weiss 6.3.4 وFig 6.14، P&H، تمرين Weiss 6.2 وThareja Q29)

طريقتين:

  1. الإدخال واحد واحد (P&H بتسميها shift-up): n × O(log n) = O(n log n) في الأسوأ.
  2. Floyd (Weiss buildHeap، P&H shift-down): انسخ العناصر زي ما هي، وبعدين نزّل من i = n/2 لـ 1. O(n).
/* Pal & Halder 10.2 "shift-up": build by n insertions (top-down) */
static void bh_build_insert(BHeap *h, const int *v, int n) { for (int i = 0; i < n; i++) bh_insert(h, v[i]); }

static void bh_sift_down(BHeap *h, int i)
{
    int x = h->a[i];
    for (;;) {
        int c = 2 * i;
        if (c > h->n) break;
        if (c + 1 <= h->n && bh_above(h, h->a[c + 1], h->a[c])) c++;
        if (!bh_above(h, h->a[c], x)) break;
        h->a[i] = h->a[c]; i = c;                      /* the hole moves down, no swap needed (Weiss Fig 6.12) */
    }
    h->a[i] = x;
}
/* Weiss Fig 6.14 / P&H "shift-down": copy the items in any order, then fix the parents from n/2 down to 1 */
static void bh_build_floyd(BHeap *h, const int *v, int n)
{
    assert(n <= h->cap);
    memcpy(h->a + 1, v, (size_t)n * sizeof(int)); h->n = n;
    for (int i = n / 2; i >= 1; i--) bh_sift_down(h, i);
}

ليه Floyd خطي؟ العقدة في ارتفاع h بتتنزّل في أسوأ الأحوال h مستوى، وفي الشجرة الكاملة عدد العقد اللي ارتفاعها h حوالي n/2h+1. فالمجموع ≤ Σ h·n/2h+1 = n · Σ h/2h+1 = n. بالظبط: مجموع ارتفاعات كل العقد في شجرة كاملة ≤ n−1. وكل نزول مستوى = مقارنتين، فالحد الأعلى < 2n مقارنة (Thareja Q29)؛ اتجرّب لكل n من 1 لـ 995.

6) عمليات تانية لما نعرف مكان العنصر (Weiss 6.3.3، Thareja Q22)

الـ heap مش بيدعم البحث عن مفتاح (O(n))، لكن لو معانا المكان: decreaseKey (percolate up) وincreaseKey (percolate down) وremove (= decreaseKey بـ −∞ ثم deleteMin): كلها O(log n).

/* Weiss 6.3: when the position of an item is known, three more operations cost O(log N) */
static void bh_percolate_up(BHeap *h, int p)
{
    int x = h->a[p];
    for (; p > 1 && bh_above(h, x, h->a[p / 2]); p /= 2) h->a[p] = h->a[p / 2];
    h->a[p] = x;
}
static void bh_change_key(BHeap *h, int p, int delta, int toward_root)   /* decreaseKey in a min-heap = toward_root */
{
    assert(p >= 1 && p <= h->n && delta >= 0);
    int sign = h->mx ? 1 : -1;                         /* "better" direction */
    h->a[p] += (toward_root ? sign : -sign) * delta;
    if (toward_root) bh_percolate_up(h, p); else bh_sift_down(h, p);
}
/* remove(p): decreaseKey(p, infinity), then deleteMin */
static void bh_remove(BHeap *h, int p)
{
    assert(p >= 1 && p <= h->n);
    h->a[p] = h->mx ? 2147483647 : -2147483647 - 1;    /* "infinity" in the good direction */
    bh_percolate_up(h, p);
    bh_delete_root(h);
}

7) d-heap (Weiss 6.14 و6.15)

كل عقدة لها d ابن. الارتفاع logd n فالإدخال أسرع (مستويات أقل)، لكن الحذف بيقارن مع d ابن في كل مستوى (d·logd n). المواقع (الترقيم من 1): parent(i) = ⌊(i−2)/d⌋ + 1 وchild k of i = d(i−1) + 2 + k للـ k = 0…d−1.

/* Weiss 6.5 / Exercise 6.14: a d-heap in an array a[1..n]. Children of i: d(i-1)+2 .. d(i-1)+d+1; parent: (i-2)/d + 1 */
static int dh_parent(int i, int d) { return (i - 2) / d + 1; }
static int dh_child(int i, int d, int k) { return d * (i - 1) + 2 + k; }          /* k = 0..d-1 */
typedef struct { int *a; int n, cap, d; long cmp; } DHeap;
static DHeap dh_new(int cap, int d) { DHeap h = { calloc((size_t)cap + 2, sizeof(int)), 0, cap, d, 0 }; return h; }
static void dh_insert(DHeap *h, int v)                 /* min-heap */
{
    int p = ++h->n;
    while (p > 1) { int q = dh_parent(p, h->d); h->cmp++; if (v >= h->a[q]) break; h->a[p] = h->a[q]; p = q; }
    h->a[p] = v;
}
static int dh_delete_min(DHeap *h)
{
    int top = h->a[1], x = h->a[h->n--], i = 1;
    if (h->n == 0) return top;
    for (;;) {
        int first = dh_child(i, h->d, 0);
        if (first > h->n) break;
        int best = first;
        for (int k = 1; k < h->d && first + k <= h->n; k++) { h->cmp++; if (h->a[first + k] < h->a[best]) best = first + k; }
        h->cmp++;
        if (h->a[best] >= x) break;
        h->a[i] = h->a[best]; i = best;
    }
    h->a[i] = x;
    return top;
}
static int dh_valid(const DHeap *h)
{
    for (int i = 2; i <= h->n; i++) if (h->a[dh_parent(i, h->d)] > h->a[i]) return 0;
    return 1;
}

8) Min-max heap (Weiss 6.18، Fig 6.58)

عايزين deleteMin وdeleteMax في نفس الهيكل. نفس شكل الـ heap بس: مستويات العمق الزوجي (0، 2، …) min levels (العقدة أصغر من كل شجرتها الفرعية)، والفردي max levels (أكبر من كل شجرتها الفرعية). الـ min = a[1] والـ max = الأكبر بين a[2] وa[3]، والتلاتة O(1)، والإدخال والحذفين O(log N).

#include <assert.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct { int a[512]; int n; } MMHeap;                 /* a[0] unused, as in the book's binary heap */

static int mm_level_is_min(int i) { int d = 0; while (i > 1) { i >>= 1; d++; } return d % 2 == 0; }
static int mm_min(const MMHeap *h) { return h->a[1]; }
static int mm_max_index(const MMHeap *h) { return h->n == 1 ? 1 : (h->n == 2 || h->a[2] >= h->a[3]) ? 2 : 3; }
static int mm_max(const MMHeap *h) { return h->a[mm_max_index(h)]; }

static void swap(int *x, int *y) { int t = *x; *x = *y; *y = t; }

/* bubble up on the levels of the same kind: grandparent (i/4) */
static void mm_up_same(MMHeap *h, int i, int min_level)
{
    while (i >= 4 && (min_level ? h->a[i] < h->a[i / 4] : h->a[i] > h->a[i / 4])) { swap(&h->a[i], &h->a[i / 4]); i /= 4; }
}
static void mm_bubble_up(MMHeap *h, int i)
{
    if (i == 1) return;
    int p = i / 2;
    if (mm_level_is_min(i)) {
        if (h->a[i] > h->a[p]) { swap(&h->a[i], &h->a[p]); mm_up_same(h, p, 0); }   /* belongs on a max level */
        else mm_up_same(h, i, 1);
    } else {
        if (h->a[i] < h->a[p]) { swap(&h->a[i], &h->a[p]); mm_up_same(h, p, 1); }
        else mm_up_same(h, i, 0);
    }
}
static void mm_insert(MMHeap *h, int x) { assert(h->n < 511); h->a[++h->n] = x; mm_bubble_up(h, h->n); }

/* the smallest (min level) or largest (max level) key among the children and grandchildren of i; 0 if none */
static int mm_best_descendant(const MMHeap *h, int i, int want_min)
{
    int best = 0, c[6] = { 2 * i, 2 * i + 1, 4 * i, 4 * i + 1, 4 * i + 2, 4 * i + 3 };
    for (int k = 0; k < 6; k++) {
        if (c[k] > h->n) continue;
        if (!best || (want_min ? h->a[c[k]] < h->a[best] : h->a[c[k]] > h->a[best])) best = c[k];
    }
    return best;
}
static void mm_trickle_down(MMHeap *h, int i)
{
    int min_level = mm_level_is_min(i);
    for (;;) {
        int m = mm_best_descendant(h, i, min_level);
        if (!m) return;
        if (m >= 4 * i) {                                                  /* m is a grandchild */
            if (min_level ? h->a[m] < h->a[i] : h->a[m] > h->a[i]) {
                swap(&h->a[m], &h->a[i]);
                int p = m / 2;                                             /* fix the parent of m on the other kind of level */
                if (min_level ? h->a[m] > h->a[p] : h->a[m] < h->a[p]) swap(&h->a[m], &h->a[p]);
                i = m;
            } else return;
        } else {                                                           /* m is a child: the last possible move */
            if (min_level ? h->a[m] < h->a[i] : h->a[m] > h->a[i]) swap(&h->a[m], &h->a[i]);
            return;
        }
    }
}
static int mm_delete_min(MMHeap *h) { assert(h->n); int r = h->a[1]; h->a[1] = h->a[h->n--]; if (h->n) mm_trickle_down(h, 1); return r; }
static int mm_delete_max(MMHeap *h)
{
    assert(h->n); int i = mm_max_index(h), r = h->a[i];
    h->a[i] = h->a[h->n--]; if (i <= h->n) mm_trickle_down(h, i);
    return r;
}
/* the definition, checked literally: min-level keys <= everything below them, max-level keys >= everything below */
static int mm_subtree_ok(const MMHeap *h, int i, int j)                    /* is key i ordered against node j ? */
{
    return mm_level_is_min(i) ? h->a[i] <= h->a[j] : h->a[i] >= h->a[j];
}
static int mm_valid(const MMHeap *h)
{
    for (int i = 1; i <= h->n; i++) {

            if (2 * j <= h->n) stack[sp++] = 2 * j;
            if (2 * j + 1 <= h->n) stack[sp++] = 2 * j + 1; }
    }
    return 1;
}

9) Leftist heaps (Weiss 6.6، Fig 6.20–6.30، تمارين 6.19–6.25)

عايزين merge سريع. الـ binary heap بياخد O(n). الفكرة: شجرة heap-ordered (مش لازم كاملة)، لكن نخليها مائلة للشمال عشان المسار اليمين يبقى قصير.

null path length: npl(x) = طول أقصر مسار من x لعقدة فيها ابن فاضي. npl(NULL) = −1. الشرط: لكل عقدة npl(left) ≥ npl(right). النتيجة: أقصر مسار لـ NULL هو المسار اليمين، وعقدة npl = r فيها على الأقل 2r+1−1 عقدة، فالمسار اليمين في heap بـ N عقدة ≤ ⌊log₂(N+1)⌋ (اتأكد بـ assert في كل اختبار).

typedef struct LNode { int key, npl; struct LNode *left, *right; } LNode;     /* npl = null path length; npl(NULL) = -1 */
static long lh_steps;                                  /* how many nodes the merges visited (cost counter) */
static int npl(const LNode *t) { return t ? t->npl : -1; }
static LNode *lh_node(int key) { LNode *t = malloc(sizeof *t); t->key = key; t->npl = 0; t->left = t->right = NULL; return t; }
/* Weiss Fig 6.26/6.27. merge1 handles the case h1->key <= h2->key. */
static LNode *lh_merge(LNode *h1, LNode *h2);
static LNode *lh_merge1(LNode *h1, LNode *h2)
{
    if (h1->left == NULL)                              /* a single node: h2 simply becomes its left child */
        h1->left = h2;
    else {
        h1->right = lh_merge(h1->right, h2);           /* the recursion goes down the RIGHT path only */
        if (npl(h1->left) < npl(h1->right)) { LNode *t = h1->left; h1->left = h1->right; h1->right = t; }  /* leftist rule */
        h1->npl = npl(h1->right) + 1;
    }
    return h1;
}
static LNode *lh_merge(LNode *h1, LNode *h2)
{
    lh_steps++;
    if (!h1) return h2;
    if (!h2) return h1;
    return h1->key < h2->key ? lh_merge1(h1, h2) : lh_merge1(h2, h1);
}
static LNode *lh_insert(LNode *h, int x) { return lh_merge(lh_node(x), h); }
static LNode *lh_delete_min(LNode *h, int *out)        /* Fig 6.30: the root goes, its two subheaps are merged */
{
    LNode *l = h->left, *r = h->right; *out = h->key; free(h);
    return lh_merge(l, r);
}

/* Weiss Fig 6.28 and the text after it: do it in two passes over the right path, with no recursion.
   Pass 1 merges the two right paths into one sorted path; pass 2 walks back up fixing npl and swapping children. */
static LNode *lh_merge_iter(LNode *a, LNode *b)
{
    LNode *path[128]; int np = 0;
    while (a && b) {                                   /* pass 1: detach nodes from the right paths in key order */
        LNode **w = a->key < b->key ? &a : &b;
        LNode *n = *w; *w = n->right; n->right = NULL;
        path[np++] = n;
    }
    LNode *rest = a ? a : b;                           /* what is left of one path is already in order */
    while (rest) { LNode *n = rest; rest = n->right; n->right = NULL; path[np++] = n; }
    LNode *below = NULL;                               /* pass 2: from the bottom of the merged path up */
    for (int i = np - 1; i >= 0; i--) {
        LNode *n = path[i]; n->right = below;
        if (npl(n->left) < npl(n->right)) { LNode *t = n->left; n->left = n->right; n->right = t; }
        n->npl = npl(n->right) + 1;
        below = n;
    }
    return below;
}

10) Skew heaps (Weiss 6.7، Fig 6.31–6.33، تمارين 6.26–6.29)

نفس الـ merge لكن من غير npl: بعد الدمج في الابن اليمين نبدّل الابنين دايمًا. مفيش ضمان على التكلفة في عملية واحدة (ممكن O(N))، لكن amortized O(log N). مميزاته: كود أقصر، ومفيش حقل إضافي.

/* Weiss 6.7: no npl at all. After merging into the right subtree, ALWAYS swap the two children
   (the book swaps every node on the right path except the last one; this recursive form is the same thing). */
static LNode *sk_merge(LNode *h1, LNode *h2)
{
    lh_steps++;
    if (!h1) return h2;
    if (!h2) return h1;
    if (h2->key < h1->key) { LNode *t = h1; h1 = h2; h2 = t; }
    LNode *r = sk_merge(h1->right, h2);
    h1->right = h1->left; h1->left = r;                /* swap, with the merged heap landing on the left */
    return h1;
}
static LNode *sk_insert(LNode *h, int x) { return sk_merge(lh_node(x), h); }
static LNode *sk_delete_min(LNode *h, int *out)
{
    LNode *l = h->left, *r = h->right; *out = h->key; free(h);
    return sk_merge(l, r);
}

11) Binomial heaps / queues (Thareja 12.2، Weiss 6.8، تمارين 6.30–6.36)

كل الـ heaps اللي فاتت ما بتديش insert بـ O(1) في أسوأ حالة. الـ binomial queue هو «عدّاد ثنائي» من الأشجار:

الشجرة binomial Bk = شجرتين Bk−1 واحدة معلّقة تحت جذر التانية. فيها 2k عقدة، وجذرها له k ابن (Bk−1 … B0)، وعدد العقد على العمق d هو C(k, d) (ومن هنا الاسم binomial). (الكتاب Thareja في Table و T/F 6 بيطبع «2i» وده أكيد 2i اتحوّلت في الاستخراج؛ والـ T/F رقم 6 «Bi فيها 2i عقدة» مفتاحها False.)

الـ heap = مجموعة أشجار binomial، أقصى واحدة من كل رتبة k، كل واحدة heap-ordered. فعدد العقد N بالثنائي يحدد الأشجار: N = 12 = 1100₂ يعني B3 وB2 (Fig 12.13: 8 + 4 عقد).

typedef struct BNode { int val, degree; struct BNode *parent, *child, *sibling; } BNode;
static long bn_links;                                  /* number of Link calls = number of key comparisons that built trees */
static BNode *bn_node(int val) { BNode *x = calloc(1, sizeof *x); x->val = val; return x; }

/* Thareja Fig 12.15: the minimum is one of the roots, so look at all of them: O(log n) */
static BNode *bn_min(BNode *head)
{
    BNode *y = NULL, *x = head; int min = INT_MAX;     /* Step 1 */
    while (x) {                                        /* Steps 2-4 */
        if (x->val < min) { min = x->val; y = x; }
        x = x->sibling;
    }
    return y;                                          /* Step 5 */
}

/* Thareja Fig 12.17: make Y (a B(i-1)) the leftmost child of Z (another B(i-1)): the result is a B(i) */
static void bn_link(BNode *y, BNode *z)
{
    y->parent = z;                                     /* Step 1 */
    y->sibling = z->child;                             /* Step 2 */
    z->child = y;                                      /* Step 3 */
    z->degree++;                                       /* Step 4 */
    bn_links++;
}

دمج (Union) = جمع ثنائي مع carry: ندمج قايمتين الجذور كأنهم مرتبين بالـ degree، وبعدين نمشي: لو فيه شجرتين بنفس الرتبة نربطهم (الجذر الأصغر يفضل جذر ويبقى التاني ابنه)، ويطلع «carry» بالرتبة اللي بعدها. Thareja بتسميها Link (Fig 12.17) وUnion (Fig 12.18)، وWeiss Fig 6.55.

/* the first half of Union: merge the two root lists into one list sorted by degree (like merging two sorted lists) */
static BNode *bn_merge_roots(BNode *a, BNode *b)
{
    BNode dummy = { 0 }, *t = &dummy;
    while (a && b) {
        if (a->degree <= b->degree) { t->sibling = a; a = a->sibling; } else { t->sibling = b; b = b->sibling; }
        t = t->sibling;
    }
    t->sibling = a ? a : b;
    return dummy.sibling;
}

/* Thareja Fig 12.18. PTR is the root being examined, NEXT the one after it, PREV the one before. */
static BNode *bn_union(BNode *h1, BNode *h2)
{
    BNode *head = bn_merge_roots(h1, h2);              /* Steps 1-3 */
    if (!head) return NULL;                            /* Step 4 */
    BNode *prev = NULL, *ptr = head, *next = ptr->sibling;   /* Step 5 */
    while (next) {                                     /* Step 6 */
        if (ptr->degree != next->degree ||
            (next->sibling && next->sibling->degree == ptr->degree)) {   /* Step 7: nothing to link here, move on */
            prev = ptr; ptr = next;
        } else if (ptr->val <= next->val) {            /* two equal degrees: the smaller key stays the root */
            ptr->sibling = next->sibling;
            bn_link(next, ptr);
        } else {
            if (!prev) head = next; else prev->sibling = next;
            bn_link(ptr, next);
            ptr = next;
        }
        next = ptr->sibling;
    }
    return head;                                       /* Step 8 */
}
static BNode *bn_insert(BNode *head, BNode *x)         /* Fig 12.20: a one-node heap united with H */
{
    x->parent = x->child = x->sibling = NULL; x->degree = 0;
    return bn_union(head, x);
}

insert = union مع heap من عقدة واحدة. extract-min (Thareja 12.21، Weiss Fig 6.56) = نشيل أصغر جذر؛ أبناؤه Bk−1 … B0 هم أصلاً binomial heap كامل (بعد عكس الترتيب) فنعمل union معاه ومع الباقي. وdecrease بنطلّع القيمة بالتبديل مع الأب، وdelete = decrease لـ −∞ ثم extract-min.

/* Thareja Fig 12.21: remove the smallest root R; R's children, reversed, become a heap of their own; unite. */
static BNode *bn_extract_min(BNode *head, BNode **out)
{
    BNode *r = bn_min(head);                           /* Step 1 */
    assert(r);
    if (head == r) head = r->sibling;                  /* Step 2: unlink R from the root list */
    else { BNode *p = head; while (p->sibling != r) p = p->sibling; p->sibling = r->sibling; }
    BNode *rev = NULL, *c = r->child;                  /* Steps 3-5: reverse the child list (degrees now rise) */
    while (c) { BNode *nx = c->sibling; c->sibling = rev; c->parent = NULL; rev = c; c = nx; }
    r->child = NULL; r->sibling = NULL; r->degree = 0;
    *out = r;
    return bn_union(head, rev);                        /* Step 6 */
}

/* Thareja Fig 12.23: lower a value and let it rise by swapping values with its parent.
   (Swapping the VALUES means a node pointer no longer names "the same item"; Weiss/CLRS note this too.) */
static void bn_decrease(BNode *x, int k)
{
    assert(k <= x->val);
    x->val = k;
    BNode *ptr = x, *par = ptr->parent;
    while (par && ptr->val < par->val) { int t = ptr->val; ptr->val = par->val; par->val = t; ptr = par; par = ptr->parent; }
}
/* Fig 12.24: delete = decrease to minus infinity, then extract the minimum */
static BNode *bn_delete(BNode *head, BNode *x)
{
    BNode *out;
    bn_decrease(x, INT_MIN);
    head = bn_extract_min(head, &out);
    free(out);
    return head;
}

12) Fibonacci heaps (Thareja 12.3، Examples 12.8–12.10، Q30)

الفكرة: نأجّل الشغل. قايمة جذور (circular doubly linked) بدون قيود، والـ min بمؤشر. insert وunion وfind-min = O(1) حقيقية (Thareja Table 12.1 بتكتب find-min وcreate بـ O(n)؛ وده غلط طباعي واضح: الاتنين O(1)). وكل الشغل بيتعمل في extract-min بـ consolidate: نربط الجذور بنفس الرتبة (زي binomial) لحد ما رتب الجذور تبقى مختلفة. ودرجة أي عقدة في heap بـ n عقدة ≤ logφ n (φ = 1.618…).

typedef struct FNode { int key, degree, mark; struct FNode *parent, *child, *left, *right; } FNode;
typedef struct { FNode *min; int n; long links, cuts; FNode *start; } FHeap;   /* min[H] and n[H]; start = where a hand-drawn root list begins */

static FNode *fn_new(int key) { FNode *x = calloc(1, sizeof *x); x->key = key; x->left = x->right = x; return x; }

/* put x (a whole circular list of one) next to y in y's list: O(1)  — "concatenate" in the book */
static void list_splice(FNode *y, FNode *x)           /* x's list is merged into y's list */
{
    FNode *yr = y->right, *xl = x->left;
    y->right = x; x->left = y; xl->right = yr; yr->left = xl;
}
static void list_remove(FNode *x) { x->left->right = x->right; x->right->left = x->left; x->left = x->right = x; }

/* Fig 12.28 */
static void fh_insert(FHeap *h, FNode *x)
{
    x->degree = 0; x->parent = x->child = NULL; x->mark = 0; x->left = x->right = x;     /* Steps 1-2 */
    if (h->min) list_splice(h->min, x);                                                  /* Step 3 */
    if (!h->min || x->key < h->min->key) h->min = x;                                     /* Step 4 */
    h->n++;                                                                              /* Step 5 */
}
/* Fig 12.30: the two root lists are joined into one in O(1); h2 is left empty */
static void fh_union(FHeap *h1, FHeap *h2)
{
    if (!h1->min) h1->min = h2->min;
    else if (h2->min) { list_splice(h1->min, h2->min); if (h2->min->key < h1->min->key) h1->min = h2->min; }
    h1->n += h2->n; h1->links += h2->links; h1->cuts += h2->cuts;
    h2->min = NULL; h2->n = 0;
}
/* Fig 12.33: remove y from the root list and make it a child of x */
static void fh_link(FHeap *h, FNode *y, FNode *x)
{
    list_remove(y);                                    /* Step 1 */
    y->parent = x;                                     /* Step 2 */
    if (!x->child) x->child = y; else list_splice(x->child, y);
    x->degree++;                                       /* Step 3 */
    y->mark = 0;                                       /* Step 4 */
    h->links++;
}
/* Fig 12.32: until every root has its own degree */
static void fh_consolidate(FHeap *h)
{
    FNode *A[64] = { 0 };                              /* A[d] = the root of degree d seen so far (D(n) <= log_phi n < 64) */
    FNode **roots = malloc(sizeof(FNode *) * (size_t)(h->n + 2)); int nr = 0;   /* at most n roots */
    /* the root list is circular and has no first element; the book simply walks it from the left of its picture.
       h->start remembers that left end when the heap was written down by hand, otherwise we begin at min[H]. */
    FNode *begin = (h->start && !h->start->parent) ? h->start : h->min, *x = begin;
    do { roots[nr++] = x; x = x->right; } while (x != begin);    /* take a snapshot: the list changes while we link */
    h->start = NULL;
    for (int i = 0; i < nr; i++) {
        FNode *ptr = roots[i]; int d = ptr->degree;
        while (A[d]) {
            FNode *temp = A[d];
            if (ptr->key > temp->key) { FNode *t = ptr; ptr = temp; temp = t; }    /* the smaller key becomes the parent */
            fh_link(h, temp, ptr);
            A[d] = NULL; d++;
        }
        A[d] = ptr;
    }
    free(roots);
    h->min = NULL;                                     /* Steps 13-18: rebuild the root list and min[H] from A[] */
    for (int i = 0; i < 64; i++)
        if (A[i]) {
            A[i]->left = A[i]->right = A[i];
            if (!h->min) h->min = A[i]; else { list_splice(h->min, A[i]); if (A[i]->key < h->min->key) h->min = A[i]; }
        }
}
/* Fig 12.31 */
static FNode *fh_extract_min(FHeap *h)
{
    FNode *x = h->min;
    if (!x) return NULL;
    if (x->child) {                                    /* every child becomes a root */
        FNode *c = x->child;
        do { c->parent = NULL; c = c->right; } while (c != x->child);
        list_splice(x, x->child);                      /* the child list joins the root list next to x */
        x->child = NULL; x->degree = 0;
    }
    FNode *r = x->right;
    if (h->start == x) h->start = (x->right == x) ? NULL : x->right;
    list_remove(x);                                    /* remove x from the root list */
    h->n--;
    if (r == x) h->min = NULL;                         /* x was the only node */
    else { h->min = r; fh_consolidate(h); }
    return x;
}

decrease-key: بنقلّل القيمة، ولو بقت أصغر من الأب نقطع العقدة ونحطها في قايمة الجذور (cut). ولو الأب كان معلّم (فقد ابن قبل كده) يتقطع كمان، وهكذا للأعلى (cascading cut). العلامات (marks) هي اللي بتضمن إن العقدة اللي رتبتها k فيها على الأقل Fk+2 عقدة، فالارتفاع لوغاريتمي. amortized O(1) للـ decrease-key، وO(log n) للـ delete وextract-min.

/* Fig 12.36 and 12.37 */
static void fh_cut(FHeap *h, FNode *x, FNode *par)
{
    if (par->child == x) par->child = (x->right == x) ? NULL : x->right;     /* Step 1 */
    list_remove(x);
    par->degree--;                                     /* Step 2 */
    list_splice(h->min, x);                            /* Step 3 */
    x->parent = NULL;                                  /* Step 4 */
    x->mark = 0;                                       /* Step 5 */
    h->cuts++;
}
static void fh_cascading_cut(FHeap *h, FNode *x)
{
    FNode *par = x->parent;
    if (par) {
        if (!x->mark) x->mark = 1;                     /* the first child lost: just remember it */
        else { fh_cut(h, x, par); fh_cascading_cut(h, par); }    /* the second one: x leaves too, and so on upwards */
    }
}
/* Fig 12.35 */
static void fh_decrease(FHeap *h, FNode *x, int v)
{
    assert(v <= x->key);
    x->key = v;
    FNode *par = x->parent;
    if (par && x->key < par->key) { fh_cut(h, x, par); fh_cascading_cut(h, par); }
    if (x->key < h->min->key) h->min = x;
}
/* Fig 12.39 */
static void fh_delete(FHeap *h, FNode *x)
{
    fh_decrease(h, x, INT_MIN);
    FNode *r = fh_extract_min(h);
    assert(r == x);
    free(r);
}
