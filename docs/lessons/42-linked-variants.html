2) الـ Header node: امسح الحالات الخاصة

مع القايمة العادية، إدخال أو حذف أول عقدة حالة خاصة لأن head نفسه بيتغيّر. لو في الأول عقدة وهمية ما بتتحذفش أبدًا، كل عقدة بيانات ليها «سابق» وكل العمليات بتبقى نفس الكود. وفي الـ header بنقدر نخزّن عدد العقد (ملحوظة Thareja وLipschutz) فالـ length بقى O(1).

typedef struct cnode { int val; struct cnode *next; } cnode;

/* the header's val field is free: many books keep the NUMBER OF NODES there. We do too,
   so length is O(1)  (header->val == number of data nodes). */
static cnode *ch_new(void)
{
    cnode *h = malloc(sizeof *h);
    if (!h) { perror("malloc"); exit(1); }
    h->val = 0; h->next = h;                 /* empty: points at itself */
    return h;
}
static cnode *ch_insert_after(cnode *h, cnode *p, int v)   /* p is the header or any node */
{
    cnode *n = malloc(sizeof *n);
    if (!n) { perror("malloc"); exit(1); }
    n->val = v; n->next = p->next; p->next = n;
    h->val++;
    return n;
}
static cnode *ch_last(cnode *h) { cnode *p = h; while (p->next != h) p = p->next; return p; }
static void ch_push_front(cnode *h, int v) { ch_insert_after(h, h, v); }
static void ch_push_back(cnode *h, int v)  { ch_insert_after(h, ch_last(h), v); }

/* delete the node AFTER p (p = predecessor). The only deletion primitive we need. */
static int ch_delete_after(cnode *h, cnode *p)
{
    cnode *t = p->next;
    if (t == h) return 0;                    /* nothing to delete (UNDERFLOW) */
    p->next = t->next;
    free(t); h->val--;
    return 1;
}

لاحظ إن ch_delete_after هي الأداة الوحيدة للحذف: كل حذف تاني (أول، آخر، k-th، بالقيمة) هو إيجاد السابق وبعدين استدعاءها.

static int ch_delete_first(cnode *h) { return ch_delete_after(h, h); }
static int ch_delete_last(cnode *h)
{
    if (h->next == h) return 0;
    cnode *p = h;
    while (p->next->next != h) p = p->next;  /* p = predecessor of the last node */
    return ch_delete_after(h, p);
}
/* k-th node, k = 1,2,...  (Thareja PE 24: delete the k-th node) */
static int ch_delete_kth(cnode *h, int k)
{
    if (k < 1 || k > h->val) return 0;
    cnode *p = h;
    while (--k) p = p->next;
    return ch_delete_after(h, p);
}
/* delete all nodes with negative data (PE 39) in one pass with a trailing pointer */
static int ch_delete_negatives(cnode *h)
{
    int removed = 0;
    for (cnode *p = h; p->next != h; ) {
        if (p->next->val < 0) { ch_delete_after(h, p); removed++; }
        else p = p->next;                    /* only advance when we did NOT delete */
    }
    return removed;
}
static int ch_count_nonzero(const cnode *h)  /* PE 27 */
{
    int c = 0;
    for (const cnode *p = h->next; p != h; p = p->next) c += p->val != 0;
    return c;
}
static cnode *ch_copy(const cnode *h)        /* PE 6 */
{
    cnode *c = ch_new(), *tail = c;
    for (const cnode *p = h->next; p != h; p = p->next) tail = ch_insert_after(c, tail, p->val);
    return c;
}
static void ch_free(cnode *h) { while (ch_delete_first(h)) ; free(h); }
فخ حذف داخل لفّةفي ch_delete_negatives مش بنتقدّم بـ p = p->next بعد الحذف، لأن العقدة اللي بعد p اتغيّرت وممكن تكون هي كمان سالبة (مثال: -1 -2 -3 ورا بعض). لو تقدّمت هتفوّت واحدة. اتقدّم بس لما ما تحذفش.

3) الـ Josephus على قايمة دايرية

n ناس في دايرة، كل m-ـة بتتشال لحد ما يفضل واحد. بالقايمة الدايرية: ندور m−1 خطوة ونحذف التالي. التكلفة O(n·m). فيه كمان علاقة تكرارية J(i) = (J(i−1) + m) mod i بتدّي الإجابة في O(n) من غير ما نبني قايمة أصلًا. البرنامج بيتأكد إن الاتنين بيدّوا نفس الرقم لكل n ≤ 40 وm ≤ 9 (والحالة التاريخية J(41,3) = 31).

/* n people stand in a circle; every m-th is eliminated until one is left.
   Returns the survivor's number (1-based). The circular list makes "skip m−1, remove m-th"
   natural; cost O(n·m). A closed form exists: J(1)=0, J(i)=(J(i−1)+m) mod i  (O(n)). */
static int josephus_list(int n, int m)
{
    cnode *h = ch_new();
    for (int i = 1; i <= n; i++) ch_push_back(h, i);
    cnode *p = h;                            /* invariant: p->next is the person being counted */
    while (h->val > 1) {
        for (int step = 1; step < m; step++) {
            p = p->next;                     /* count one person...                          */
            if (p->next == h) p = h;         /* ...and wrap: the successor of the last is the first */
        }
        ch_delete_after(h, p);               /* eliminate the m-th person                    */
        if (p->next == h) p = h;             /* the next count starts after the victim       */
    }
    int survivor = h->next->val;
    ch_free(h);
    return survivor;
}
static int josephus_formula(int n, int m)
{
    int j = 0;
    for (int i = 2; i <= n; i++) j = (j + m) % i;
    return j + 1;
}

4) القايمة المزدوجة (doubly) الدايرية بـ header

أقوى نسخة: كل عقدة ليها prev وnext، والـ header بيقفل الدايرة. النتيجة: مفيش NULL في أي مكان وبالتالي مفيش if (p == NULL) في الكود.

typedef struct dnode { int val; struct dnode *prev, *next; } dnode;
typedef struct { dnode head; int size; } dlist;     /* head is the sentinel (its val is unused) */

static void dl_init(dlist *l) { l->head.prev = l->head.next = &l->head; l->size = 0; }
static int  dl_empty(const dlist *l) { return l->head.next == &l->head; }

/* insert a new node holding v right AFTER node p (p may be the sentinel): 4 pointer writes */
static dnode *dl_insert_after(dlist *l, dnode *p, int v)
{
    dnode *n = malloc(sizeof *n);
    if (!n) { perror("malloc"); exit(1); }
    n->val = v;
    n->prev = p;  n->next = p->next;
    p->next->prev = n;  p->next = n;
    l->size++;
    return n;
}
static dnode *dl_push_front(dlist *l, int v) { return dl_insert_after(l, &l->head, v); }
static dnode *dl_push_back (dlist *l, int v) { return dl_insert_after(l, l->head.prev, v); }

/* unlink a node (not the sentinel!) and return it WITHOUT freeing: lets us "move" nodes */
static dnode *dl_unlink(dlist *l, dnode *p)
{
    assert(p != &l->head);
    p->prev->next = p->next;  p->next->prev = p->prev;
    l->size--;
    return p;
}
static void dl_erase(dlist *l, dnode *p) { free(dl_unlink(l, p)); }

العمليات المهمة: القلب بتبديل prev/next في كل عقدة (بما فيها الـ header)، والدمج O(1) لأن بنوصّل آخر a بأول b بس، ونقل عقدة للأول من غير allocation.

static dnode *dl_find(dlist *l, int v)
{
    for (dnode *p = l->head.next; p != &l->head; p = p->next) if (p->val == v) return p;
    return NULL;
}
static void dl_clear(dlist *l) { while (!dl_empty(l)) dl_erase(l, l->head.next); }

/* reverse: swap prev/next in EVERY node including the sentinel */
static void dl_reverse(dlist *l)
{
    dnode *p = &l->head;
    do { dnode *t = p->next; p->next = p->prev; p->prev = t; p = t; } while (p != &l->head);
}
/* concatenate b onto the end of a in O(1); b becomes empty (Thareja PE 12) */
static void dl_concat(dlist *a, dlist *b)
{
    if (dl_empty(b)) return;
    dnode *al = a->head.prev, *bf = b->head.next, *bl = b->head.prev;
    al->next = bf;  bf->prev = al;
    bl->next = &a->head;  a->head.prev = bl;
    a->size += b->size;
    dl_init(b);
}
/* delete the first node and add it as the last one: no allocation, just relinking (PE 13) */
static void dl_rotate_left(dlist *l)
{
    if (l->size < 2) return;
    dnode *f = dl_unlink(l, l->head.next);
    f->prev = l->head.prev;  f->next = &l->head;
    l->head.prev->next = f;  l->head.prev = f;
    l->size++;
}
/* move an existing node to the front (PE 32 moves the middle node) */
static void dl_move_to_front(dlist *l, dnode *p)
{
    dl_unlink(l, p);
    p->prev = &l->head;  p->next = l->head.next;
    l->head.next->prev = p;  l->head.next = p;
    l->size++;
}

تبديل عقدتين بالـ relinking والترتيب

لما العقدة كبيرة (record)، تبديل البيانات غالي. تبديل الروابط بيكلّف ثابت. وقاعدة ترتيب الإدراج (insertion sort) على القايمة المزدوجة بتمشي لليسار من العقدة لحد مكانها — اللي ممكن لأن عندنا prev.

/* swap node p with its successor by RELINKING (the data stays where it is): PE 36 */
static void dl_swap_with_next(dlist *l, dnode *p)
{
    dnode *q = p->next;
    assert(p != &l->head && q != &l->head);
    p->prev->next = q;   q->prev = p->prev;
    p->next = q->next;   q->next->prev = p;
    q->next = p;         p->prev = q;
    (void)l;
}
/* insertion sort by relinking: stable, O(n^2) worst case, O(n) on sorted input (PE 8) */
static void dl_sort(dlist *l)
{
    dnode *p = l->head.next;
    while (p != &l->head) {
        dnode *nxt = p->next;
        dnode *q = p->prev;
        while (q != &l->head && q->val > p->val) q = q->prev;      /* find p's place going left */
        if (q != p->prev) {                                         /* only move if necessary    */
            dl_unlink(l, p);
            p->prev = q;  p->next = q->next;
            q->next->prev = p;  q->next = p;
            l->size++;
        }
        p = nxt;
    }
}

5) خوارزميات القوايم الكلاسيكية

القلب (reverse)

/* three pointers: prev, cur, next. Each link is flipped exactly once: O(n) time, O(1) space. */
static node *reverse_iter(node *h)
{
    node *prev = NULL;
    while (h) {
        node *next = h->next;   /* 1. remember the rest      */
        h->next = prev;         /* 2. flip this link         */
        prev = h;               /* 3. advance both           */
        h = next;
    }
    return prev;
}
/* recursion: reverse the tail, then hang the head after the (new) last node. O(n) stack. */
static node *reverse_rec(node *h)
{
    if (!h || !h->next) return h;
    node *rest = reverse_rec(h->next);
    h->next->next = h;
    h->next = NULL;
    return rest;
}

الدمج والتقسيم

/* merge two SORTED lists by relinking nodes (no allocation). Stable: ties take from a first. */
static node *merge_sorted(node *a, node *b)
{
    node dummy; node *t = &dummy; dummy.next = NULL;
    while (a && b) {
        if (b->val < a->val) { t->next = b; b = b->next; }
        else                 { t->next = a; a = a->next; }
        t = t->next;
    }
    t->next = a ? a : b;
    return dummy.next;
}
/* split into two lists: nodes 1,3,5,... go to *odd and 2,4,6,... to *even (relinking only) */
static void split_alternate(node *h, node **odd, node **even)
{
    node od, ev; node *to = &od, *te = &ev; od.next = ev.next = NULL;
    int i = 0;
    while (h) {
        node *next = h->next;
        h->next = NULL;
        if (i++ % 2 == 0) { to->next = h; to = h; } else { te->next = h; te = h; }
        h = next;
    }
    *odd = od.next; *even = ev.next;
}

عقدة dummy على الـ stack بتلغي حالة «أول عقدة في النتيجة». ده نفس فكرة الـ header بس مؤقتة.

مؤشرين بسرعتين

/* slow/fast pointers: when fast reaches the end, slow is at the middle. */
static node *middle(node *h)
{
    node *slow = h, *fast = h;
    while (fast && fast->next) { slow = slow->next; fast = fast->next->next; }
    return slow;               /* for an even length this is the SECOND of the two middles */
}
/* n-th node from the end in a single pass: advance 'lead' n steps, then move both. */
static node *nth_from_end(node *h, int n)
{
    node *lead = h, *lag = h;
    for (int i = 0; i < n; i++) { if (!lead) return NULL; lead = lead->next; }
    while (lead) { lead = lead->next; lag = lag->next; }
    return lag;
}
/* Floyd's tortoise and hare: they meet iff there is a cycle. */
static int has_cycle(const node *h)
{
    const node *slow = h, *fast = h;
    while (fast && fast->next) {
        slow = slow->next; fast = fast->next->next;
        if (slow == fast) return 1;
    }
    return 0;
}

ليه has_cycle بتشتغل؟ لو فيه دايرة، الاتنين هيدخلوها، والمسافة بينهم بتقلّ واحد كل خطوة (الأرنب بيسبق بخطوة زيادة)، فلازم يتقابلوا. ولو مفيش، الأرنب بيوصل NULL.

التكرار والترتيب (merge sort)

static int is_sorted(const node *h)
{
    for (; h && h->next; h = h->next) if (h->val > h->next->val) return 0;
    return 1;
}
/* remove EVERY later duplicate (unsorted list): O(n^2), no extra memory */
static node *remove_duplicates(node *h)
{
    for (node *p = h; p; p = p->next) {
        node *q = p;
        while (q->next) {
            if (q->next->val == p->val) { node *d = q->next; q->next = d->next; free(d); }
            else q = q->next;
        }
    }
    return h;
}
/* merge sort on a list: split in the middle, sort halves, merge. O(n log n), O(log n) stack,
   and (unlike arrays) O(1) extra memory because merging only relinks nodes. */
static node *list_msort(node *h)
{
    if (!h || !h->next) return h;
    node *slow = h, *fast = h->next;              /* slow ends at the LAST node of the first half */
    while (fast && fast->next) { slow = slow->next; fast = fast->next->next; }
    node *second = slow->next; slow->next = NULL;
    return merge_sorted(list_msort(h), list_msort(second));
}

Merge sort هو الترتيب الأنسب للقوايم: الدمج بيعيد ربط العقد بس، فمفيش مصفوفة مساعدة (O(1) مساحة زيادة، مقابل O(n) في المصفوفات)، والوصول العشوائي اللي quicksort/heapsort محتاجينه مش موجود. التحليل الكامل في درس mergesort.

6) كثيرات الحدود (polynomials) كقايمة

كثيرة حدود زي 7x³y² − 8x²y + 3xy + 11x − 4 مصفوفة معاملات هتكون مضيّعة (مصفوفة 2D فيها أغلبها أصفار — راجع sparse في الدرس السابق). الحل: قايمة حدود (معامل، أس x، أس y) مرتبة تنازليًا. كده الجمع = دمج (merge) وبيحذف الحد لو المعاملين اتلغوا.

typedef struct term { int c, ex, ey; struct term *next; } term;

/* order: bigger x-exponent first, then bigger y-exponent. returns >0 if (ax,ay) comes first */
static int before(int ax, int ay, int bx, int by) { return ax != bx ? ax > bx : ay > by; }

static term *new_term(int c, int ex, int ey)
{
    term *t = malloc(sizeof *t);
    if (!t) { perror("malloc"); exit(1); }
    t->c = c; t->ex = ex; t->ey = ey; t->next = NULL;
    return t;
}
/* append at the tail through a "pointer to the last next-field" (no special case for empty) */
static term **append(term **tail, int c, int ex, int ey)
{
    if (c == 0) return tail;
    *tail = new_term(c, ex, ey);
    return &(*tail)->next;
}
static void poly_free(term *p) { while (p) { term *t = p->next; free(p); p = t; } }
/* p + q : two-pointer merge. Equal exponents add their coefficients (and vanish if 0). */
static term *poly_add(const term *p, const term *q)
{
    term *res = NULL, **tail = &res;
    while (p && q) {
        if (p->ex == q->ex && p->ey == q->ey) { tail = append(tail, p->c + q->c, p->ex, p->ey); p = p->next; q = q->next; }
        else if (before(p->ex, p->ey, q->ex, q->ey)) { tail = append(tail, p->c, p->ex, p->ey); p = p->next; }
        else { tail = append(tail, q->c, q->ex, q->ey); q = q->next; }
    }
    for (; p; p = p->next) tail = append(tail, p->c, p->ex, p->ey);
    for (; q; q = q->next) tail = append(tail, q->c, q->ex, q->ey);
    return res;
}
/* k * p  (Thareja PE 26) */
static term *poly_scale(const term *p, int k)
{
    term *res = NULL, **tail = &res;
    for (; p; p = p->next) tail = append(tail, p->c * k, p->ex, p->ey);
    return res;
}

الضرب: ضرب كل حد من q في كل p، وتجميع النتائج. لـ m وn حد: O(m·n) ضربة. الطريقة اللي فوق بتجمّع الجزئيات واحد واحد (أبطأ شوية من heap-merge) بس أبسط وأسهل في الفهم. (قارن بـ درس 10 اللي عمل نفس الـ ADT بمصفوفات.)

/* p * q : multiply p by each term of q and add the partial products.
   with m and n terms it costs O(m*n) multiplications plus n merges of growing size. */
static term *poly_mul(const term *p, const term *q)
{
    term *res = NULL;
    for (; q; q = q->next) {
        term *part = NULL, **tail = &part;
        for (const term *t = p; t; t = t->next) tail = append(tail, t->c * q->c, t->ex + q->ex, t->ey + q->ey);
        term *sum = poly_add(res, part);
        poly_free(res); poly_free(part);
        res = sum;
    }
    return res;
}
/* d/dx : c*ex x^(ex-1) y^ey (terms with ex==0 vanish) */
static term *poly_dx(const term *p)
{

7) الـ Cursor: قايمة من غير pointers

في FORTRAN القديمة وأنظمة الـ embedded من غير malloc، وحتى في قواعد البيانات (السجل بيشاور على سجل برقمه)، مفيش pointers. الحل: array ضخمة من العقد، و«pointer» = رقم خانة، والصفر = NULL. الخانة 0 هي header الـ free list.

#define SPACE 16
typedef struct { int val; int next; } cnode;       /* next == 0  means "no node" */
static cnode pool[SPACE];

static void pool_init(void)
{
    for (int i = 0; i < SPACE; i++) pool[i].next = (i + 1) % SPACE;   /* 0 -> 1 -> ... -> SPACE-1 -> 0 */
    pool[0].val = 0;                                                  /* slot 0 = free-list header */
}
static int cursor_alloc(void)                  /* returns 0 when the pool is exhausted (OVERFLOW) */
{
    int p = pool[0].next;
    pool[0].next = pool[p].next;               /* unlink first free cell */
    return p;
}
static void cursor_free(int p)
{
    pool[p].next = pool[0].next;               /* push on the free list */
    pool[0].next = p;
}
/* a list is a header cell; the header is allocated like any other cell */
static int list_make(void)
{
    int h = cursor_alloc();
    if (!h) return 0;
    pool[h].next = 0;
    return h;
}
static int list_empty(int h) { return pool[h].next == 0; }

static int list_find(int h, int v)             /* returns the cell index or 0 */
{
    int p = pool[h].next;
    while (p && pool[p].val != v) p = pool[p].next;
    return p;
}
static int list_find_prev(int h, int v)
{
    int p = h;
    while (pool[p].next && pool[pool[p].next].val != v) p = pool[p].next;
    return p;                                   /* pool[p].next is the target, or 0 if absent */
}
static int list_insert_after(int p, int v)      /* returns new cell or 0 on overflow */
{
    int c = cursor_alloc();
    if (!c) return 0;
    pool[c].val = v;
    pool[c].next = pool[p].next;
    pool[p].next = c;
    return c;
}
static int list_delete(int h, int v)            /* 1 if something was removed */
{
    int p = list_find_prev(h, v);
    int t = pool[p].next;
    if (!t) return 0;
    pool[p].next = pool[t].next;
    cursor_free(t);
    return 1;
}
