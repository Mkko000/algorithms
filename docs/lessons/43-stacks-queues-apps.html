1) التعبيرات الحسابية: infix وpostfix وprefix

الكتابة اللي إحنا متعودين عليها infix: العملية بين المعاملين A + B، وعشان نفهمها محتاجين أولويات وأقواس. Postfix (Reverse Polish، RPN): العملية بعد المعاملين A B +. Prefix (Polish): قبلهم + A B. السر: في الشكلين دول مفيش أقواس ومفيش أولويات، الترتيب وحده بيحدد المعنى. وده ليه تطبيق كبير: المترجمات (compilers) بتحوّل infix لـ postfix، والحاسبات اللي بالـ stack (HP RPN) بتقيّم postfix مباشرة.

A - B - CA B - C -- - A B C
A ^ B ^ CA B C ^ ^^ A ^ B C

2) infix → postfix بالـ stack (shunting-yard)

الفكرة: المعاملات (A، 10) بتروح للـ output على طول. العمليات بتستنى على stack لحد ما يجي عملية أضعف (أو مساوية لو شمالية) تقولها «معاملاتك اكتملت». القوس المفتوح بيتحط، وعند القوس المقفول بنفرّغ لحد المفتوح. وفي الآخر نفرّغ الباقي.

typedef struct { char t[MAXT][TLEN]; int n; } toks;

/* split "(A-B)+C*10" into  ( A - B ) + C * 10   returns -1 on a bad character or overflow */
static int tokenize(const char *s, toks *o)
{
    o->n = 0;
    for (; *s; ) {
        if (isspace((unsigned char)*s)) { s++; continue; }
        if (o->n >= MAXT) return -1;
        if (isalnum((unsigned char)*s)) {
            int k = 0;
            while (isalnum((unsigned char)*s)) { if (k < TLEN - 1) o->t[o->n][k++] = *s; s++; }
            o->t[o->n][k] = 0;
        } else if (strchr("+-*/%^()", *s)) { o->t[o->n][0] = *s++; o->t[o->n][1] = 0; }
        else return -1;
        o->n++;
    }
    return o->n;
}
static int is_op(const char *t)    { return t[1] == 0 && strchr("+-*/%^", t[0]) != NULL; }
static int prec(char c)            { return c == '^' ? 3 : (c == '*' || c == '/' || c == '%') ? 2 : 1; }
static int right_assoc(char c)     { return c == '^'; }
static void join(const toks *t, char *out, size_t cap)
{
    out[0] = 0;
    for (int i = 0; i < t->n; i++) { if (i) strncat(out, " ", cap - strlen(out) - 1); strncat(out, t->t[i], cap - strlen(out) - 1); }
}
/* shunting-yard. An operator waits on the stack until an operator of LOWER precedence
   (or equal, for left-associative ones) arrives and tells it "your operands are complete". */
static int to_postfix(const toks *in, toks *out)
{
    char stack[MAXT]; int sp = 0; out->n = 0;
    for (int i = 0; i < in->n; i++) {
        const char *t = in->t[i];
        if (t[0] == '(' ) stack[sp++] = '(';
        else if (t[0] == ')') {
            while (sp && stack[sp - 1] != '(') { out->t[out->n][0] = stack[--sp]; out->t[out->n++][1] = 0; }
            if (!sp) return -1;                              /* unmatched ')' */
            sp--;                                            /* discard '(' */
        } else if (is_op(t)) {
            char c = t[0];
            while (sp && stack[sp - 1] != '(' &&
                   (prec(stack[sp - 1]) > prec(c) || (prec(stack[sp - 1]) == prec(c) && !right_assoc(c)))) {
                out->t[out->n][0] = stack[--sp]; out->t[out->n++][1] = 0;
            }
            stack[sp++] = c;
        } else { strcpy(out->t[out->n++], t); }              /* operand goes straight to the output */
    }
    while (sp) { if (stack[sp - 1] == '(') return -1; out->t[out->n][0] = stack[--sp]; out->t[out->n++][1] = 0; }
    return out->n;
}
الشرط اللي بيتغلط فيهالـ while بتفرّغ لو prec(top) > prec(c) أو المساواة مع c شمالية. لو كتبت >= بس، الـ ^ بتبقى شمالية غلط: A^B^C تطلع A B ^ C ^ بدل A B C ^ ^. الاختبار في البرنامج بيمسك ده.

3) infix → prefix: بطريقتين

الطريقة 1 (عكس + shunting + عكس): اعكس التعبير وبدّل الأقواس، طبّق نفس الخوارزمية بس بـ عكس شرط الـ associativity (لأن بعد العكس العمليات الشمالية بقت بتتجمّع من الناحية التانية)، وبعدين اعكس الناتج. الطريقة 2: حوّل لـ postfix وبعدين من postfix لـ prefix بـ stack من النصوص: لكل عملية خد آخر نصّين x وy، وادفع op x y. نفس الطريقة بتديك infix مقوّس كامل: (x op y). وتحويل prefix → infix بنفس الـ stack بس بنقرأ من اليمين لليسار.

/* method 1: reverse + swap parentheses + shunting-yard with the associativity test FLIPPED
   (because after reversing, left-associative operators group from the other side) */
static int to_prefix_reverse(const toks *in, toks *out)
{
    toks rev; rev.n = in->n;
    for (int i = 0; i < in->n; i++) {
        const char *t = in->t[in->n - 1 - i];
        strcpy(rev.t[i], t);
        if (t[0] == '(') strcpy(rev.t[i], ")"); else if (t[0] == ')') strcpy(rev.t[i], "(");
    }
    char stack[MAXT]; int sp = 0; toks tmp; tmp.n = 0;
    for (int i = 0; i < rev.n; i++) {
        const char *t = rev.t[i];
        if (t[0] == '(' && t[1] == 0) stack[sp++] = '(';
        else if (t[0] == ')' && t[1] == 0) {
            while (sp && stack[sp - 1] != '(') { tmp.t[tmp.n][0] = stack[--sp]; tmp.t[tmp.n++][1] = 0; }
            if (!sp) return -1;
            sp--;
        } else if (is_op(t)) {
            char c = t[0];
            while (sp && stack[sp - 1] != '(' &&
                   (prec(stack[sp - 1]) > prec(c) || (prec(stack[sp - 1]) == prec(c) && right_assoc(c)))) {
                tmp.t[tmp.n][0] = stack[--sp]; tmp.t[tmp.n++][1] = 0;
            }
            stack[sp++] = c;
        } else strcpy(tmp.t[tmp.n++], t);
    }
    while (sp) { if (stack[sp - 1] == '(') return -1; tmp.t[tmp.n][0] = stack[--sp]; tmp.t[tmp.n++][1] = 0; }
    out->n = tmp.n;
    for (int i = 0; i < tmp.n; i++) strcpy(out->t[i], tmp.t[tmp.n - 1 - i]);
    return out->n;
}

/* concatenate NULL-terminated pieces into a fresh heap string (nothing can be truncated) */
static char *catv(const char *first, ...)
{
    va_list ap; size_t len = 0;
    va_start(ap, first);
    for (const char *p = first; p; p = va_arg(ap, const char *)) len += strlen(p);
    va_end(ap);
    char *s = malloc(len + 1), *w = s;
    if (!s) { perror("malloc"); exit(1); }
    va_start(ap, first);
    for (const char *p = first; p; p = va_arg(ap, const char *)) { size_t k = strlen(p); memcpy(w, p, k); w += k; }
    va_end(ap);
    *w = 0;
    return s;
}
/* One stack-of-strings routine for the three string conversions:
     mode 0  postfix -> prefix           "op left right"
     mode 1  postfix -> infix            "(left op right)"
     mode 2  prefix  -> infix            same, but scan right to left
   returns a heap string (caller frees) or NULL if the input is malformed. */
static char *convert(const toks *pf, int mode)
{
    char *st[MAXT]; int sp = 0;
    for (int k = 0; k < pf->n; k++) {
        const char *t = pf->t[mode == 2 ? pf->n - 1 - k : k];
        if (!is_op(t)) { st[sp++] = catv(t, NULL); continue; }
        if (sp < 2) { while (sp) free(st[--sp]); return NULL; }
        char *y = st[--sp], *x = st[--sp];
        char *left = mode == 2 ? y : x, *right = mode == 2 ? x : y;   /* scanning backwards: first pop = LEFT */
        st[sp++] = mode == 0 ? catv(t, " ", left, " ", right, NULL)
                             : catv("(", left, " ", t, " ", right, ")", NULL);
        free(x); free(y);
    }
    if (sp != 1) { while (sp) free(st[--sp]); return NULL; }
    return st[0];
}

دالة convert الواحدة بتغطّي التلات تحويلات (mode 0: postfix→prefix، 1: postfix→infix، 2: prefix→infix). هي بتبني نصوص على الـ heap بـ catv عشان ما نحطش حجم ثابت يقطع التعبير.

4) تقييم postfix وprefix

قراءة من الشمال: معامل → push. عملية → pop اتنين (الأول اللي بيطلع هو المعامل اليمين)، طبّق، push النتيجة. في الآخر لازم يفضل عنصر واحد بالظبط (غير كده التعبير غلط). ممكن نقيّم prefix بنفس الطريقة لو قرينا من اليمين (وساعتها أول pop هو اليسار).

static double apply(char op, double a, double b)
{
    switch (op) {
    case '+': return a + b;  case '-': return a - b;
    case '*': return a * b;  case '/': return a / b;
    case '%': return fmod(a, b);
    default:  return pow(a, b);
    }
}
/* postfix: scan LEFT to RIGHT. operand -> push; operator -> pop b, pop a, push a op b */
static int eval_postfix(const toks *pf, double *res)
{
    double st[MAXT]; int sp = 0;
    for (int i = 0; i < pf->n; i++) {
        const char *t = pf->t[i];
        if (is_op(t)) { if (sp < 2) return -1; double b = st[--sp], a = st[--sp]; st[sp++] = apply(t[0], a, b); }
        else st[sp++] = atof(t);
    }
    if (sp != 1) return -1;
    *res = st[0]; return 0;
}
/* prefix: scan RIGHT to LEFT. the first operand popped is the LEFT operand: a op b */
static int eval_prefix(const toks *pf, double *res)
{
    double st[MAXT]; int sp = 0;
    for (int i = pf->n - 1; i >= 0; i--) {
        const char *t = pf->t[i];
        if (is_op(t)) { if (sp < 2) return -1; double a = st[--sp], b = st[--sp]; st[sp++] = apply(t[0], a, b); }
        else st[sp++] = atof(t);
    }
    if (sp != 1) return -1;
    *res = st[0]; return 0;
}

5) الأقواس المتوازنة (balancing symbols)

أبسط تطبيق للـ stack وبيظهر في كل compiler. كل فاتح بيدفع مقفوله المتوقع، وكل مقفول لازم يطابق القمة. الأخطاء التلاتة: مقفول زيادة (stack فاضي)، مقفول غلط (القمة مش هو)، وفاتح ماتقفلش (stack مش فاضي في الآخر). الدالة بترجع مكان أول غلطة مش بس صح/غلط.

/* ( [ { push their closer; a closer must match the top. returns -1 if balanced,
   otherwise the 0-based index of the first offending character. */
static int check_brackets(const char *s)
{
    char st[256]; int sp = 0;
    for (int i = 0; s[i]; i++) {
        char c = s[i];
        if (c == '(') st[sp++] = ')';
        else if (c == '[') st[sp++] = ']';
        else if (c == '{') st[sp++] = '}';
        else if (c == ')' || c == ']' || c == '}') {
            if (sp == 0 || st[--sp] != c) return i;      /* underflow, or mismatched pair */
        }
    }
    return sp == 0 ? -1 : (int)strlen(s);               /* leftover openers: error "at the end" */
}

6) أكتر من stack في array واحدة

لو عندك stackين وكل واحد محجوز له نص المصفوفة، واحد منهم ممكن يفيض والتاني فاضي نص مساحته. الحل: واحد يبدأ من الأول (topA) والتاني من الآخر (topB) ويكبروا لبعض. الـ overflow بس لما topA + 1 == topB: يعني المصفوفة كلها اتملت فعلًا. (للـ queues الحل بيبقى بالـ circular أو بتقسيم ديناميكي لأن الـ queue بتتحرك للأمام.)

/* two stacks in ONE array: stack A grows from the left end, stack B from the right end.
   Overflow happens only when the WHOLE array is full — much better than two fixed halves. */
typedef struct { int a[CAP]; int topA, topB; } two_stacks;     /* topA: next free on the left, topB on the right */
static void ts_init(two_stacks *s) { s->topA = -1; s->topB = CAP; }
static int ts_push(two_stacks *s, int which, int v)
{
    if (s->topA + 1 == s->topB) return 0;                         /* the two tops met: full */
    if (which == 0) s->a[++s->topA] = v; else s->a[--s->topB] = v;
    return 1;
}
static int ts_pop(two_stacks *s, int which, int *v)
{
    if (which == 0) { if (s->topA < 0) return 0; *v = s->a[s->topA--]; }
    else            { if (s->topB >= CAP) return 0; *v = s->a[s->topB++]; }
    return 1;
}

7) stack بـ getMin في O(1)

مع كل push بنخزّن كمان «أصغر قيمة لحد دلوقتي» في stack تاني. الـ pop بيشيل من الاتنين، فالـ min دايمًا فوق التاني. نفس الفكرة بتنفع لأي دالة تجميعية (max، gcd، sum).

/* every push also pushes min(v, current min) on a second stack; getMin = peek of that stack */
typedef struct { int v[CAP], m[CAP], n; } min_stack;
static void ms_push(min_stack *s, int x)
{
    assert(s->n < CAP);
    s->m[s->n] = (s->n == 0 || x < s->m[s->n - 1]) ? x : s->m[s->n - 1];
    s->v[s->n++] = x;
}
static int ms_pop(min_stack *s) { assert(s->n > 0); return s->v[--s->n]; }
static int ms_min(const min_stack *s) { assert(s->n > 0); return s->m[s->n - 1]; }

8) queue من stackين، وstack من queueين

queue من stackين: enqueue بيدفع في IN. dequeue بيعمل pop من OUT، ولو OUT فاضي بينقل كل IN لـ OUT (فبيتعكس الترتيب). كل عنصر بينتقل مرة واحدة بس في عمره، فـ n عملية تكلّف O(n): amortised O(1) للعملية (رغم إن dequeue الواحدة ممكن تكلّف O(n)). البرنامج بيعدّ النقلات ويتأكد إنها ≤ عدد العمليات.

stack من queueين: push بيدخل العنصر الجديد في queue فاضية، ويدخل وراه كل القديم، وبعدين نبدّل الدورين. فالجديد بقى في المقدمة: push بـ O(n) وpop بـ O(1). (أو العكس.)

/* queue from two stacks: enqueue pushes on IN; dequeue pops from OUT, and only when OUT is
   empty moves EVERYTHING from IN to OUT (reversing it).  Each element moves at most once
   from IN to OUT, so n operations cost O(n) in total = amortised O(1) per operation. */
typedef struct { int in[CAP], out[CAP], ni, no; long moves; } queue2s;
static void q2_enqueue(queue2s *q, int x) { assert(q->ni < CAP); q->in[q->ni++] = x; }
static int  q2_dequeue(queue2s *q)
{
    if (q->no == 0) while (q->ni) { q->out[q->no++] = q->in[--q->ni]; q->moves++; }
    assert(q->no > 0);
    return q->out[--q->no];
}

/* stack from two queues (circular int queues): push makes the new element the FRONT by
   rotating the old ones behind it: O(n) push, O(1) pop. */
typedef struct { int a[CAP]; int head, n; } iqueue;
static void iq_put(iqueue *q, int x) { assert(q->n < CAP); q->a[(q->head + q->n++) % CAP] = x; }
static int  iq_get(iqueue *q) { assert(q->n > 0); int x = q->a[q->head]; q->head = (q->head + 1) % CAP; q->n--; return x; }
typedef struct { iqueue q1, q2; } stack2q;
static void s2_push(stack2q *s, int x)
{
    iq_put(&s->q2, x);
    while (s->q1.n) iq_put(&s->q2, iq_get(&s->q1));
    iqueue t = s->q1; s->q1 = s->q2; s->q2 = t;          /* swap the roles */
}
static int s2_pop(stack2q *s) { return iq_get(&s->q1); }

9) المونوتونك stack: next greater وstock span

سؤال متكرر: لكل عنصر، أول عنصر أكبر منه على يمينه. الحل الساذج O(n²). بـ stack فيه أرقام الخانات اللي لسه ما عرفناش إجابتها (قيمها متناقصة): لما يجي عنصر أكبر منهم، هو الإجابة بتاعتهم كلهم فنشيلهم. كل خانة تتدفع مرة وتتشال مرة: O(n). نفس الفكرة بتحل stock span (كام يوم متتالي السعر ≤ النهاردة).

/* next greater element: for each i, the first a[j] > a[i] with j > i (or -1).
   The stack holds indices whose answer is still unknown, with DECREASING values; each index
   is pushed once and popped once, so the whole scan is O(n). */
static void next_greater(const int a[], int n, int out[])
{
    int st[CAP], sp = 0;
    for (int i = 0; i < n; i++) {
        while (sp && a[st[sp - 1]] < a[i]) out[st[--sp]] = a[i];
        st[sp++] = i;
    }
    while (sp) out[st[--sp]] = -1;
}
/* stock span: span[i] = number of consecutive days up to i with price <= price[i] */
static void stock_span(const int p[], int n, int span[])
{
    int st[CAP], sp = 0;
    for (int i = 0; i < n; i++) {
        while (sp && p[st[sp - 1]] <= p[i]) sp--;
        span[i] = sp ? i - st[sp - 1] : i + 1;
        st[sp++] = i;
    }
}

ولعمليات الـ stack بس (copy، compare، reverse) بدون أي هيكل تاني:

typedef struct { int a[CAP]; int n; } istack;
static void is_push(istack *s, int x) { assert(s->n < CAP); s->a[s->n++] = x; }
static int  is_pop(istack *s)          { assert(s->n > 0);   return s->a[--s->n]; }
/* copy preserving order: pop into a temporary (reverses), pop back into both (restores) */
static void is_copy(istack *src, istack *dst)
{
    istack tmp = {{0}, 0};
    while (src->n) is_push(&tmp, is_pop(src));
    while (tmp.n) { int x = is_pop(&tmp); is_push(src, x); is_push(dst, x); }
}
/* compare two stacks: they are equal iff the pops agree one by one (both are restored) */
static int is_equal(istack *a, istack *b)
{
    if (a->n != b->n) return 0;
    istack ta = {{0}, 0}, tb = {{0}, 0}; int eq = 1;
    while (a->n) { int x = is_pop(a), y = is_pop(b); is_push(&ta, x); is_push(&tb, y); if (x != y) eq = 0; }
    while (ta.n) is_push(a, is_pop(&ta));
    while (tb.n) is_push(b, is_pop(&tb));
    return eq;
}
/* reverse a stack using only push/pop and two temporary stacks (three moves = identity on order twice, reverse once) */
static void is_reverse(istack *s)
{
    istack t1 = {{0}, 0}, t2 = {{0}, 0};
    while (s->n) is_push(&t1, is_pop(s));
    while (t1.n) is_push(&t2, is_pop(&t1));
    while (t2.n) is_push(s, is_pop(&t2));
}

10) الريكيرشن = stack: نحاكي الـ call stack بإيدينا

كل نداء recursive بيعمل frame (الـ arguments والمتغيرات المحلية و«أكمّل من فين») على الـ call stack، وكل return بيشيله. عشان كده الريكيرشن العميق جدًا بيعمل stack overflow. أي دالة recursive ممكن نحوّلها لحلقة + stack صريح. الأمثلة هنا: factorial (الأبسط: stack من «الضربات المستنية»)، Fibonacci (frame بيه حالة 0/1/2 ومسجّل ret)، Hanoi (stack من «مهام»، بنحط المهام بعكس ترتيب التنفيذ)، وAckermann (stack من الـ m المستنية).

/* recursive: 1 frame per level, n+1 frames deep at the bottom */
static long fact_rec(int n) { return n <= 1 ? 1 : n * fact_rec(n - 1); }

/* explicit stack: phase 1 "descend" pushes n, n-1, ..., 2 (the pending multiplications);
   phase 2 "return" pops them multiplying on the way back — exactly the order the real stack unwinds. */
static long fact_stack(int n, int *max_depth)
{
    int st[64], sp = 0;
    while (n > 1) st[sp++] = n--;                 /* "call fact(n-1), remember to multiply by n" */
    *max_depth = sp + 1;
    long result = 1;                               /* fact(1) */
    while (sp) result *= st[--sp];                 /* "return": result = n * result */
    return result;
}
/* tail recursion: the recursive call is the LAST thing done, so no frame has to be remembered;
   a compiler (or you) can turn it into a loop that reuses one frame. */
static long fact_tail(int n, long acc) { return n <= 1 ? acc : fact_tail(n - 1, acc * n); }
static long fact_loop(int n) { long acc = 1; while (n > 1) acc *= n--; return acc; }
static int gcd_rec(int a, int b)  { return b == 0 ? a : gcd_rec(b, a % b); }
static int gcd_loop(int a, int b) { while (b) { int t = a % b; a = b; b = t; } return a; }

Tail recursion: لو النداء الـ recursive هو آخر حاجة بتتعمل (مفيش ضرب مستني)، فمش محتاجين نفتكر frame القديم: compiler بيحوّلها لحلقة. fact_tail بتحمل المجموع التراكمي في acc، وgcd بتتحوّل مباشرة. لاحظ إن fact_rec مش tail لأن n * مستنية بعد الرجوع.

static long fib_rec(int n) { return n < 2 ? n : fib_rec(n - 1) + fib_rec(n - 2); }
static long calls;                                   /* counts frames pushed */
static long fib_count(int n) { calls++; return n < 2 ? n : fib_count(n - 1) + fib_count(n - 2); }

/* Fibonacci with an explicit stack. A frame = (n, state). state 0: need to compute fib(n-1);
   state 1: have fib(n-1) in 'a', need fib(n-2); state 2: both done, return a+b.
   'ret' is the value being returned to the frame below (the "return register"). */
typedef struct { int n, state; long a; } frame;
static long fib_stack(int n0)
{
    frame st[128]; int sp = 0; long ret = 0;
    st[sp++] = (frame){n0, 0, 0};
    while (sp) {
        frame *f = &st[sp - 1];
        if (f->n < 2) { ret = f->n; sp--; continue; }               /* base case: return n */
        if (f->state == 0)      { f->state = 1; st[sp++] = (frame){f->n - 1, 0, 0}; }   /* call fib(n-1) */
        else if (f->state == 1) { f->a = ret; f->state = 2; st[sp++] = (frame){f->n - 2, 0, 0}; }   /* call fib(n-2) */
        else                    { ret = f->a + ret; sp--; }          /* return fib(n-1)+fib(n-2) */
    }
    return ret;
}

أبراج هانوي

انقل n قرص من A لـ C بقرص واحد كل مرة وبدون كبير فوق صغير: انقل n−1 لـ B، انقل الأكبر لـ C، انقل n−1 من B لـ C. عدد الحركات T(n) = 2T(n−1) + 1 = 2ⁿ − 1، ومفيش حل أسرع (إثبات بالاستقراء: الأكبر لازم يتحرك مرة على الأقل، وقبلها n−1 لازم يكونوا على القضيب الوسيط).

/* Towers of Hanoi: move n disks from 'from' to 'to' using 'via'. 2^n - 1 moves. */
static long hmoves; static int hlog[2048][2];
static void hanoi_rec(int n, int from, int to, int via)
{
    if (n == 0) return;
    hanoi_rec(n - 1, from, via, to);
    hlog[hmoves][0] = from; hlog[hmoves][1] = to; hmoves++;
    hanoi_rec(n - 1, via, to, from);
}
/* Same with an explicit stack of "tasks": either "solve (n,from,to,via)" or "do this one move".
   Push in REVERSE order of execution: move n-1 (via->to) last, the disk, then move n-1 (from->via) first. */
typedef struct { int kind, n, from, to, via; } task;       /* kind 0 = solve, 1 = move one disk */
static long hanoi_stack(int n, int from, int to, int via, int out[][2])
{
    task st[256]; int sp = 0; long m = 0;
    st[sp++] = (task){0, n, from, to, via};
    while (sp) {
        task t = st[--sp];
        if (t.kind == 1) { out[m][0] = t.from; out[m][1] = t.to; m++; continue; }
        if (t.n == 0) continue;
        st[sp++] = (task){0, t.n - 1, t.via, t.to, t.from};
        st[sp++] = (task){1, 0, t.from, t.to, 0};
        st[sp++] = (task){0, t.n - 1, t.from, t.via, t.to};
    }
    return m;
}

دالة Ackermann

بتنمو أسرع من أي دالة primitive recursive: A(2,3) = 9 وA(3,3) = 61 وA(4,2) رقم بـ 19729 خانة. مثال كلاسيكي على recursion متداخل (النداء جوه argument نداء).

/* Ackermann: grows faster than any primitive recursive function. A(2,3)=9, A(3,3)=61, A(4,*) is astronomical.
   m = 0:  A = n+1;   n = 0:  A(m-1, 1);   else  A(m-1, A(m, n-1)).  */
static long ack_rec(long m, long n)
{
    if (m == 0) return n + 1;
    if (n == 0) return ack_rec(m - 1, 1);
    return ack_rec(m - 1, ack_rec(m, n - 1));
}
/* the stack of pending "m" values: A(m, n) with m>0 and n>0 becomes  push m-1, then compute A(m, n-1) */
static long ack_stack(long m, long n, long *max_stack)
{
    long *st = malloc(sizeof(long) * (1 << 20)); long sp = 0; *max_stack = 0;
    st[sp++] = m;
    while (sp) {
        m = st[--sp];
        if (m == 0) { n = n + 1; }
        else if (n == 0) { st[sp++] = m - 1; n = 1; }
        else { st[sp++] = m - 1; st[sp++] = m; n = n - 1; }
        if (sp > *max_stack) *max_stack = sp;
    }
    free(st);
    return n;
}

11) الـ queue الدايرية: فاضية ولا ممتلئة؟

في queue بـ array عادية، بعد ما REAR يوصل آخر المصفوفة بنقول OVERFLOW حتى لو الخانات اللي قبل FRONT فاضية (بعد dequeues). الحل: الدوران (i+1) % N. المشكلة الجديدة: لما FRONT == REAR+1 الـ queue ممكن تكون فاضية أو ممتلئة. تلات حلول (كلها في الكود ومتقارنة بنماذج مرجعية على 20000 خطوة عشوائية):

(a) حقل countn == 0n == NN
(b) خانة فاضية دايمًاfront == rear(rear+1)%N == frontN−1

/* (a) count field: front = index of the oldest element, n = how many.
       rear = (front + n) % N  — it is never stored, it is computed. Full: n == N. Empty: n == 0. */
typedef struct { int a[N]; int front, n; } cqueue;
static int cq_put(cqueue *q, int x)
{
    if (q->n == N) return 0;                           /* overflow */
    q->a[(q->front + q->n++) % N] = x; return 1;
}
static int cq_get(cqueue *q, int *x)
{
    if (q->n == 0) return 0;                           /* underflow */
    *x = q->a[q->front]; q->front = (q->front + 1) % N; q->n--; return 1;
}

/* (b) front/rear only, so N-1 usable cells: empty is front == rear, full is (rear+1)%N == front */
typedef struct { int a[N]; int front, rear; } cqueue_b;
static int cb_put(cqueue_b *q, int x)
{
    if ((q->rear + 1) % N == q->front) return 0;
    q->a[q->rear] = x; q->rear = (q->rear + 1) % N; return 1;
}
static int cb_get(cqueue_b *q, int *x)
{
    if (q->front == q->rear) return 0;
    *x = q->a[q->front]; q->front = (q->front + 1) % N; return 1;
}

/* (c) the book's convention: front = rear = -1 means empty.  N usable cells.
       full  <=>  (front == 0 && rear == N-1)  ||  (front == rear + 1)   — the two faces of the same state */
typedef struct { int a[N]; int front, rear; } cqueue_c;
static int cc_full(const cqueue_c *q) { return (q->front == 0 && q->rear == N - 1) || q->front == q->rear + 1; }
static int cc_put(cqueue_c *q, int x)
{
    if (cc_full(q)) return 0;
    if (q->front == -1) q->front = q->rear = 0;
    else q->rear = (q->rear + 1) % N;
    q->a[q->rear] = x; return 1;
}
static int cc_get(cqueue_c *q, int *x)
{
    if (q->front == -1) return 0;
    *x = q->a[q->front];
    if (q->front == q->rear) q->front = q->rear = -1;        /* took the last one: back to empty */
    else q->front = (q->front + 1) % N;
    return 1;
}

ملحوظة على أسئلة Thareja: مفتاح الإجابات في آخر الكتاب بيقول إن الـ circular queue بتبقى ممتلئة لما FRONT = 0 and REAR = MAX−1 (اختيار (b))، وده فعلًا حالة من الحالتين في الجدول (c). والمفتاح بيقول كمان إن «front = MAX−1 وrear = 0 تعني ممتلئة» صح؛ ده صحيح حرفيًا بس لما MAX = 2، والقاعدة العامة هي front == (rear+1) % MAX. التفصيل في صفحة التمارين.

12) الـ Deque (double-ended queue)

إدخال وحذف من الطرفين. بـ circular array: push_left بيرجّع left خطوة لورا (بالـ mod) ويخزّن، push_right بيخزّن عند (left+n)%N. نوعين مقيّدين: input-restricted (إدخال من طرف واحد، حذف من الاتنين) وoutput-restricted (حذف من طرف واحد، إدخال من الاتنين). همّا نفس الهيكل مع منع عمليتين. الـ stack = deque بتستخدم طرف واحد، والـ queue = deque (إدخال يمين، حذف شمال). تطبيقات: undo/redo، work-stealing، sliding window maximum.

/* deque on a circular array.  left = index of the leftmost element, n = count.
   push_left moves left one step BACK (mod N) and stores; push_right stores at (left+n)%N. */
typedef struct { int a[N]; int left, n; } deque;
static int dq_push_left(deque *d, int x)
{
    if (d->n == N) return 0;
    d->left = (d->left + N - 1) % N; d->a[d->left] = x; d->n++; return 1;
}
static int dq_push_right(deque *d, int x)
{
    if (d->n == N) return 0;
    d->a[(d->left + d->n++) % N] = x; return 1;
}
static int dq_pop_left(deque *d, int *x)
{
    if (!d->n) return 0;
    *x = d->a[d->left]; d->left = (d->left + 1) % N; d->n--; return 1;
}
static int dq_pop_right(deque *d, int *x)
{
    if (!d->n) return 0;
    *x = d->a[(d->left + --d->n) % N]; return 1;
}
/* Input-restricted deque: insert only at the right; delete at both ends.
   Output-restricted deque: delete only at the left; insert at both ends.
   They are the same structure with two of the four operations simply not offered. */
#define ir_insert(d, x)      dq_push_right((d), (x))
#define ir_delete_left(d, x) dq_pop_left((d), (x))
#define ir_delete_right(d,x) dq_pop_right((d), (x))
#define or_insert_left(d, x) dq_push_left((d), (x))
#define or_insert_right(d,x) dq_push_right((d), (x))
#define or_delete(d, x)      dq_pop_left((d), (x))

13) الـ Priority Queue بطريقتين

كل عنصر ليه أولوية، والأعلى أولوية يتخدم الأول، ولو اتساووا فالـ FCFS. (1) قايمة مرتبة: الإدخال O(n) (نمشي لحد أول عقدة أولويتها أقل بشدة، فالجديد يدخل بعد المتساوين)، والحذف O(1) (الأول). (2) مصفوفة طوابير: صف لكل مستوى أولوية، الإدخال O(1)، والحذف بيدوّر على أول صف مش فاضي O(P) (P عدد المستويات). الاتنين «غير مثاليين»؛ الـ heap بيدّي O(log n) للاتنين.

/* priority queue = sorted linked list; SMALLER number = HIGHER priority; ties are FCFS.
   insert: walk while the next node's priority is <= the new one's (so the new node goes AFTER equals).
   delete: remove the head.   insert O(n), delete O(1). */
typedef struct pnode { int data, pri; struct pnode *next; } pnode;
static pnode *pq_insert(pnode *head, int data, int pri)
{
    pnode *n = malloc(sizeof *n);
    if (!n) { perror("malloc"); exit(1); }
    n->data = data; n->pri = pri;
    if (!head || pri < head->pri) { n->next = head; return n; }
    pnode *p = head;
    while (p->next && p->next->pri <= pri) p = p->next;
    n->next = p->next; p->next = n;
    return head;
}
static pnode *pq_delete(pnode *head, int *data)
{
    assert(head);
    *data = head->data; pnode *rest = head->next; free(head); return rest;
}
/* priority queue as an array of circular queues, one per priority level 0..P-1 (0 = most urgent).
   insert: put at the rear of row p, O(1).   delete: scan rows from 0 for the first non-empty row, O(P). */
#define P 5
typedef struct { cqueue row[P]; } pq_array;
static int pa_insert(pq_array *q, int data, int pri) { return cq_put(&q->row[pri], data); }
static int pa_delete(pq_array *q, int *data, int *pri)
{
    for (int k = 0; k < P; k++) if (q->row[k].n) { *pri = k; return cq_get(&q->row[k], data); }
    return 0;
}
