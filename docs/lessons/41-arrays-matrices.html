الـ array هي أبسط هيكل بيانات في الدنيا، وكل هيكل تاني (stack, queue, heap, hash table) غالبًا بيتبني فوقها. بس الكتب الأربعة بتحط عليها أسئلة كتير مش واضحة من أول نظرة: إزاي المترجم بيلاقي A[i][j]؟ إيه تكلفة الإدخال في النص؟ ليه نخزّن ٩٩٪ أصفار في مصفوفة؟ الدرس ده بيجاوب على كل ده، وكل سطر كود هنا اتجرّب وشغّال في src/17-arrays-matrices/.

بدون تكرارالريبو أصلًا فيه الأنواع والمصفوفات في درس 05 والـ binary search في درس التحليل والـ sorting كله في الجزء الرابع. هنا بنغطّي بس اللي ناقص: عمليات الـ array الأساسية بتحليل تكلفتها، حساب العناوين، الـ sparse والـ special matrices.
هدف الدرس: تحسب عنوان أي عنصر في array متعددة الأبعاد (row/column-major وحدود دنيا مش صفر)، تكتب الإدخال والحذف والدمج والقلب والـ rotation بتكلفتها الصح، وتختار تمثيل مضغوط لمصفوفة sparse أو مثلثية أو tri-diagonal وتحسب وفّرت قد إيه.

1) الـ array في الذاكرة

الـ array = عناصر من نفس النوع محطوطة ورا بعض في الذاكرة. ده بالظبط اللي بيدّيك الـ O(1) في الوصول: لو عارف عنوان أول عنصر (Base) وحجم العنصر (w بالبايت)، عنوان العنصر رقم i حساب بسيط:

Addr(A[i]) = Base + w · (i − LB)        // LB = أول index (في C بتساوي 0)

سؤال Thareja رقم 8: int arr[50] و Base=1000 و w=2، عنوان arr[35] = 1000 + 2·35 = 1070. ولو اللغة بتبدأ من 1 (زي Pascal) أو من رقم تاني (A[-5..5]) بنطرح الـ LB. اللغات دي بتحتاج الطرح ده؛ C مش بتحتاجه، وده سبب إن الـ index بيبدأ من صفر في C.

ملاحظة على سؤال 9 في الكتابسؤال Thareja رقم 9 بيطلب Marks[8][5] في مصفوفة Marks[10][5] — بس الـ index الصحيح للأعمدة من 0 لـ 4، فـ [5] برّه الحدود! الحساب بالمعادلة بيدّي 2090 (والعنصر ده عمليًا هو Marks[9][0] لأن C بتخزّن row-major). مهم تفرّق بين «اللي المعادلة بتحسبه» و«اللي legal فعلًا».

2) الإدخال والحذف: تكلفة الإزاحة

الـ array مكانها ثابت، فلو عايز تدخّل عنصر في النص لازم تحرّك اللي بعده. الشكل ده بيوضّح ليه الكتب كلها بتقولك ابدأ من الآخر:

/* insert value v at position pos (0-based); shifts a[pos..n-1] one step right.
 * cost: n - pos moves -> O(n) worst case (pos = 0), O(1) best (pos = n). */
int array_insert(int a[], int n, int cap, int pos, int v)
{
    if (n >= cap || pos < 0 || pos > n) return -1;   /* overflow / bad index */
    for (int i = n - 1; i >= pos; i--)               /* go from the END */
        a[i + 1] = a[i];
    a[pos] = v;
    return n + 1;
}

الحذف عكسه: بنزيح من الأول لقدّام (لو بدأت من الآخر هتنسخ نفس القيمة على كل الخانات). في الحالتين التكلفة n − pos حركة: O(n) في الأسوأ، O(1) لو في الآخر. وده بيجاوب سؤال Thareja رقم 14: في 7 أسماء، إدخال Andrew (ترتيبه 1) بيحرّك 6 أسماء، وحذف Esha (index 3) بيحرّك 3.

/* delete the element at position pos; shifts a[pos+1..n-1] one step left. */
int array_delete(int a[], int n, int pos)
{
    if (pos < 0 || pos >= n) return -1;              /* underflow / bad index */
    for (int i = pos; i < n - 1; i++)                /* go from the FRONT */
        a[i] = a[i + 1];
    return n - 1;
}

الدمج (merging)

«الدمج» في Thareja يعني تلزيق مصفوفتين ورا بعض. لكن الدمج المهم هو دمج مصفوفتين مرتبتين في مرتبة (نفس قلب mergesort): بنمشي بمؤشرين ونختار الأصغر. O(n₁+n₂)، ومش بيحتاج ترتيب تاني.

/* merge two SORTED arrays into c; returns n1 + n2. Stable, O(n1 + n2). */
int array_merge_sorted(const int a[], int n1, const int b[], int n2, int c[])
{
    int i = 0, j = 0, k = 0;
    while (i < n1 && j < n2) c[k++] = (b[j] < a[i]) ? b[j++] : a[i++];
    while (i < n1) c[k++] = a[i++];
    while (j < n2) c[k++] = b[j++];
    return k;
}

القلب والـ rotation بتلات reversals

عايز تلف المصفوفة k خانات شمال من غير مصفوفة مساعدة؟ قسّمها [A | B] وبدّل أماكنهم. الحيلة: reverse(A) و reverse(B) بعدين reverse(الكل) بيطلّع [B | A]. ليه؟ لأن reverse(reverse(A) reverse(B)) = B A. ثلاث مرّات قلب = O(n) زمن و O(1) مساحة.

static void rev(int a[], int i, int j) { while (i < j) { int t = a[i]; a[i++] = a[j]; a[j--] = t; } }

void array_reverse(int a[], int n) { rev(a, 0, n - 1); }

/* rotate LEFT by k using three reversals: O(n) time, O(1) extra space.
 *   [A | B]  ->  reverse(A), reverse(B), reverse(all)  ->  [B | A]       */
void array_rotate_left(int a[], int n, int k)
{
    if (n == 0) return;
    k %= n;
    rev(a, 0, k - 1);
    rev(a, k, n - 1);
    rev(a, 0, n - 1);
}

3) المصفوفات المتعددة الأبعاد وحساب العنوان

الذاكرة خط واحد، فلازم نحوّل (i, j) لرقم واحد. فيه طريقتين متفق عليهم:

C بتستخدم row-major. وده مش مجرد تفصيلة: لو بتلف على مصفوفة كبيرة بـ for i { for j { a[i][j] } } بتمشي ورا بعض في الذاكرة (الـ cache مبسوط)، ولو عكست الحلقتين بتقفز كل مرة C عنصر — والفرق ممكن يبقى 5–10 أضعاف في السرعة على مصفوفات كبيرة.

/* 1-D:  Addr(A[i]) = Base + w * (i - LB)                                   */
long addr1(long base, long w, long lb, long i) { return base + w * (i - lb); }

/* 2-D with arbitrary lower bounds, R rows x C columns:
 *   row-major    Addr(A[i][j]) = Base + w * ((i - LBr) * C + (j - LBc))
 *   column-major Addr(A[i][j]) = Base + w * ((j - LBc) * R + (i - LBr))    */
long addr2_row(long base, long w, long lbr, long lbc, long R, long C, long i, long j)
{ (void)R; return base + w * ((i - lbr) * C + (j - lbc)); }

long addr2_col(long base, long w, long lbr, long lbc, long R, long C, long i, long j)
{ (void)C; return base + w * ((j - lbc) * R + (i - lbr)); }

/* n-D row-major: the LAST subscript varies fastest. Horner-style: */
long addrN_row(long base, long w, int n, const long lb[], const long len[], const long idx[])
{
    long off = 0;
    for (int d = 0; d < n; d++) off = off * len[d] + (idx[d] - lb[d]);
    return base + w * off;
}

/* n-D column-major: the FIRST subscript varies fastest. */
long addrN_col(long base, long w, int n, const long lb[], const long len[], const long idx[])
{
    long off = 0;
    for (int d = n - 1; d >= 0; d--) off = off * len[d] + (idx[d] - lb[d]);
    return base + w * off;
}

الدالة addrN_row بتعمل نفس الحاجة لأي عدد أبعاد بأسلوب Horner: off = off·len[d] + idx[d]. وده بالظبط اللي المترجم بيعمله لـ T[i][j][k] ((i·4 + j)·5 + k في int T[3][4][5]). الكود بيتأكد إن المعادلة تطابق &M[i][j] الحقيقي لكل الخلايا.

4) المصفوفة الـ sparse: ليه نخزّن الأصفار؟

جدول 1000×1000 بيحجز مليون خانة. لو فيه 3000 قيمة بس غير صفرية (زي مصفوفة الجيران لشبكة طرق، أو مصفوفة معاملات معادلات)، ٩٩٫٧٪ من الذاكرة بيضيع على أصفار. الحل: نخزّن بس الحدود غير الصفرية كـ triplets (row, col, val) مرتبة بالصف ثم العمود.

void sp_from_dense(Sparse *s, int rows, int cols, const int *d)
{
    s->rows = rows; s->cols = cols; s->n = 0;
    for (int i = 0; i < rows; i++)
        for (int j = 0; j < cols; j++)
            if (d[i * cols + j] != 0) {
                assert(s->n < MAXT);
                s->t[s->n++] = (Term){ i, j, d[i * cols + j] };
            }
}

Transpose: الطريقة البطيئة والسريعة

transpose معناها نبدّل الصف بالعمود، بس لازم نحافظ على ترتيب (صف، عمود) في النتيجة. الطريقة الساذجة: لكل عمود c من A، لف على كل الحدود ودوّر على اللي عمودها c. تكلفتها O(cols · n)، ولو المصفوفة مش sparse جدًا (n ≈ rows·cols) تبقى O(rows·cols²) — أبطأ من التخزين العادي!

/* Simple transpose: for each column c of A (in order), scan ALL terms.
 * Cost O(cols * n). When n ~ rows*cols this is O(rows*cols^2): worse than dense! */
void sp_transpose_slow(const Sparse *a, Sparse *b)
{
    b->rows = a->cols; b->cols = a->rows; b->n = a->n;
    int k = 0;
    for (int c = 0; c < a->cols; c++)
        for (int p = 0; p < a->n; p++)
            if (a->t[p].col == c) b->t[k++] = (Term){ c, a->t[p].row, a->t[p].val };
}

الطريقة السريعة هي نفس key-indexed counting اللي شفناه في الفصل 6: (1) نعدّ عدد الحدود في كل عمود، (2) نحوّل العدّ لمواقع بداية (prefix sums)، (3) نحط كل حد في مكانه من غير ما نبحث. تكلفتها O(cols + n).

/* Fast transpose: first COUNT the terms in each column, turn the counts into
 * starting positions (prefix sums), then place every term once.
 * Cost O(cols + n)  — the same trick as key-indexed counting. */
void sp_transpose_fast(const Sparse *a, Sparse *b)
{
    int cnt[MAXT] = {0}, pos[MAXT] = {0};
    b->rows = a->cols; b->cols = a->rows; b->n = a->n;
    for (int p = 0; p < a->n; p++) cnt[a->t[p].col]++;
    for (int c = 1; c < a->cols; c++) pos[c] = pos[c - 1] + cnt[c - 1];
    for (int p = 0; p < a->n; p++) {
        int q = pos[a->t[p].col]++;
        b->t[q] = (Term){ a->t[p].col, a->t[p].row, a->t[p].val };
    }
}

الجمع والضرب

الجمع = دمج قايمتين مرتبتين (نفس merge فوق) مع مراعاة إن حدّين في نفس الخلية بيتجمعوا وممكن يطلعوا صفر (ساعتها نشيل الحد). الضرب: نحوّل B لـ Bᵀ بالـ fast transpose، فصف في A وصف في Bᵀ هما قايمتين مرتبتين بالعمود، والـ dot product بيتعمل بـ merge.

/* C = A + B : merge two sorted term lists (like merging two sorted arrays). */
int sp_add(const Sparse *a, const Sparse *b, Sparse *c)
{
    if (a->rows != b->rows || a->cols != b->cols) return 0;
    c->rows = a->rows; c->cols = a->cols; c->n = 0;
    int i = 0, j = 0;
    while (i < a->n || j < b->n) {
        long ka = i < a->n ? (long)a->t[i].row * a->cols + a->t[i].col : -1;
        long kb = j < b->n ? (long)b->t[j].row * b->cols + b->t[j].col : -1;
        if (j >= b->n || (i < a->n && ka < kb)) c->t[c->n++] = a->t[i++];
        else if (i >= a->n || kb < ka)          c->t[c->n++] = b->t[j++];
        else {                                  /* same cell */
            int v = a->t[i].val + b->t[j].val;
            if (v) c->t[c->n++] = (Term){ a->t[i].row, a->t[i].col, v };
            i++; j++;
        }
    }
    return 1;
}
/* C = A * B using B^T: every row of A is dotted with every row of B^T,
 * walking two sorted lists (like merge). Cost O(rowsA * n_B + colsB * n_A ...). */
int sp_mul(const Sparse *a, const Sparse *b, Sparse *c)
{
    if (a->cols != b->rows) return 0;
    Sparse bt; sp_transpose_fast(b, &bt);
    c->rows = a->rows; c->cols = b->cols; c->n = 0;
    int ra = 0;
    while (ra < a->n) {
        int row = a->t[ra].row, rend = ra;
        while (rend < a->n && a->t[rend].row == row) rend++;      /* terms [ra,rend) */
        int cb = 0;
        while (cb < bt.n) {
            int col = bt.t[cb].row, cend = cb;                    /* row of B^T = col of B */
            while (cend < bt.n && bt.t[cend].row == col) cend++;
            int i = ra, j = cb, sum = 0;
            while (i < rend && j < cend) {
                if (a->t[i].col < bt.t[j].col) i++;
                else if (a->t[i].col > bt.t[j].col) j++;
                else sum += a->t[i++].val * bt.t[j++].val;
            }
            if (sum) { assert(c->n < MAXT); c->t[c->n++] = (Term){ row, col, sum }; }
            cb = cend;
        }
        ra = rend;
    }
    return 1;
}

5) المصفوفات الخاصة: مثلثية وtri-diagonal

لو المصفوفة ليها شكل ثابت (كل اللي فوق القطر أصفار)، مش محتاجين triplets أصلًا: بنعرف مكان كل خلية بمعادلة. Lower-triangular فيها n(n+1)/2 خلية، ونرقّمها صف بصف:

/* 0-based (i, j).  Row i of a lower triangle holds i+1 cells, so the rows
 * before it hold 0+1+...+i = i(i+1)/2 cells.                                 */
static long lower_idx(int i, int j)  { return (long)i * (i + 1) / 2 + j; }    /* needs j <= i */

/* upper triangle, row by row: row r holds n-r cells; rows before i hold
 * n + (n-1) + ... + (n-i+1) = i*n - i(i-1)/2                                  */
static long upper_idx(int n, int i, int j) { return (long)i * n - (long)i * (i - 1) / 2 + (j - i); } /* j >= i */

/* tri-diagonal: only |i-j| <= 1. Row 0 has 2 cells (slots 0,1); every later row
 * has 3, so row i starts at slot 2 + 3(i-1) = 3i-1  =>  slot(i,j) = 3i-1 + (j-i+1) = 2i+j. */
static long tri_idx(int i, int j) { return 2L * i + j; }                      /* j in {i-1,i,i+1} */

اشتقاق tri_idx: الصف 0 ليه خليتين (slots 0 و1)، وأي صف بعده ليه 3، فالصف i بيبدأ عند 2 + 3(i−1) = 3i−1، والخلية (i, j) بتبقى في 3i−1 + (j−i+1) = 2i + j. مجموع الخلايا 3n−2. والـ symmetric matrix بنخزّن نصها بس (lower) ونقرأ (i,j) كـ (j,i) لو i<j: n(n+1)/2 بدل n².

typedef struct { int n; int *a; } TriM;                 /* 3n-2 cells */
static TriM tri_new(int n) { TriM m = { n, calloc((size_t)(3 * n - 2), sizeof(int)) }; return m; }
static int  tri_get(const TriM *m, int i, int j) { return (j < i - 1 || j > i + 1) ? 0 : m->a[tri_idx(i, j)]; }
static void tri_set(TriM *m, int i, int j, int v) { assert(j >= i - 1 && j <= i + 1); m->a[tri_idx(i, j)] = v; }
