1) المصطلحات (Thareja 9.1، Pal 8.1، B&D 12.1)

الشجرة (tree) مجموعة عقد (nodes) فيها عقدة واحدة اسمها root والباقي موزّع على أشجار فرعية (sub-trees) منفصلة، كل واحدة تعريفها نفس التعريف (يعني التعريف recursive). الشجرة اللي فيها n عقدة ليها n − 1 edge بالظبط: كل عقدة غير الجذر ليها أب واحد، يعني حافة واحدة لفوق.

2) أنواع الـ binary trees وخصائصها

Binary tree: كل عقدة ليها على الأكتر ابنين، ومميّزين: شمال ويمين (فالشجرتين «A ابنها الشمال B» و«A ابنها اليمين B» مختلفتين). الأنواع (Thareja 9.2، Pal 8.2–8.7):

Completeكل المستويات ممتلئة ما عدا الأخير، والأخير ممتلئ من الشمالاللي بيتخزن في array من غير فراغات (الـ heap)

الخصائص (وكلها متحقّق منها في الكود)

  • أقصى عدد عقد في level r هو 2r (الجذر r = 0). فمستوى 3 فيه 8، ومستوى 4 فيه 16، ومستوى 12 فيه 4096.
  • الشجرة بارتفاع h (بعدد العقد) فيها على الأقل h وعلى الأكتر 2h − 1 عقدة.
  • الشجرة بـ n عقدة ارتفاعها على الأقل ⌈log₂(n+1)⌉ وعلى الأكتر n.
  • الـ complete tree بـ n عقدة ارتفاعها بالظبط ⌈log₂(n+1)⌉. مثال: 27 عقدة: المستويات 0–3 فيها 15، والأخير (level 4) فيه 12، الارتفاع 5.
  • في الـ extended tree اللي فيها n عقدة داخلية: عدد العقد الخارجية = n + 1، وE = I + 2n (E مجموع أطوال المسارات للخارجية وI للداخلية).
  • الـ full tree: عدد الأوراق = عدد العقد الداخلية + 1.
typedef struct Node {
    char lab[12];
    struct Node *l, *r;
} Node;

static Node *mk(const char *lab, Node *l, Node *r)
{
    Node *n = malloc(sizeof *n);
    size_t k = strlen(lab); if (k > sizeof n->lab - 1) k = sizeof n->lab - 1;     /* labels are short */
    memcpy(n->lab, lab, k); n->lab[k] = 0;
    n->l = l; n->r = r;
    return n;
}
static void free_tree(Node *t) { if (t) { free_tree(t->l); free_tree(t->r); free(t); } }
/* grammar:  tree := '.' | label | label '(' tree ',' tree ')'      e.g.  A(B(D,E),C(.,G))   */
static Node *parse_rec(const char **p)
{
    if (**p == '.') { (*p)++; return NULL; }
    char lab[12]; int k = 0;
    while (**p && **p != '(' && **p != ',' && **p != ')') lab[k++] = *(*p)++;
    lab[k] = 0;
    Node *n = mk(lab, NULL, NULL);
    if (**p == '(') {
        (*p)++; n->l = parse_rec(p);
        assert(**p == ','); (*p)++;
        n->r = parse_rec(p);
        assert(**p == ')'); (*p)++;
    }
    return n;
}
static Node *parse(const char *s) { const char *p = s; Node *t = parse_rec(&p); assert(*p == 0); return t; }

الـ parse بتخلّيك تكتب أي شكل من الكتاب في سطر: "A(B(D,E),C(F,G))" و. معناها فاضي. كل اختبارات الدرس بتستعملها.

/* depth = number of edges from the root (root has depth 0); -1 if absent */
static int depth_of(const Node *t, const char *lab, int d)
{
    if (!t) return -1;
    if (!strcmp(t->lab, lab)) return d;
    int x = depth_of(t->l, lab, d + 1);
    return x >= 0 ? x : depth_of(t->r, lab, d + 1);
}
/* ancestors of lab, written root-first into b; returns 1 if lab was found */
static int ancestors(const Node *t, const char *lab, Buf *b)
{
    if (!t) return 0;
    if (!strcmp(t->lab, lab)) return 1;
    Buf tmp = {{0}};
    if (ancestors(t->l, lab, &tmp) || ancestors(t->r, lab, &tmp)) {
        Buf mine = {{0}}; bput(&mine, t->lab);
        if (tmp.s[0]) { strcat(mine.s, " "); strcat(mine.s, tmp.s); }
        *b = mine; return 1;
    }
    return 0;
}

{
    if (!t) return NULL;
    if (!strcmp(t->lab, lab)) return t;
    const Node *x = find(t->l, lab);
    return x ? x : find(t->r, lab);
}
static const Node *parent_of(const Node *t, const char *lab)
{

/* complete: all levels full except possibly the last, which is filled left to right.
   full (strict): every node has 0 or 2 children. */
static int is_complete(const Node *t)
{
    const Node *q[256]; int f = 0, r = 0, gap = 0;
    q[r++] = t;
    while (f < r) {
        const Node *x = q[f++];
        if (!x) { gap = 1; continue; }
        if (gap) return 0;                       /* a real node after an empty slot */
        q[r++] = x->l; q[r++] = x->r;
    }
    return 1;
}
static int is_full(const Node *t)
{
    if (!t) return 1;
    if (!t->l && !t->r) return 1;
    if (!t->l || !t->r) return 0;
    return is_full(t->l) && is_full(t->r);
}

similar بتقول إن الشجرتين ليهم نفس الشكل (بغض النظر عن القيم)، وcopies بتقول نفس الشكل والمحتوى (Thareja بيسمّيهم similar وcopies، وده سؤال Fill-in رقم 6). is_complete بتعمل BFS وبتتأكد إن أول NULL مايجيش بعده عقدة حقيقية.

أ) التمثيل الرابط (linked) بثلاث arrays

Thareja 9.2.3: ثلاث arrays LEFT[] وDATA[] وRIGHT[] وROOT وAVAIL (رأس قايمة الخانات الفاضية). الخانة i هي «عقدة»؛ LEFT[i] وRIGHT[i] أرقام خانات (أو −1 لـ NULL). ده هو الـ pointer بس بـ index. تمرين Q14 و Q15 من الفصل بيطلب تحويل في الاتجاهين وهنحلّه.

/* Linked representation exactly as in Thareja Fig. 9.10/9.11:  arrays LEFT, DATA, RIGHT, a ROOT
   index and an AVAIL list threaded through the unused slots using the LEFT field. Index 0 unused,
   -1 plays the role of NULL. */
#define MEM 32
typedef struct {
    int left[MEM + 1], right[MEM + 1];
    char data[MEM + 1][12];
    int root, avail;
} Mem;

static void mem_init(Mem *m)
{
    memset(m, 0, sizeof *m);
    m->root = -1; m->avail = 1;
    for (int i = 1; i <= MEM; i++) { m->left[i] = i < MEM ? i + 1 : -1; m->right[i] = -1; }
}
static int mem_new(Mem *m, const char *lab, int l, int r)
{
    int i = m->avail;
    assert(i != -1);                              /* OVERFLOW */
    m->avail = m->left[i];
    strcpy(m->data[i], lab); m->left[i] = l; m->right[i] = r;
    return i;
}
static int mem_from_tree(Mem *m, const Node *t)
{
    if (!t) return -1;
    int l = mem_from_tree(m, t->l), r = mem_from_tree(m, t->r);   /* children first -> parent index larger */
    return mem_new(m, t->lab, l, r);
}
static Node *tree_from_mem(const Mem *m, int i)
{
    if (i == -1) return NULL;
    return mk(m->data[i], tree_from_mem(m, m->left[i]), tree_from_mem(m, m->right[i]));
}

ب) التمثيل المتتابع (sequential) في array واحدة

الجذر في TREE[1]. لو أب في K: الشمال في 2K واليمين في 2K+1، والأب لعقدة في K هو ⌊K/2⌋. مفيش pointers، ومحتاجين 2h − 1 خانة. مناسب للـ complete trees (مفيش فراغ) وسيء جدًا للـ skewed: شجرة بـ 20 عقدة chain ممكن تحتاج مليون خانة.

/* sequential representation: TREE[1] = root, children of K at 2K and 2K+1, "" = empty slot.
   returns the array size needed (2^h - 1 for height h) */
static void seq_fill(const Node *t, char out[][12], int k, int cap)
{
    if (!t || k > cap) return;
    strcpy(out[k], t->lab);
    seq_fill(t->l, out, 2 * k, cap);
    seq_fill(t->r, out, 2 * k + 1, cap);
}
لو الـ index بيبدأ من 0الأب في i: الشمال 2i+1 واليمين 2i+2. الكتاب (وكل الصيغ دي) بتبدأ من 1 عشان الحسبة تطلع أنضف.

4) الـ Traversals: أربع طرق

أي تجوال بيمر على كل عقدة مرة واحدة في O(n). التلاتة العميقة (depth-first) بتختلف في إمتى نزور الجذر بالنسبة لأبنائه، والرابع (level-order / breadth-first) بيمشي مستوى مستوى:

Pre-orderN ← L ← Rنسخ الشجرة، prefix من expression tree، listing الـ directories (Weiss)
In-orderL ← N ← Rفي BST بيطلّع المفاتيح مرتبة؛ infix من expression tree
Post-orderL ← R ← Nحذف الشجرة (الأبناء الأول)، postfix، حساب حجم الـ directory
Level-orderمستوى مستوى بـ queueإيجاد أقصر مسار، طباعة بالمستويات، فحص الـ complete
static void pre(const Node *t, Buf *b)  { if (t) { bput(b, t->lab); pre(t->l, b); pre(t->r, b); } }   /* N L R */
static void in(const Node *t, Buf *b)   { if (t) { in(t->l, b); bput(b, t->lab); in(t->r, b); } }     /* L N R */
static void post(const Node *t, Buf *b) { if (t) { post(t->l, b); post(t->r, b); bput(b, t->lab); } } /* L R N */

على الشجرة A(B(D(G,H(.,L)),E),C(.,F(I,J(K,.)))) (Fig 9.19(a) في Thareja، Examples 9.6–9.8): pre = A B D G H L E C F I J K، in = G D H L B E A C I F K J، post = G L H D E B I K J F C A، level = A B C D E F G H I J L K. البرنامج بيتأكد منهم بـ assert.

/* breadth-first: a queue of node pointers (array queue, enough for the sizes used here) */
static void level(const Node *t, Buf *b)
{
    const Node *q[256]; int f = 0, r = 0;
    if (t) q[r++] = t;
    while (f < r) {
        const Node *x = q[f++];

من غير recursion: stack صريح (Lipschutz 7.4، Thareja 9.3)

الـ recursion نفسها stack (راجع درس 43). لو الشجرة عميقة (skewed بمليون عقدة) الـ call stack هيطفح. النسخة الـ iterative: pre-order بتدفع اليمين وبعدين الشمال. in-order بتنزل شمال لأقصى حد وبتدفع في السكة. post-order أصعب لأن العقدة بتتزار بعد مرتين رجوع: الحل بـ last (آخر عقدة اتزارت) عشان نعرف جاي من الشمال ولا من اليمين.

/* Lipschutz 7.5: traversals with a stack, no recursion. */
#define STK 256
static void pre_iter(const Node *t, Buf *b)
{
    const Node *st[STK]; int sp = 0;
    if (t) st[sp++] = t;
    while (sp) {
        const Node *x = st[--sp];
        bput(b, x->lab);
        if (x->r) st[sp++] = x->r;           /* right first, so left comes out first */
        if (x->l) st[sp++] = x->l;
    }
}
static void in_iter(const Node *t, Buf *b)
{
    const Node *st[STK]; int sp = 0;
    while (t || sp) {
        while (t) { st[sp++] = t; t = t->l; }  /* go as far left as possible, remembering the way */
        t = st[--sp];
        bput(b, t->lab);
        t = t->r;
    }
}
static void post_iter(const Node *t, Buf *b)
{
    const Node *st[STK]; int sp = 0; const Node *last = NULL;
    while (t || sp) {
        while (t) { st[sp++] = t; t = t->l; }
        const Node *top = st[sp - 1];
        if (top->r && top->r != last) t = top->r;      /* right subtree not finished yet */
        else { bput(b, top->lab); last = top; sp--; }
    }
}

Morris: in-order بـ O(1) مساحة (من غير stack ومن غير recursion)

الفكرة (وهي أصل الـ threaded trees في الدرس الجاي): قبل ما ننزل شمال من عقدة cur، ندوّر على predecessor بتاعها (أقصى يمين في الشمال) ونخلّي right بتاعه يشاور مؤقتًا على cur. لما نرجع للعقدة عن طريق الـ thread نعرف إننا خلّصنا الشمال فنشيل الـ thread ونزور cur ونروح يمين. الشجرة بترجع زي ما كانت بالظبط، والبرنامج بيتأكد.

/* Morris traversal: O(1) extra space. It temporarily makes the in-order predecessor point
   back to the current node (a thread!), follows it, then removes it again. */
static void in_morris(Node *t, Buf *b)
{
    Node *cur = t;
    while (cur) {
        if (!cur->l) { bput(b, cur->lab); cur = cur->r; }
        else {
            Node *p = cur->l;
            while (p->r && p->r != cur) p = p->r;     /* in-order predecessor of cur */
            if (!p->r) { p->r = cur; cur = cur->l; }  /* create the thread */
            else { p->r = NULL; bput(b, cur->lab); cur = cur->r; }   /* remove it, visit */
        }
    }
}
ماتستعملش Morris مع threadsلأنها بتعدّل الشجرة مؤقتًا: لو فيه thread آخر بيقرأ الشجرة في نفس الوقت، أو استثناء نصّ الـ traversal، الشجرة هتفضل بـ links مقلوبة. للاستعمال المتزامن استعمل stack.

5) بناء الشجرة من traversalين

traversal واحد مش بيحدد الشجرة. اتنين بيحددوها في حالتين: (pre + in) أو (post + in). السبب: pre بيقولك مين الجذر (أول عنصر)، وin بيقولك مين في الشمال ومين في اليمين (اللي قبل الجذر وبعده). نكرر على الجزئين. (pre + post) مش كفاية لما فيه عقدة بابن واحد: ماقدرش أعرف الابن شمال ولا يمين.

/* Build from IN-ORDER + PRE-ORDER (arrays of labels). The first pre-order item is the root; its
   position in the in-order list splits left and right. */
static Node *from_pre_in(const char **pre, const char **in, int n, int *pi)
{
    if (n == 0) return NULL;
    const char *root = pre[(*pi)++];
    int k = 0; while (strcmp(in[k], root)) k++;
    Node *t = mk(root, NULL, NULL);
    t->l = from_pre_in(pre, in, k, pi);
    t->r = from_pre_in(pre, in + k + 1, n - k - 1, pi);
    return t;
}
/* IN-ORDER + POST-ORDER: the LAST post-order item is the root; build the RIGHT subtree first. */
static Node *from_post_in(const char **post, const char **in, int n, int *pi)
{
    if (n == 0) return NULL;
    const char *root = post[(*pi)--];
    int k = 0; while (strcmp(in[k], root)) k++;
    Node *t = mk(root, NULL, NULL);
    t->r = from_post_in(post, in + k + 1, n - k - 1, pi);
    t->l = from_post_in(post, in, k, pi);
    return t;
}

مثال (Thareja 9.4.5): in = D B E A F C G و pre = A B D E C F G بيطلّعوا شجرة جذرها A و post بتاعها D E B F G C A. ولو اديتها post = D E B F G C A مع نفس الـ in بتطلع نفس الشجرة. البرنامج بيعمل الاتنين ويقارنهم.

6) شجرة التعبير (Expression Tree) — Thareja 9.4، Weiss 4.2.2

الأوراق = معاملات والعقد الداخلية = عمليات. القراءة: in-order = infix (بتحتاج أقواس)، pre-order = prefix، post-order = postfix. والبناء من postfix بـ stack: معامل → اعمل له عقدة وادفعها. عملية → pop اتنين (الأول هو اليمين) وركّب عقدة ليهم وادفعها. ده بالظبط Weiss 4.2.2.

/* POSTFIX:  operand -> push a leaf;  operator -> pop right, pop left, push the new node. */
static Node *from_postfix(const char *s)
{
    Node *st[64]; int sp = 0;
    char buf[256]; strncpy(buf, s, 255); buf[255] = 0;
    for (char *tok = strtok(buf, " "); tok; tok = strtok(NULL, " ")) {
        if (is_op(tok)) {
            Node *r = st[--sp], *l = st[--sp];
            st[sp++] = mk(tok, l, r);
        } else st[sp++] = mk(tok, NULL, NULL);
    }
    assert(sp == 1);
    return st[0];
}
/* PREFIX: read an operator, then build its two subtrees recursively. */
static Node *from_prefix_rec(char **toks, int *i, int n)
{
    assert(*i < n);
    const char *t = toks[(*i)++];
    if (!is_op(t)) return mk(t, NULL, NULL);
    Node *l = from_prefix_rec(toks, i, n);
    Node *r = from_prefix_rec(toks, i, n);
    return mk(t, l, r);
}
static Node *from_prefix(const char *s)
{
    char buf[256]; strncpy(buf, s, 255); buf[255] = 0;
    char *toks[64]; int n = 0;
    for (char *tok = strtok(buf, " "); tok; tok = strtok(NULL, " ")) { toks[n] = malloc(strlen(tok) + 1); strcpy(toks[n++], tok); }
    int i = 0; Node *t = from_prefix_rec(toks, &i, n);
    assert(i == n);
    for (int k = 0; k < n; k++) free(toks[k]);
    return t;
}

/* recursive-descent parser for single-token operands (letters or numbers) and ( ).
     expr   := term   { (+|-) term }
     term   := factor { (*|/|%) factor }
     factor := atom [ ^ factor ]            <- right recursion = right associativity
     atom   := name | number | '(' expr ')'                                                         */
static const char *ip;
static Node *p_expr(void);
static Node *p_atom(void)
{
    while (*ip == ' ') ip++;
    if (*ip == '(') { ip++; Node *e = p_expr(); while (*ip == ' ') ip++; assert(*ip == ')'); ip++; return e; }
    char lab[12]; int k = 0;
    while (isalnum((unsigned char)*ip)) lab[k++] = *ip++;
    assert(k > 0); lab[k] = 0;
    return mk(lab, NULL, NULL);
}
static Node *p_factor(void)
{
    Node *a = p_atom();
    while (*ip == ' ') ip++;
    if (*ip == '^') { ip++; Node *b = p_factor(); return mk("^", a, b); }
    return a;
}
static Node *p_term(void)
{
    Node *a = p_factor();
    for (;;) {
        while (*ip == ' ') ip++;
        if (*ip == '*' || *ip == '/' || *ip == '%') { char op[2] = { *ip++, 0 }; a = mk(op, a, p_factor()); }
        else return a;
    }
}
static Node *p_expr(void)
{
    Node *a = p_term();
    for (;;) {
        while (*ip == ' ') ip++;
        if (*ip == '+' || *ip == '-') { char op[2] = { *ip++, 0 }; a = mk(op, a, p_term()); }
        else return a;
    }
}
static Node *from_infix(const char *s) { ip = s; Node *t = p_expr(); while (*ip == ' ') ip++; assert(*ip == 0); return t; }
/* fully parenthesised infix (every operator node gets brackets) */
static void infix_full(const Node *t, char *out)
{
    if (!is_op(t->lab) || !t->l) { strcat(out, t->lab); return; }
    strcat(out, "("); infix_full(t->l, out); strcat(out, t->lab); infix_full(t->r, out); strcat(out, ")");
}
/* minimal parentheses: a child needs brackets only if it binds weaker than its parent, or equally
   on the "wrong" side (left side of ^, right side of - / % and friends) */
static void infix_min(const Node *t, char *out)
{
    if (!is_op(t->lab) || !t->l) { strcat(out, t->lab); return; }
    char p = t->lab[0];
    const Node *kids[2] = { t->l, t->r };
    for (int k = 0; k < 2; k++) {
        const Node *c = kids[k];
        int need = 0;
        if (is_op(c->lab) && c->l) {
            char q = c->lab[0];
            if (prec(q) < prec(p)) need = 1;
            else if (prec(q) == prec(p)) need = (p == '^') ? (k == 0) : (k == 1);
        }
        if (need) strcat(out, "(");
        infix_min(c, out);
        if (need) strcat(out, ")");
        if (k == 0) strcat(out, t->lab);
    }
}

typedef struct { const char *name; double v; } Bind;
static double lookup(const Bind *b, int nb, const char *name)
{
    if (isdigit((unsigned char)name[0])) return atof(name);
    for (int i = 0; i < nb; i++) if (!strcmp(b[i].name, name)) return b[i].v;
    assert(!"unbound variable");
    return 0;
}
static double eval_tree(const Node *t, const Bind *b, int nb)
{
    if (!t->l) return lookup(b, nb, t->lab);
    double x = eval_tree(t->l, b, nb), y = eval_tree(t->r, b, nb);
    switch (t->lab[0]) {
        case '+': return x + y;
        case '-': return x - y;
        case '*': return x * y;
        case '/': return x / y;
        case '%': return fmod(x, y);
        default:  return pow(x, y);
    }
}

7) كام شجرة ممكنة؟ والـ general trees

عدد الأشكال المختلفة لشجرة ثنائية بـ n عقدة هو عدد كاتالان C(n) = (2n)! / ((n+1)! n!): 1, 1, 2, 5, 14, 42, 132 … (والصيغة: C(n) = Σ C(i)·C(n−1−i) لأن كل عقدة جذر بتقسم الباقي لشمال ويمين). مع 3 عقد: 5 أشكال؛ مع 4: 14. الكود بيعدّها وبيطبعها.

/* number of structurally different binary trees with n nodes = Catalan(n) */
static long catalan(int n)
{
    long c[20] = {1};
    for (int i = 1; i <= n; i++) { c[i] = 0; for (int j = 0; j < i; j++) c[i] += c[j] * c[i - 1 - j]; }
    return c[n];
}
/* list all shapes with n nodes into out[] (each written like  x(x(.,.),.)  ) */
static int shapes(int n, char out[][64])
{
    if (n == 0) { strcpy(out[0], "."); return 1; }
    int cnt = 0;
    for (int k = 0; k < n; k++) {
        char left[16][64], right[16][64];
        int a = shapes(k, left), b = shapes(n - 1 - k, right);
        for (int i = 0; i < a; i++)
            for (int j = 0; j < b; j++) {
                if (!strcmp(left[i], ".") && !strcmp(right[j], ".")) strcpy(out[cnt], "x");
                else {
                    char tmp[160];                                  /* 2 children (< 64 each) + "x(" "," ")" fit easily */
                    snprintf(tmp, sizeof tmp, "x(%.60s,%.60s)", left[i], right[j]);
                    assert(strlen(tmp) < 64);                       /* n <= 4 is all we ever list */
                    strcpy(out[cnt], tmp);
                }
                cnt++;
            }
    }
    return cnt;
}

الشجرة العامة (General tree) → شجرة ثنائية

عقدة ممكن تبقى ليها أي عدد أبناء. بدل ما نحجز array pointers، كل عقدة بتحتفظ بـ أول ابن وبـ الأخ التالي (first-child / next-sibling، Weiss 4.1.1). ده نفسه التحويل لشجرة ثنائية: ابن أول = left، أخ تالي = right. والـ forest بتتحوّل بنفس الطريقة (الجذور إخوات).

/* A general tree (any number of children) stored left-child / right-sibling. The SAME two pointers
   then ARE a binary tree:  left = first child,  right = next sibling   (Thareja 9.3). */
static Node *gparse_list(const char **p);
static Node *gparse_one(const char **p)
{
    char lab[12]; int k = 0;
    while (**p && **p != '(' && **p != ',' && **p != ')') lab[k++] = *(*p)++;
    lab[k] = 0;
    Node *n = mk(lab, NULL, NULL);
    if (**p == '(') { (*p)++; n->l = gparse_list(p); assert(**p == ')'); (*p)++; }
    return n;
}
static Node *gparse_list(const char **p)           /* a,b,c  ->  a --right--> b --right--> c */
{
    Node *first = gparse_one(p), *cur = first;
    while (**p == ',') { (*p)++; cur->r = gparse_one(p); cur = cur->r; }
    return first;
}
/* "A(B(E,F),C,D(G))" : A has children B,C,D;  B has E,F;  D has G.  returns the binary tree */
static Node *general_to_binary(const char *s) { const char *p = s; Node *t = gparse_one(&p); assert(!*p); return t; }

typedef struct TNode {
    char name[24];
    int own;                        /* the file's own size in blocks (0 for a plain directory) */
    struct TNode *first_child, *next_sibling;
} TNode;

static TNode *tn(const char *name, int own)
{
    TNode *t = calloc(1, sizeof *t);
    assert(strlen(name) < sizeof t->name);
    strcpy(t->name, name); t->own = own;
    return t;
}
static TNode *add_child(TNode *parent, TNode *c)          /* append at the END of the sibling chain, O(#children) */
{
    if (!parent->first_child) parent->first_child = c;
    else { TNode *s = parent->first_child; while (s->next_sibling) s = s->next_sibling; s->next_sibling = c; }
    return c;
}
static void tfree(TNode *t) { if (t) { tfree(t->first_child); tfree(t->next_sibling); free(t); } }

/* PRE-order with indentation (Weiss Fig 4.6): print the node, THEN its children one tab deeper.
   Total work is O(N): every node is visited once. */
static void list_all(const TNode *t, int depth, char *out)
{
    for (int i = 0; i < depth; i++) strcat(out, "  ");
    strcat(out, t->name); strcat(out, "\n");
    for (const TNode *c = t->first_child; c; c = c->next_sibling) list_all(c, depth + 1, out);
}
/* POST-order (Weiss Fig 4.9): a directory's size is known only AFTER all its children's sizes are known. */
static int size_of(const TNode *t, char *out)
{
    int s = t->own;
    for (const TNode *c = t->first_child; c; c = c->next_sibling) s += size_of(c, out);
    char line[64]; snprintf(line, sizeof line, "%-10s %4d\n", t->name, s); strcat(out, line);
    return s;
}
static int count_nodes(const TNode *t) { int n = 1; for (const TNode *c = t->first_child; c; c = c->next_sibling) n += count_nodes(c); return n; }
static int tree_height(const TNode *t)                     /* nodes on the longest path (single node = 1) */
{
    int h = 0;
    for (const TNode *c = t->first_child; c; c = c->next_sibling) { int x = tree_height(c); if (x > h) h = x; }
    return h + 1;
}

Weighted path length

لو للأوراق (العقد الخارجية) أوزان wi وعمق di: WEPL = Σ wi·di. الشجرة اللي تقلّلها لنفس مجموعة الأوزان هي شجرة Huffman. (Thareja Q16: الأشجار T1 وT2 وT3 بتطلع 77 و49 و36.)

/* In an EXTENDED binary tree every leaf is an "external node". Depth of root = 0. */
static void paths(const Node *t, int d, int *E, int *I, long *W, int leafnum)
{
    if (!t) return;
    if (!t->l && !t->r) { *E += d; if (leafnum) *W += (long)atoi(t->lab) * d; return; }
    *I += d;
    paths(t->l, d + 1, E, I, W, leafnum);
    paths(t->r, d + 1, E, I, W, leafnum);
}

خوارزمية Huffman

عايزين كود لكل حرف: prefix-free (ولا كود بداية كود تاني، فنفك التشفير من غير فواصل) وأقصر ما يمكن للحروف الكتيرة. الخوارزمية الجشعة (greedy): ابدأ بغابة أوراق بأوزانها، وكرّر: خد أصغر شجرتين وادمجهم تحت جذر جديد وزنه المجموع (الأصغر شمال)، لحد ما تبقى شجرة واحدة. الكود لكل حرف = مسار الجذر للورقة (0 شمال، 1 يمين). إثبات الأمثلية: أصغر وزنين لازم يبقوا إخوات في أعمق مستوى في أي شجرة مثالية (لو ماكانوش نبدّلهم مع اللي هناك ومفيش خسارة).

typedef struct HNode {
    int w, ch;                     /* weight, character (or -1 for an internal node) */
    struct HNode *l, *r;
} HNode;

/* The "priority queue" is a plain array: remove_min scans for the smallest weight and closes the gap
   by shifting, so items keep their age order and ties always go to the OLDER item (repeatable results).
   O(n^2) overall - perfect for 26 letters and easy to read; a heap would make it O(n log n). */
static HNode *remove_min(HNode **q, int *m)
{
    int a = 0;
    for (int i = 1; i < *m; i++) if (q[i]->w < q[a]->w) a = i;      /* strict <  keeps the first (oldest) */
    HNode *x = q[a];
    memmove(q + a, q + a + 1, (size_t)(*m - a - 1) * sizeof *q);
    (*m)--;
    return x;
}
static HNode *huffman(const int *w, const int *ch, int n)
{
    HNode *q[256]; int m = n;
    for (int i = 0; i < n; i++) {
        q[i] = malloc(sizeof(HNode));
        q[i]->w = w[i]; q[i]->ch = ch[i]; q[i]->l = q[i]->r = NULL;
    }
    while (m > 1) {
        HNode *x = remove_min(q, &m), *y = remove_min(q, &m);
        HNode *z = malloc(sizeof(HNode));
        z->w = x->w + y->w; z->ch = -1; z->l = x; z->r = y;
        q[m++] = z;                                                 /* newest goes to the back */
    }
    return q[0];
}
static void hfree(HNode *t) { if (t) { hfree(t->l); hfree(t->r); free(t); } }

/* walk the tree: left edge = '0', right edge = '1' */
static void codes(const HNode *t, char *buf, int len, char table[256][32])
{
    if (!t->l && !t->r) { buf[len] = 0; strcpy(table[t->ch], len ? buf : "0"); return; }
    buf[len] = '0'; codes(t->l, buf, len + 1, table);
    buf[len] = '1'; codes(t->r, buf, len + 1, table);
}
static int hwepl(const HNode *t, int d)
{
    if (!t->l && !t->r) return t->w * d;
    return hwepl(t->l, d + 1) + hwepl(t->r, d + 1);
}
static int encode(const char *text, char table[256][32], char *bits)
{
    bits[0] = 0;
    for (const char *p = text; *p; p++) strcat(bits, table[(unsigned char)*p]);
    return (int)strlen(bits);
}
static void decode(const HNode *root, const char *bits, char *out)
{
    const HNode *t = root; int k = 0;
    for (const char *p = bits; *p; p++) {
        t = (*p == '0') ? t->l : t->r;
        if (!t->l && !t->r) { out[k++] = (char)t->ch; t = root; }
    }
    out[k] = 0;
}

Tournament (winner) trees

شجرة complete فيها الأوراق هي اللاعبين وكل عقدة داخلية هي الفائز من ابنيها: الجذر هو الأصغر في O(n) بنا. لما نغيّر قيمة ورقة نعيد المباريات على مسارها بس: O(log n). ده أساس الـ replacement selection في الـ external sorting.

/* Tournament (winner) tree for n = 2^k players: leaves at T[n..2n-1], T[i] = winner of T[2i], T[2i+1].
   Here the "winner" is the SMALLER value (a selection tree used for merging runs). */
static void tour_build(int *T, const int *v, int n)
{
    for (int i = 0; i < n; i++) T[n + i] = v[i];
    for (int i = n - 1; i >= 1; i--) T[i] = T[2 * i] < T[2 * i + 1] ? T[2 * i] : T[2 * i + 1];
}
/* replace leaf i by x and replay only the matches on the way to the root: O(log n) */
static void tour_replay(int *T, int n, int i, int x)
{
    T[n + i] = x;
    for (int k = (n + i) / 2; k >= 1; k /= 2) T[k] = T[2 * k] < T[2 * k + 1] ? T[2 * k] : T[2 * k + 1];
}
